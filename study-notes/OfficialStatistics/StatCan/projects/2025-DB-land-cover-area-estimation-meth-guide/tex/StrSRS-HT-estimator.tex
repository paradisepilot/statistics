\textbf{Coefficient of variation (CV) estimator:}
\begin{eqnarray*}
\widehat{\textnormal{CV}}\!\left(\,\widehat{A^{(k)}_{{\color{black}r}}}^{\textnormal{HT}}\,\right)
& = &
	\left.
		\sqrt{{\color{white}.}\Var\!\left(\,\widehat{A^{(k)}_{{\color{black}r}}}^{\textnormal{HT}}\,\right){\color{white}.}}
		\right/
	\widehat{A^{(k)}_{{\color{black}r}}}^{\textnormal{HT}}
\end{eqnarray*}

\begin{frame}{\vskip -0.5cm \large HT \& variance estimators for an ecozone, say, $z = r_{1} \cup r_{2} \cup r_{3}$}

\scriptsize

\begin{eqnarray*}
\widehat{A^{(k)}_{{\color{red}z}}}^{\textnormal{HT}}
& \!\! = &
	\widehat{A^{(k)}_{{\color{red}r_{1}}}}^{\textnormal{HT}}
	\,+\,
	\widehat{A^{(k)}_{{\color{red}r_{2}}}}^{\textnormal{HT}}
	\,+\,
	\widehat{A^{(k)}_{{\color{red}r_{3}}}}^{\textnormal{HT}}
\;\; = \quad
	\overset{K}{\underset{j=1}{\sum}}\;
	M_{{\color{red}r_{1}},j}
	\cdot
	\overline{y}^{(k)}_{{\color{red}r_{1}},j}
	\;+\;
	\overset{K}{\underset{j=1}{\sum}}\;
	M_{{\color{red}r_{2}},j}
	\cdot
	\overline{y}^{(k)}_{{\color{red}r_{2}},j}
	\;+\;
	\overset{K}{\underset{j=1}{\sum}}\;
	M_{{\color{red}r_{3}},j}
	\cdot
	\overline{y}^{(k)}_{{\color{red}r_{3}},j}
\\
& \textnormal{\LARGE\color{white}1} &
\\
\widehat{\Var}\!\left(\,\widehat{A^{(k)}_{{\color{red}z}}}^{\textnormal{HT}}\,\right)
& = &
	\widehat{\Var}\!\left(\,\widehat{A^{(k)}_{{\color{red}r_{1}}}}^{\textnormal{HT}}\,\right)
	\,+\,
	\widehat{\Var}\!\left(\,\widehat{A^{(k)}_{{\color{red}r_{2}}}}^{\textnormal{HT}}\,\right)
	\,+\,
	\widehat{\Var}\!\left(\,\widehat{A^{(k)}_{{\color{red}r_{3}}}}^{\textnormal{HT}}\,\right)
\\
& \!\! = \!\! &
	\textnormal{\tiny$
	\overset{K}{\underset{j=1}{\sum}}
	\left(\,1 \,-\, \dfrac{m_{{\color{red}r_{1}},j}}{M_{{\color{red}r_{1}},j}}\,\right)
	\cdot
	M_{{\color{red}r_{1}},j}^{2}
	\cdot
	\dfrac{s_{{\color{red}r_{1}},j}^{2}}{m_{{\color{red}r_{1}},j}}
	\;+\;
	\overset{K}{\underset{j=1}{\sum}}
	\left(\,1 \,-\, \dfrac{m_{{\color{red}r_{2}},j}}{M_{{\color{red}r_{2}},j}}\,\right)
	\cdot
	M_{{\color{red}r_{2}},j}^{2}
	\cdot
	\dfrac{s_{{\color{red}r_{2}},j}^{2}}{m_{{\color{red}r_{2}},j}}
	\;+\;
	\overset{K}{\underset{j=1}{\sum}}
	\left(\,1 \,-\, \dfrac{m_{{\color{red}r_{3}},j}}{M_{{\color{red}r_{3}},j}}\,\right)
	\cdot
	M_{{\color{red}r_{3}},j}^{2}
	\cdot
	\dfrac{s_{{\color{red}r_{3}},j}^{2}}{m_{{\color{red}r_{3}},j}}
	$}
\\
& \textnormal{\LARGE\color{white}1} &
\\
\widehat{\textnormal{CV}}\!\left(\,\widehat{A^{(k)}_{{\color{red}z}}}^{\textnormal{HT}}\,\right)
%\;\; = \;\;
%	\dfrac{
%		\sqrt{{\color{white}.}\Var\!\left(\,\widehat{A^{(k)}_{{\color{red}z}}}^{\textnormal{HT}}\,\right){\color{white}.}}
%		}{
%		\widehat{A^{(k)}_{{\color{red}z}}}^{\textnormal{HT}}
%		}
& = &
	\left.
		\sqrt{
			{\color{white}.}
			\widehat{\Var}\!\left(\,\widehat{A^{(k)}_{{\color{red}z}}}^{\textnormal{HT}}\,\right)
			{\color{white}.}
			}
		\right/
	\widehat{A^{(k)}_{{\color{red}z}}}^{\textnormal{HT}}
\end{eqnarray*}

\end{frame}
