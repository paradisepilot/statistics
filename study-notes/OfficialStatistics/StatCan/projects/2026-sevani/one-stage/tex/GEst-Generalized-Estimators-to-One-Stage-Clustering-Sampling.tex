Statistics Canada's Integrated Business Survey Program (IBSP)
uses a three-level hierarchy of ``units'':
\begin{itemize}
\item
    sampling units
    \begin{itemize}
    \item
        The sampling design is defined at the level of sampling units.
    \item
        Each sampling unit is composed of one or more imputation units.
    \end{itemize}
\item
    imputation units
    \begin{itemize}
    \item
        Collection and imputation is performed at the level of imputation units.
    \item
        Each imputation unit is composed of one or more estimation units.
    \item
        It is assumed that the sampling design is {\color{red}one-stage cluster sampling},
        in the sense that, the sample is a probability sample of \textit{sampling units}, and
        for each selected \textit{sampling unit},
        all of its constituent \textit{imputation units} are selected.
        In standard cluster sampling terminology,
        the \textit{sampling units} are called the \textit{primary sampling units} (PSU),
        while the \textit{imputation units} are called the \textit{secondary sampling units} (SSU).
    \end{itemize}
\item
    estimation units
    \begin{itemize}
    \item
        {\color{red}Domain membership indicators} are defined at the level of estimation units.
    \end{itemize}
\end{itemize}

\noindent
G-Est is a Statistics Canada generalized system that computes (among other functionalities)
point and variance estimates (from probability sample data) for population parameters of the following form:

\vskip 0.3cm
\begin{definition}[Generalized total]
\mbox{}
\vskip 0.1cm
\noindent
A \textbf{generalized total} is a population parameter of the following form:
\begin{equation*}
T_{\alpha Y}
\;\; := \;\;
    \underset{i \in U}{\sum}\;\;
    \underset{j \in \mathcal{C}_{i}}{\sum}\;
    \underset{k \in \mathcal{E}_{ij}}{\sum}\;
    \alpha_{ijk}Y_{ijk}\,,
\end{equation*}
where
\begin{itemize}
\item
    $U$\, is the population of sampling units (i.e., primary sampling units),
\item
    $\mathcal{C}_{i}$,\, for \,$i \in U$,\,
    is the collection of imputation units (i.e., secondary sampling units)
    belonging to sampling unit $i \in U$,
\item
    $\mathcal{E}_{ij}$,\, for \,$i \in U$\, and \,$j \in \mathcal{C}_{i}$,\,
    is the collection of estimation units
    belonging to imputation unit \,$j \in \mathcal{C}_{i}$\, of sampling unit $i \in U$,
\item
    $Y_{ijk} \in \Re$\, is the value
    of an ($\Re$-valued) population characteristic \,$\mathbf{Y}$\,
    defined at the level of the estimation units; more precisely,
    \,$Y_{ijk}$\, is the value of \,$\mathbf{Y}$\,
    for estimation unit \,$k \in \mathcal{E}_{ij}$\,
    of imputation unit \,$j \in \mathcal{C}_{i}$\,
    belonging to sampling unit \,$i \in U$,\,
\item
    $\alpha_{ijk} \in \Re$\, is the coefficient defined at the level of the estimation unit.
\end{itemize}
\end{definition}

\textbf{Sampling variance component:}
\vskip 0.1cm
\begin{lemma}[Unbiased estimator for sampling variance component; Equation (5.5) in \cite{Beaumont2011}]
\label{DomainTotalSamplingVariance}
\mbox{}
\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{DomainTotalFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{DomainTotalProbabilisticFramework}.
\vskip 0.1cm
\noindent
Then, the following statements are true:
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\mbox{}\;\;\,(\theenumi)$\quad$}
\begin{enumerate}
\item
    If
    \;$v(\,s\,;\abullet\ydot\,)$\,
    is an (arbitrary) design-unbiased estimator for the sampling variance, i.e.,
    \begin{equation*}
    E_{S}\!\left[\;
        v(\,s\,;\abullet\Ydot\,)
        \,\left\vert\;
        \Ydot,\Xobs
        \right.
        \,\right]
    \;\; = \;\;
        \Var_{S}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \,\right\vert\,
            \Ydot,
            \Xobs
            \,\right),
    \end{equation*}
    then
    \,$
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\abullet\Ydot\,)
        \left\vert\,
        s,s_{R},\YsubR,\Xobs
        \right.
        \right]
    $\,
    is an $(\Sdot,\Rdot,\YsubR)$-unbiased estimator for
    \,$
    E_{Y}\!\left[\,
        \Var_{S}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \,\right\vert\,
            \Ydot,
            \Xobs
            \,\right)
    \,\right]
    $,\,
    i.e.,
    \begin{equation*}
    E_{SR}\,
    E_{Y_{R}}\,
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\abullet\Ydot\,)
        \,\left\vert\;
        s,s_{R},\YsubR,\Xobs
        \right.
        \,\right]
    \;\; = \;\;
        E_{Y}\!\left[\,
            \Var_{S}\!\left(\,
                \left.
                \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
                \,\right\vert\,
                \Ydot,
                \Xobs
                \,\right)
            \,\right]
    \end{equation*}
\item
    If
    \,$v(\,s\,;\abullet\ydot\,)$\,
    is taken to be the (design-unbiased) Horvitz-Thompson estimator for the sampling variance, i.e.,
    \begin{equation*}
    v(\,s\,;\abullet\ydot\,)
    \;\; := \;\;
        \underset{k \in s}{\sum}\,
        \underset{l \in s}{\sum}\,
        \dfrac{\pi_{kl} - \pi_{k}\pi_{l}}{\pi_{kl}}\,
        (w_{k}d_{k}y_{k})\,(w_{l}d_{l}y_{l})
    \end{equation*}

    \,$
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\dbullet\Ydot\,)
        \left\vert\,
        s,s_{R},\YsubR,\Xobs
        \right.
        \right]

    \begin{equation*}
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\dbullet\Ydot\,)
        \left\vert\,
        s,s_{R},\YsubR,\Xobs
        \right.
        \right]

        \;+\;
        \underset{i \in s}{\sum}\,
        \underset{j\in(\mathcal{C}_{i})_{M}}{\sum}
        \dfrac{1 - \pi_{i}}{\pi_{i}^{2}}\;

    \end{equation*}
\end{enumerate}
\end{lemma}
\proof
\begin{eqnarray*}
&&

        v(\,s\,;\abullet\Ydot\,)
        \,\left\vert\!
        \begin{array}{c} s,s_{R},\YsubR, \\ \Xobs \end{array}
        \right.
        \!\!\right]
\;\; = \;\;
    E_{Y_{M}}\!\left[\,
        \left.

        \underset{k \in s}{\sum}\,

        \left.

        \underset{k \in s}{\sum}\,

        \dfrac{1}{\pi_{i}}

        \underset{k \in s}{\sum}\,

        \,\right\vert\!\!

        \!\!\right]

    \underset{j\in\mathcal{C}_{i}}{\sum}\;
    \underset{k \in s}{\sum}\,
    \underset{l\in\mathcal{C}_{k}}{\sum}\,
    \dfrac{\pi_{ik} - \pi_{i}\pi_{k}}{\pi_{ik}}\,
    \dfrac{\alpha_{ij}}{\pi_{i}}
    \,
    \dfrac{\alpha_{kl}}{\pi_{k}}

    E_{Y_{M}}\!\left[\,
        Y_{ij}Y_{kl}
        \left\vert\!\!
        \begin{array}{c} s,s_{R},\YsubR, \\ \Xobs \end{array}
        \right.
        \!\!\right]
\\
& = &
    \underset{i \in s}{\sum}\,
    \underset{j\in\mathcal{C}_{i}}{\sum}\;
    \underset{k \in s}{\sum}\,
    \underset{l\in\mathcal{C}_{k}}{\sum}\,

    \dfrac{\alpha_{kl}}{\pi_{k}}

        Y_{ij}Y_{kl}
        \left\vert\!\!
        \begin{array}{c} \YsubR, \\ \Xobs \end{array}
        \right.
        \!\!\right]
\\
& = &
    \cdots

    v(\,s\,;\abullet\widetilde{y}_{\bullet}\,)

    \underset{j\in(\mathcal{C}_{i})_{M}}{\sum}
    \dfrac{1 - \pi_{i}}{\pi_{i}^{2}}\;
    \alpha_{ij}^{2}\,\sigma_{ij}^{2}\,,
\end{eqnarray*}
where the last equality follows from the observation:
\begin{eqnarray*}
E_{Y_{M}}\!\left[\,
    Y_{ij}Y_{kl}
    \left\vert\!\!
    \begin{array}{c} \YsubR, \\ \Xobs \end{array}
    \right.
    \!\!\right]
& = &
    \left\{\begin{array}{cl}
        y_{ij} \cdot y_{kl}\,, &
        \underset{{\color{white}.}}
        {\textnormal{for \,$j \in (\mathcal{C}_{i})_{R}$, \,$l \in (\mathcal{C}_{k})_{R}$}}
        \\
        y_{ij} \cdot E[\,Y_{kl}\,\vert\,\Xobs\,]\,, &
        \underset{{\color{white}.}}
        {\textnormal{for \,$j \in (\mathcal{C}_{i})_{R}$, \,$l \in (\mathcal{C}_{k})_{M}$}}
        \\
        E[\,Y_{ij}\,\vert\,\Xobs\,] \cdot y_{kl}\,, &
        \underset{{\color{white}.}}
        {\textnormal{for \,$j \in (\mathcal{C}_{i})_{M}$, \,$l \in (\mathcal{C}_{k})_{R}$}}
        \\
        E[\,Y_{ij}\,\vert\,\Xobs\,] \cdot E[\,Y_{kl}\,\vert\,\Xobs\,]\,, &
        \underset{{\color{white}.}}
        {\textnormal{for \,$(i,j) \neq (k,l)$,\, $j \in (\mathcal{C}_{i})_{M}$, \,$l \in (\mathcal{C}_{k})_{M}$}}
        % {\textnormal{for \,$k,l \in s_{M}$,\, $k \neq l$}}
        \\
        \Var(\,Y_{ij}\,\vert\,\Xobs) + E[\,Y_{ij}\,\vert\,\Xobs\,]^{2}\,, &
        \underset{{\color{white}.}}
        {\textnormal{for \,$(i,j) = (k,l)$,\, $j \in (\mathcal{C}_{i})_{M}$}}
        \end{array}\right.
\end{eqnarray*}
which is equivalent to the following more compact expression:
\begin{eqnarray*}
E_{Y_{M}}\!\left[\,
    Y_{ij}Y_{kl}
    \left\vert\!\!
    \begin{array}{c} \YsubR, \\ \Xobs \end{array}
    \right.
    \!\!\right]
& = &
    \left\{\begin{array}{cl}
        \widetilde{y}_{ij}^{2} \,+\, \sigma_{ij}^{2}\,, &
        \textnormal{for \,$(i,j) = (k,l)$,\, $j \in (\mathcal{C}_{i})_{M}$}
        \\
        \widetilde{y}_{ij} \cdot \widetilde{y}_{kl}\,, & \overset{{\color{white}1}}{\textnormal{otherwise}}
        \end{array}\right.
\end{eqnarray*}

\begin{lemma}[Unbiased estimator for nonresponse variance; Equations (5.9) \& (5.10) in \cite{Beaumont2011}]
\label{DomainTotalNonresponseVariance}
\mbox{}
\vskip 0.1cm

\ref{DomainTotalProbabilisticFramework}.

\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\mbox{}\;\;\,(\theenumi)$\quad$}
\begin{enumerate}
\item
    $

        \right]
    $.
\item

        \left.

        \,\right\vert\!\!
        \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}

    \begin{eqnarray*}
    E_{Y}\!\left[

        \,\right\vert\!\!
        \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
        \!\!\right]
    & = &
        \left(\;
            \underset{i \in s}{\sum}\,
            \underset{j \in (\mathcal{C}_{i})_{R}}{\sum}\!
            \left(\,W^{(+)}_{\alpha,ij}\,\right)^{\!2}

            \sigma^{2}_{ij}

\subsection{Formulation of the target variability quantification problem}
\label{DomainTotalFormulationOfVarianceEstimationProblem}
\setcounter{theorem}{0}

Suppose:
\begin{itemize}
\item
    We would like to estimate
    -- using data from a probability sample --
    a domain total
    \begin{equation*}
    T_{d,Y}
    \;\; := \;\;
        \underset{i \in U}{\sum}\;d_{i}Y_{i}\,,
    \end{equation*}
    where \,$U = \{\,1,2,\ldots,N\,\} \subset \N$\, represents the population,
    $\dbullet = (d_{1},\ldots,d_{N}) \in \{0,1\}^{N}$ is the vector of domain membership indicators,
    and
    \,$\Ydot = (Y_{1},\ldots,Y_{N})$\,
    is an $\Re$-valued population characteristic.
\item
    Aside/Reminder:
    \vskip 0.01cm
    \noindent
    If there were no missing data in \,$\ydot$\, for the units in the selected sample,
    then the \textit{Horvitz-Thompson domain total estimator\! \footnote{
        We will use the alternative notations
        \,$\widehat{T}_{d,\xi}^{\textnormal{(HT)}}$\,
        and
        \,$\widehat{T}^{\textnormal{(HT)}}(s\,;\dbullet\xi_{\bullet})$\,
        interchangeably, noting that the former is notationally more succinct,
        while the latter allows us to emphasize that the population characteristic
        \,$\xi_{\bullet}$\,
        is being considered as an argument to the formula
        \begin{equation*}
        \widehat{T}^{\textnormal{(HT)}}(s\,;\dbullet\xi_{\bullet})
        \;\; := \;\;
            \underset{i\,\in\,s}{\sum}\;w_{i}\,d_{i}\,\xi_{i}
        \end{equation*}
        for the Horvitz-Thompson population total estimator.
        }
    }
    \begin{eqnarray*}
    \widehat{T}^{\textnormal{(HT)}}_{d,Y}
    & :=: &
        % \widehat{T}^{\textnormal{(HT)}}(s\,;\dydot)
        \widehat{T}^{\textnormal{(HT)}}(\,\Sdot,\Ydot\,;\dbullet\,)
    \;\; := \;\;
        \underset{i\,\in\,s}{\sum}\;w_{i}\,d_{i}\,Y_{i}\,,
    \end{eqnarray*}
    is a design-unbiased estimator for
    \,$T_{d,Y}$,\,
    where
    \begin{itemize}
    \item
        $s \subset U$\, is the (random) selected sample, and
        \,$\Sdot \,=\, (S_{1},\ldots,S_{N})^{T}$\,
        is the vector of $\{0,1\}$-valued selection indicator variables,
    \item
        $w_{i} \,:=\, \dfrac{1}{\pi_{i}}$,\, for each population unit \,$i \in U$,\,
        is the (\textit{a priori} known, non-random) design weight of \,$i \in U$,\,
        i.e., the reciprocal of the selection probability \,$\pi_{i}$\,
        (assumed strictly positive) of \,$i \in U$.\,
    \end{itemize}
    Recall (Result 2.8.1, p.43, \cite{Sarndal1992}) also that
    the sampling variance of 
    \,$
    \widehat{T}^{\textnormal{(HT)}}_{d,\xi}
    % \,=\,
    %     \widehat{T}^{\textnormal{(HT)}}(s\,;\dbullet\xi_{\bullet})
    $\,
    is given by
    \begin{equation*}
    \Var_{S}\!\left(\,
        \left.
        \widehat{T}^{\textnormal{(HT)}}_{d,\xi}
        \,\right\vert\!
        \begin{array}{c} \xi_{\bullet} \end{array}
        \!\!\right)
    \;\; = \;\;\,
        \underset{i \in U}{\sum}\,
        \underset{j \in U}{\sum}\;
        \dfrac{\pi_{ij} - \pi_{i}\pi_{j}}{\pi_{ij}}
        \cdot
        \dfrac{d_{i}\,\xi_{i}}{\pi_{i}}
        \cdot
        \dfrac{d_{j}\,\xi_{j}}{\pi_{j}}\,,
    \end{equation*}
    and the following Horvitz-Thompson variance estimator
    \begin{equation*}
    \widehat{V}^{\textnormal{(HT)}}(\,s\,;\dbullet\xi_{\bullet}\,)
    \;\; := \;\;
        \underset{i \in s}{\sum}\,
        \underset{j \in s}{\sum}\;
        \dfrac{\pi_{ij} - \pi_{i}\pi_{j}}{\pi_{ij}}
        \cdot
        \dfrac{d_{i}\,\xi_{i}}{\pi_{i}}
        \cdot
        \dfrac{d_{j}\,\xi_{j}}{\pi_{j}}
    \end{equation*}
    is a design-unbiased estimator for the sampling variance of
    % \,$\widehat{T}_{d,\xi_{\bullet}}(s) \,=\, \widehat{T}_{d}(s\,;\xi_{\bullet})$,\,
    \,$
    \widehat{T}^{\textnormal{(HT)}}_{d,\xi_{\bullet}}
    \,=\,
        \widehat{T}^{\textnormal{(HT)}}(s\,;\dbullet\xi_{\bullet})
    $,\,
    i.e.,
    \begin{equation*}
    E_{S}\!\left[\;
        \left.
        \widehat{V}^{\textnormal{(HT)}}(\,s\,;\dbullet\xi_{\bullet}\,)
        \;\right\vert\!
        \begin{array}{c} \xi_{\bullet} \end{array}
        \!\right]
    \;\; = \;\;
        \Var_{S}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{d,\xi} %(s\,;\dbullet\xi_{\bullet})
            \;\right\vert\!
            \begin{array}{c} \xi_{\bullet} \end{array}
            \!\!\right)
    \end{equation*}
\item
    The value of \,$Y_{l}$\, has been successfully observed
    only for units \,$l$\, in the (random) ``responding'' subsample
    \,$s_{R} \subset s \subset U$;\,
    in other words, the measurement of \,$Y_{k}$\, somehow failed for each unit \,$k$\,
    in the ``nonresponding'' subsample
    \,$s_{M} \, := \,s \,\backslash\, s_{R} \,\subset\, U$.\,
\item
    We would like to estimate \,$T_{d,Y}$\,
    with an imputation estimator of the following form:
    \begin{eqnarray*}
    \widetilde{T}_{d,Y}
    & = &
        \widetilde{T}_{d,Y}(\,
            \Sdot\,,\,
            \Rdot\,,\,
            \Ydot\,;\,
            \Xdotdot\,,\,
            \dbullet
            \,)
    \;\; = \;\;
        \underset{i\,\in\,s}{\sum}\;w_{i}\,d_{i}\,Y^{\dagger}_{i}
    \;\; = \;\;
        \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,d_{i}\left(\,
            R_{i}\,Y_{i}
            \overset{{\color{white}.}}{+}
            (1-R_{i})\,Y^{*}_{i}
            \,\right),
    \end{eqnarray*}
    where
    \begin{itemize}
    \item
        $\Rdot \,=\, (R_{1},\ldots,R_{N})^{T}$ is the vector of $\{0,1\}$-valued response indicator variables,
    \item
        $\Xdotdot \,=\, \left(\,X_{ij}\,\right) \,\in\, \overline{\Re}^{N \times r}$\,
        is a random matrix containing the values of the $\Re$predictor variables for $\Ydot$,
        where the $i^{\textnormal{th}}$ row of \,$\Xdotdot$\, corresponds to populaton unit \,$i \in U$,\,
        and where \,$\overline{R} \,:=\, \Re \sqcup \{\,\texttt{NA}\,\}$\,
        is an extension of $\Re$ alllowing for missing values in \,$\Xdotdot$,\, and
    \item
        for each \,$i \in s$,\,
        \,$Y^{\dagger}_{i} \;=\; R_{i}\,Y_{i}+(1-R_{i})\,Y^{*}_{i}$,\,
        and
        \begin{equation*}
        Y^{*}_{k}
        \;\; = \;\;
            Y^{*}_{k}(\,\Sdot,\Rdot,\Ydot,\Xdotdot\,)\,,
        \quad
        \textnormal{for each \,$k \in s_{M} = s\,\backslash\,s_{R}$},
        \end{equation*}
        is an imputed value for
        \,$Y_{k}$\, and \,$Y^{*}_{k}$\,
        may depend on
        \,$\Sdot,\Rdot,\Ydot,\Xdotdot$.\,
    \end{itemize}
\item
    Furthermore, \,$Y^{*}_{k}$\, ($k \in s_{M}$) is \textbf{composite-linear},
    i.e., there exists \,$J \in \N$\, such that
    \begin{center}
    \begin{minipage}{6.2in}
    for each admissible sample \,$s \subset U$,\,
    each \,$s_{R} \subset s$,\,
    {\color{red}each} \,{\color{red}$k \in s_{M} := s \,\backslash\, s_{R}$},\,
    there exist
    \begin{equation*}
    \left\{\;\;\begin{array}{ccl}
    j
    & = &
        j(\,k,s,s_{R}\,;\Xobs\,) \;\in\; \{1,2,\ldots,J\},
    \\
    \varphi^{(j)}_{0k}

    \end{equation*}
    such that
    \begin{equation*}
    Y_{{\color{red}k}}^{*}
    \;\; = \;\;
        Y^{*}_{k}(\,\Sdot,\Rdot,\Ydot,\Xdotdot\,)
    \;\; = \;\;
        \varphi^{(j)}_{0{\color{red}k}}

\vskip 1.0cm
\subsection{Probabilistic framework of SEVANI \cite{Beaumont2011}}
\label{DomainTotalProbabilisticFramework}
\setcounter{theorem}{0}

        %%%%% ~~~~~~~~~~~~~~~~~~~~ %%%%%

\noindent
\textbf{Probability spaces}
\begin{itemize}
\item
    Let
    \,$U \,=\, \{1,2,\ldots,N\}$.\,
    Let
    \,$\mathcal{P}(U) \,=\, \mathcal{P}(\{1,2,\ldots,N\})$\,
    denote the power set (set of all subsets) of \,$U$.\,
    Let
    \,$\left(\,\mathcal{P}(U)\,,\mathcal{P}(\mathcal{P}(U))\,,\overset{{\color{white}-}}{\pi}\,\right)$\,
    be the probability space induced by a sampling design, i.e.,
    a probability mass function defined on
    \,$\mathcal{P}(U)$.\,
\item
    Let \,$\left(\,\Omega,\mathscr{A},\lambda\,\right)$\, be a probability space.
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Random variables}
\begin{itemize}
\item
    $\Ydot \,=\, (Y_{1},Y_{2},\ldots,Y_{N})^{T} : \Omega \longrightarrow \Re^{N \times 1}$\,
    is an $\Re^{N \times 1}$-valued random variable defined on \,$\Omega$,\,
    i.e., $\mathscr{A}$-measurable map.
\item
    $\Xdotdot\,=\,\left(\,\overset{{\color{white}.}}{X_{ij}}\,\right) : \Omega \longrightarrow \Re^{N \times r}$\,
    is an $\Re^{N \times r}$-valued random variable defined on \,$\Omega$.
\item
    $\Sdot \,=\, (S_{1},S_{2},\ldots,S_{N})^{T} : \mathcal{P}(U) \longrightarrow \{0,1\}^{N \times 1}$\,
    is a $\{0,1\}^{N \times 1}$-valued random variable defined on \,$\mathcal{P}(U)$,\,
    i.e., $\mathcal{P}(\mathcal{P}(U))$-measurable map.
\item
    $\Rdot \,=\, (R_{1},R_{2},\ldots,R_{N})^{T} : {\color{red}\Omega \times \mathcal{P}(U)} \longrightarrow \{0,1\}^{N \times 1}$\,
    is a $\{0,1\}^{N \times 1}$-valued random variable defined on \,$\Omega \times \mathcal{P}(U)$,\,
    i.e., $\left(\mathscr{A} \overset{{\color{white}.}}{\otimes} \mathcal{P}(\mathcal{P}(U))\right)$-measurable map
    that satisfies:
    \begin{equation*}
    (\,\lambda\otimes\pi\,)\!\left(\;\,
        \underset{k \in U}{\bigcup}\left\{\;
            (\omega,s) \in \Omega\times\mathcal{P}(U)
            \;\left\vert
            \begin{array}{c}
            S_{k}(s) = 0, %\,\textnormal{and}\;
            \\
            \overset{{\color{white}.}}{R_{k}(\omega,s) = 1}
            \end{array}
            \right.
            \!\!\right\}
        \;\right)
    \; = \;
        0
    \end{equation*}
    Note that this condition stipulates that,
    whenver a population unit \,$k \in U$\, is not selected in the (random) sample \,$s \subset U$,\,
    then it is not a respondent either.
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Interpretation}
\begin{itemize}
\item
    $U = \{1,2,\ldots,N\}$
    \,--\,
    index set of the target (finite) population of interest,
    and \,$N$\, is the (finite) size of the target population.
\item
    $\left(\,\Omega,\mathscr{A},\lambda\,\right)$
    \,--\,
    underlying source of stochasticity of the imputation model (or superpopulation model).
\item
    $\Ydot$
    \,--\,
    $\Re$-valued target variable (population characteristic).
\item
    $\Xdotdot$
    \,--\,
    $\Re^{r}$-valued auxiliary variable (to be used as predictor variables for $\Ydot$). 
\item
    $\Sdot$
    \,--\,
    $\{0,1\}$-valued binary variable indicating, for each \,$s \subset U$,\,
    whether a population unit \,$k \in U$\, is selected in \,$s$.\,
    More precisely, for each \,$s \subset U$,\, we have:
    \begin{equation*}
    S_{k}(s)
    \;\; = \;\;
        \left\{\begin{array}{cl}
            1\,, & \textnormal{if \,$k \in s$},
            \\
            0\,, & \textnormal{otherwise}.
            \end{array}\right.
    \end{equation*}
\item
    $\Rdot$
    \,--\,
    $\{0,1\}$-valued binary variable indicating, for almost every outcome
    \,$\omega \in \Omega$\,
    and each
    \,$(s,s_{R}) \in \mathcal{P}(U) \times \mathcal{P}(U)$\,
    satisfying
    \,$s_{R} \subset s$,\,
    whether a selected population unit \,$ k \in s$\,
    is a respondent or not.
    % More precisely, for each \,$s \subset U$ and each \,$k \in s$,\, we have:
    % \begin{equation*}
    % R_{k}(\omega,s_{R})
    % \;\; = \;\;
    %     \left\{\begin{array}{cl}
    %         1\,, & \textnormal{if \,$k \in s_{R}$},
    %         \\
    %         0\,, & \textnormal{otherwise}.
    %         \end{array}\right.
    % \end{equation*}
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Probabilistic assumptions}
\begin{itemize}
\item
    The unit-level selection probabilities are all strictly positive, i.e.,
    \begin{equation*}
    \pi_{i}
    \; := \;
        P(\,S_{i} = 1\,)
    \; := \;
        \pi\!\left(\,\left\{\,
            \left.
            s \overset{{\color{white}.}}{\subset} U
            \,\right\vert\,
            i \in s
            \,\right\}
        \,\right)
    \;\; > \;\;
        0,\,
    \textnormal{for each \,$i \in U$\,
    % and \,$w_{i} \,:=\, \dfrac{1}{P(\,S_{i} = 1\,)}$,\, for each \,$i \in U$.
    }
    \end{equation*}
\item
    The second-order joint selection probabilities are all strictly positive, i.e.,
    \begin{equation*}
    \pi_{ij}
    \; := \;
        P(\,S_{i} = S_{j} = 1\,)
    \; := \;
        \pi\!\left(\,\left\{\,
            \left.
            s \overset{{\color{white}.}}{\subset} U
            \,\right\vert\,
            i, j \in s
            \,\right\}
        \,\right)
    \;\; > \;\;
        0,\,
    \textnormal{for each \,$i,j \in U$,\, with $i \neq j$}
    \end{equation*}
\item
    $\left(\,\Ydot\,,\Xdotdot\,\right)$\, and \,$\Sdot$\, are (unconditionally) independent.
\item
    $\Ydot$\, and \,$(\,\Sdot\,,\Rdot\,)$\, are {\color{red}conditionally independent} given \,$\Xdotdot$.
\item
    $\Cov\!\left(\;Y_{k}\,,\,Y_{l}\;\left\vert\;\Xdotdot\right.\right) \,=\, 0$,\,
    for each \,$k, l \in U$\, with \,$k \neq l$.
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{(Non-random) parameters}
\begin{itemize}
\item
    $\dbullet \,=\, (d_{1},d_{2},\ldots,d_{N}) \in \{0,1\}^{N}$\,
    is a (non-random, binary) domain membership indicator.
\end{itemize}

\vskip 0.1cm
\begin{lemma}[A formula for \,$\widetilde{T}_{d,Y}$]
\mbox{}
\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{DomainTotalFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{DomainTotalProbabilisticFramework}.
\vskip 0.1cm
\noindent
Then, we have:
\begin{eqnarray*}
\widetilde{T}_{d,Y}
& = &
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,d_{i}\,Y_{i}
    \;\overset{{\color{white}.}}{+}\;
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,(1-R_{i})\,d_{i}\,(Y^{*}_{i}-Y_{i})
\end{eqnarray*}
\end{lemma}
\proof
\begin{eqnarray*}
\widetilde{T}_{d,Y}
& := &
    \underset{i\,\in\,s}{\sum}\;w_{i}\,d_{i}\,\widetilde{Y}_{i}
\;\; = \;\;
    \underset{i\,\in\,s}{\sum}\;w_{i}d_{i}\left(\,
        R_{i}\,Y_{i}
        \overset{{\color{white}.}}{+}
        (1-R_{i})\,Y^{*}_{i}
        \,\right)
\\
& = &
    \underset{i\,\in\,s}{\sum}\;w_{i}\,R_{i}\,d_{i}\,Y_{i}
    \;\overset{{\color{white}.}}{+}\;
    \underset{i\,\in\,s}{\sum}\;w_{i}\,(1-R_{i})\,d_{i}\,Y^{*}_{i}
\;\; = \;\;
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,R_{i}\,d_{i}\,Y_{i}
    \;\overset{{\color{white}.}}{+}\;
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,(1-R_{i})\,d_{i}\,Y^{*}_{i}
\\
& = &
    {\color{blue}
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,d_{i}\,Y_{i}
    \;\overset{{\color{white}.}}{-}\;
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,d_{i}\,Y_{i}
    }
    \;\overset{{\color{white}.}}{+}\;
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,R_{i}\,d_{i}\,Y_{i}
    \;\overset{{\color{white}.}}{+}\;
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,(1-R_{i})\,d_{i}\,Y^{*}_{i}
\\
& = &
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,d_{i}\,Y_{i}
    \;\overset{{\color{white}.}}{-}\;
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,(1-R_{i})\,d_{i}\,Y_{i}
    \;\overset{{\color{white}.}}{+}\;
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,(1-R_{i})\,d_{i}\,Y^{*}_{i}
\\
& = &
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,d_{i}\,Y_{i}
    \;\overset{{\color{white}.}}{+}\;
    \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,(1-R_{i})\,d_{i}\,(Y^{*}_{i}-Y_{i})
\end{eqnarray*}
\vskip -0.5cm
\qed

\vskip 0.3cm
\begin{lemma}[Conditional bias of \,$\widetilde{T}_{d,Y}$; see Equation (5.2) in \cite{Beaumont2011}]
\label{biasTtilde}
\mbox{}
\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{DomainTotalFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{DomainTotalProbabilisticFramework}.
\vskip 0.1cm
\noindent
Then, the following statements are true:
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\mbox{}\;\;\,(\theenumi)$\quad$}
\begin{enumerate}
\item
    The \textbf{overall bias}
    \,$
    \bias\!\left(\;
        \left.
        \widetilde{T}_{d,Y}
        \;\right\vert\,
        \Xobs
        \,\right)
    $\,
    of
    \,$\widetilde{T}_{d,Y}$\,
    satisfies:
    \begin{eqnarray*}
    \bias\!\left(\;
        \left.
        \widetilde{T}_{d,Y}
        \;\right\vert\,
        \Xobs
        \,\right)
    & := &
        E\!\left[\;
            \left.
            \widetilde{T}_{d,Y} - T_{d,Y}
            \,\right\vert\,
            \Xobs
            \,\right]
    \;\; = \;\;
        E_{SR}\!\left(\;
            \overset{{\color{white}.}}{E_{Y}\!\left[\;
                \left.
                \widetilde{T}_{d,Y} \overset{{\color{white}.}}{-} \widehat{T}^{\textnormal{(HT)}}_{d,Y}
                \;\right\vert\,
                s,s_{R},\Xobs
                \,\right]}
            \;\right),
    \end{eqnarray*}
    where
    \,$
    \widehat{T}^{\textnormal{(HT)}}_{d,Y}
    \; = \;
        \widehat{T}^{\textnormal{(HT)}}(\,s\,;\dbullet\Ydot\,)
    \; = \;
        \underset{k\,\in\,s}{\sum}\;d_{k}\,w_{k}\,Y_{k}
    $\,
    is the Horvitz-Thompson estimator for the domain total.
\item
    The \textbf{conditional model bias}
    \,$
    E_{Y}\!\left[\;
        \left.
        \widetilde{T}_{d,Y} \overset{{\color{white}.}}{-} \widehat{T}^{\textnormal{(HT)}}_{d,Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    $\,
    can be expressed as follows:
    \begin{eqnarray*}
    E_{Y}\!\left[\;
        \left.
        \widetilde{T}_{d,Y} \overset{{\color{white}.}}{-} \widehat{T}^{\textnormal{(HT)}}_{d,Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    & = &
        \underset{i\,\in\,{\color{red}s_{M}}}{\sum}\,d_{i}\,w_{i}\,
        E_{Y}\!\left[\;
            \left.
            Y^{*}_{i} \overset{{\color{white}.}}{-} Y_{i}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \end{eqnarray*}
\item
    Consequently, we have the following alternative expression for the overall bias:
    \begin{eqnarray*}
    \bias\!\left(\;
        \left.
        \widetilde{T}_{d,Y}
        \;\right\vert\,
        \Xobs
        \,\right)
    & := &
        E\!\left[\;
            \left.
            \widetilde{T}_{d,Y} - T_{d,Y}
            \,\right\vert\,
            \Xobs
            \,\right]

        E_{SR}\!\left(\;
            \overset{{\color{white}.}}{E_{Y}\!\left[\;
                \left.
                \widetilde{T}_{d,Y} \overset{{\color{white}.}}{-} \widehat{T}^{\textnormal{(HT)}}_{d,Y}
                \;\right\vert\,
                s,s_{R},\Xobs
                \,\right]}
            \;\right)

    \end{eqnarray*}
\end{enumerate}
\end{lemma}

\begin{lemma}[Unbiased estimator for sampling variance component; Equation (5.5) in \cite{Beaumont2011}]
\label{DomainTotalSamplingVariance}
\mbox{}
\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{DomainTotalFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{DomainTotalProbabilisticFramework}.
\vskip 0.1cm
\noindent
Then, the following statements are true:
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\mbox{}\;\;\,(\theenumi)$\quad$}
\begin{enumerate}
\item
    If
    \;$v(\,s\,;y_{\bullet}\,)$\,
    is an (arbitrary) design-unbiased estimator for the sampling variance, i.e.,
    \begin{equation*}
    E_{S}\!\left[\;
        v(\,s\,;\Ydot\,)
        \,\left\vert\;
        \Ydot,\Xobs
        \right.
        \,\right]
    \;\; = \;\;
        \Var_{S}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,\right\vert\,
            \Ydot,
            \Xobs
            \,\right),
    \end{equation*}
    then
    \,$
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\Ydot\,)
        \left\vert\,
        s,s_{R},\YsubR,\Xobs
        \right.
        \right]
    $\,
    is an $(\Sdot,\Rdot,\YsubR)$-unbiased estimator for
    \,$
    E_{Y}\!\left[\,
        \Var_{S}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,\right\vert\,
            \Ydot,
            \Xobs
            \,\right)
    \,\right]
    $,\,
    i.e.,
    \begin{equation*}
    E_{SR}\,
    E_{Y_{R}}\,
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\Ydot\,)
        \,\left\vert\;
        s,s_{R},\YsubR,\Xobs
        \right.
        \,\right]
    \;\; = \;\;
        E_{Y}\!\left[\,
            \Var_{S}\!\left(\,
                \left.
                \widehat{T}^{\textnormal{(HT)}}_{d,Y}
                \,\right\vert\,
                \Ydot,
                \Xobs
                \,\right)
            \,\right]
    \end{equation*}
\item
    If
    \,$v(\,s\,;\dydot\,)$\,
    is taken to be the (design-unbiased) Horvitz-Thompson estimator for the sampling variance, i.e.,
    \begin{equation*}
    v(\,s\,;\dydot\,)
    \;\; := \;\;
        \underset{k \in s}{\sum}\,
        \underset{l \in s}{\sum}\,
        \dfrac{\pi_{kl} - \pi_{k}\pi_{l}}{\pi_{kl}}\,
        (w_{k}d_{k}y_{k})\,(w_{l}d_{l}y_{l})
    \end{equation*}
    then
    \,$
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\dbullet\Ydot\,)
        \left\vert\,
        s,s_{R},\YsubR,\Xobs
        \right.
        \right]
    $\,
    can be expressed as follows:
    \begin{equation*}
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\dbullet\Ydot\,)
        \left\vert\,
        s,s_{R},\YsubR,\Xobs
        \right.
        \right]
    \;\; = \;\;
        v(\,s\,;\,\dbullet\widetilde{y}_{\bullet}\,)
        \;+\;
        \underset{k \in s_{M}}{\sum}\;
        \left(1-\dfrac{1}{w_{k}}\right)w_{k}^{2}\,d_{k}^{2}\,\sigma_{k}^{2}
    \end{equation*}
\end{enumerate}
\end{lemma}
\proof
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\mbox{}\;\;\,(\theenumi)$\quad$}
\begin{enumerate}
\item
    We simply compute:
    \begin{eqnarray*}
    E_{SR}\,
    E_{Y_{R}}\,
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\dbullet\Ydot\,)
        \,\left\vert\;
        s,s_{R},\YsubR,\Xobs
        \right.
        \,\right]
    & = &
        E_{SR}\,
        E_{Y}\!\left[\,
            v(\,s\,;\dbullet\Ydot\,)
            \,\left\vert\;
            s,s_{R},\Xobs
            \right.
            \,\right]
    \\
    & = &
        E_{Y}\,
        E_{SR}\!\left[\,
            v(\,s\,;\dbullet\Ydot\,)
            \,\left\vert\;
            \Ydot,\Xobs
            \right.
            \,\right]
    \\
    & = &
        E_{Y}\,
        E_{S}\!\left[\,
            v(\,s\,;\dbullet\Ydot\,)
            \,\left\vert\;
            \Ydot,\Xobs
            \right.
            \,\right]
    \\
    & = &
        E_{Y}\!\left[\,
            \Var_{S}\!\left(\,
                \left.
                \widehat{T}^{\textnormal{(HT)}}_{d,Y}
                \,\right\vert\,
                \Ydot,
                \Xobs
                \,\right)
            \,\right],
    \;\;
    \textnormal{by hypothesis on \,$v(\,s\,;\dydot\,)$}
    \end{eqnarray*}
\item
    We compute:
    \begin{eqnarray*}
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\dbullet\Ydot\,)
        \left\vert\,
        s,s_{R},\YsubR,\Xobs
        \right.
        \right]
    & = &
        E_{Y_{M}}\!\left[\,
            \left.
            \underset{k \in s}{\sum}\,
            \underset{l \in s}{\sum}\,
            \dfrac{\pi_{kl} - \pi_{k}\pi_{l}}{\pi_{kl}}\,
            (d_{k}w_{k}Y_{k})\,(d_{l}w_{l}Y_{l})
            \;\right\vert\,
            s,s_{R},\YsubR,\Xobs
            \right]
    \\
    & = &
        \underset{k \in s}{\sum}\,
        \underset{l \in s}{\sum}\;
        \dfrac{\pi_{kl} - \pi_{k}\pi_{l}}{\pi_{kl}}\;
        d_{k}w_{k}\,d_{l}w_{l}\;
        E_{Y_{M}}\!\left[\;
            Y_{k}Y_{l}
            \,\left\vert\,
            s,s_{R},\YsubR,\Xobs
            \right.
            \right]
    \\
    & = &
        \underset{k \in s}{\sum}\,
        \underset{l \in s}{\sum}\;
        \dfrac{\pi_{kl} - \pi_{k}\pi_{l}}{\pi_{kl}}\;
        d_{k}w_{k}\,d_{l}w_{l}\;
        E_{Y_{M}}\!\left[\;
            Y_{k}Y_{l}
            \,\left\vert\,
            \YsubR,\Xobs
            \right.
            \right]
    \\
    & = &
        \cdots
        \;\; = \;\;
        v(\,s\,;\dbullet\widetilde{y}_{\bullet}\,)
        \;+\;
        \underset{k \in s_{M}}{\sum}\!
        \left(1-\dfrac{1}{w_{k}}\right)
        w_{k}^{2}\,d_{k}^{2}\,\sigma_{k}^{2}\,,
    \end{eqnarray*}
    where the last equality follows from the observation:
    \begin{eqnarray*}
    E_{Y_{M}}\!\left[\;
        Y_{k}Y_{l}
        \,\left\vert\,
        \YsubR,\Xobs
        \right.
        \right]
    & = &
        \left\{\begin{array}{cl}
            y_{k} \cdot y_{l}\,, & \underset{{\color{white}.}}{\textnormal{for \,$k,l \in s_{R}$}}
            \\
            y_{k} \cdot E[\,Y_{l}\,\vert\,\Xobs\,]\,, & \underset{{\color{white}.}}{\textnormal{for \,$k \in s_{R}$,\, $l \in s_{M}$}}
            \\
            E[\,Y_{k}\vert\,\Xobs\,] \cdot y_{l}\,, & \underset{{\color{white}.}}{\textnormal{for \,$k \in s_{M}$,\, $l \in s_{R}$}}
            \\
            E[\,Y_{k}\vert\,\Xobs\,] \cdot E[\,Y_{l}\vert\,\Xobs\,]\,, & \underset{{\color{white}.}}{\textnormal{for \,$k,l \in s_{M}$,\, $k \neq l$}}
            \\
            \Var(\,Y_{k}\,\vert\,\Xobs) + E[\,Y_{k}\,\vert\,\Xobs\,]^{2}\,, & \textnormal{for \,$k = l \in s_{M}$}
            \end{array}\right.
    \end{eqnarray*}
    which is equivalent to the following more compact expression:
    \begin{eqnarray*}
    E_{Y_{M}}\!\left[\;
        Y_{k}Y_{l}
        \,\left\vert\,
        \YsubR,\Xobs
        \right.
        \right]
    & = &
        \left\{\begin{array}{cl}
            \widetilde{y}_{k}^{2} \,+\, \sigma_{k}^{2}\,, & \textnormal{for \,$k = l \in s_{M}$}
            \\
            \widetilde{y}_{k} \cdot \widetilde{y}_{l}\,, & \overset{{\color{white}.}}{\textnormal{otherwise}}
            \end{array}\right.
    \end{eqnarray*}
    \vskip -0.7cm
    \qed
\end{enumerate}

\subsection{$\widetilde{T}_{d,Y} \,-\, \widehat{T}^{\textnormal{(HT)}}_{d,Y}$
as sum of \,$\Xobs$-conditionally independent summands; compare Equation (5.8) in \cite{Beaumont2011}}

\begin{lemma}[$\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}$ as a sum of \,$\Xobs$-conditionally independent summands]
\label{DomainTotalImputationError}
\mbox{}
\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{DomainTotalFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{DomainTotalProbabilisticFramework}.
\vskip 0.1cm
\noindent
Then, the following equalities hold:
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\mbox{}\;\;\,(\theenumi)$\quad$}
\begin{enumerate}
\item
    $
    \widetilde{T}_{d,Y}
    \;\; = \;\;
        \widetilde{T}_{d,Y}(\,s,s_{R},\Ydot\,;\Xobs,\dbullet\,)
    \;\; = \;\;
        W^{(+)}_{d,0}
        \;+\;
        \underset{l \in s_{R}}{\sum}\!\left(\,d_{l}\,w_{l} \,+\, W^{(+)}_{d,l}\,\right)Y_{l}
    $
\item
    $
    \widetilde{T}_{d,Y}
    \, - \,
    \widehat{T}^{\textnormal{(HT)}}_{d,Y}
    \;\; = \;\;
        W^{(+)}_{d,0}
        \;+\;
        \underset{l \in s_{R}}{\sum}W^{(+)}_{d,l}\,Y_{l}
        \;\;-\;
        \underset{k \in s_{M}}{\sum}w_{k}\,d_{k}\,Y_{k}
    $,\\
    where
    \,$
    \widehat{T}^{\textnormal{(HT)}}_{d,Y}
    \; = \;
    \widehat{T}^{\textnormal{(HT)}}(\,s\,;\dYdot\,)
    \; = \;
        \underset{k \in s}{\sum}\;w_{k}\,d_{k}\,Y_{k}
    $\,
    is the Horvitz-Thompson estimator.
\end{enumerate}
\end{lemma}
\proof
\begin{enumerate}
\item
    Straightforward algebra:
    \begin{eqnarray*}
    \widetilde{T}_{d,Y}
    & = &
        \widetilde{T}_{d,Y}\!\left(\,s,s_{R},\Ydot\,;\,\Xobs,\dbullet\,\right)
    \;\; = \;\;
        \underset{l \in s_{R}}{\sum}\;w_{l}\,d_{l}\,Y_{l}
        \,+\,
        \underset{k \in s_{M}}{\sum}\;w_{k}\,d_{k}\,Y_{k}^{*}
    \;\; = \;\;
        \underset{l \in s_{R}}{\sum}\;w_{l}\,d_{l}\,Y_{l}
        \,+\,
        \overset{J}{\underset{j=1}{\sum}}\;
        \underset{k \in s_{M}^{(j)}}{\sum}\,
        w_{k}\,d_{k}\,Y_{k}^{*}
    \\
    & = &
        \underset{l \in s_{R}}{\sum}\;w_{l}\,d_{l}\,Y_{l}
        \,+\,
        \overset{J}{\underset{j=1}{\sum}}\;
        \underset{k \in s_{M}^{(j)}}{\sum}\,
        w_{k}\,d_{k}\left(\,
            \varphi^{(j)}_{0k}
            \,+\,
            \underset{l \in s_{R}}{\sum}\,\varphi^{(j)}_{lk}\,Y_{l}
            \right)
    \\
    & = &
        \left(\;
            \overset{J}{\underset{j=1}{\sum}}\;
            \underset{k \in s_{M}^{(j)}}{\sum}\;
            w_{k}\,d_{k}\,\varphi^{(j)}_{0k}
            \,\right)
        \,+\,
        \left(\;
            \underset{l \in s_{R}}{\sum}\;w_{l}\,d_{l}\,Y_{l}
            \,+\,
            \overset{J}{\underset{j=1}{\sum}}\;
            \underset{k \in s_{M}^{(j)}}{\sum}\,
            w_{k}\,d_{k}\left(\,
                \underset{l \in s_{R}}{\sum}\,\varphi^{(j)}_{lk}\,Y_{l}
                \right)
            \,\right)
    \\
    & = &
        \left(\;
            \overset{J}{\underset{j=1}{\sum}}\;
            \underset{k \in s_{M}^{(j)}}{\sum}\;
            w_{k}\,d_{k}\,\varphi^{(j)}_{0k}
            \,\right)
        \,+\,
        \left(\;
            \underset{l \in s_{R}}{\sum}\;w_{l}\,d_{l}\,Y_{l}
            \,+\,
            \underset{l \in s_{R}}{\sum}\!\left(\;
                \overset{J}{\underset{j=1}{\sum}}\;
                \underset{k \in s_{M}^{(j)}}{\sum}\,w_{k}\,d_{k}\,\varphi^{(j)}_{lk}
                \right)
            Y_{l}
            \right)
    \\
    & = &
        \left(\;
            \overset{J}{\underset{j=1}{\sum}}\;
            \underset{k \in s_{M}^{(j)}}{\sum}\;
            w_{k}\,d_{k}\,\varphi^{(j)}_{0k}
            \,\right)
        \,+\,
        \underset{l \in s_{R}}{\sum}
        \left(\;
            w_{l}\,d_{l}
            \,+\,
            \left(\;
                \overset{J}{\underset{j=1}{\sum}}\;
                \underset{k \in s_{M}^{(j)}}{\sum}\,w_{k}\,d_{k}\,\varphi^{(j)}_{lk}
                \right)
            \right)
        Y_{l}
    \\
    & \overset{{\color{white}\textnormal{\LARGE$1$}}}{=} &
        W^{(+)}_{d,0}
        \;+\;
        \underset{l \in s_{R}}{\sum}\left(\,w_{l}d_{l} \,+\, W^{(+)}_{d,l}\,\right)Y_{l}
    \end{eqnarray*}
\item
    We again compute:
    \begin{eqnarray*}
    \widetilde{T}_{d,Y}
    \,-\,
    \widehat{T}^{\textnormal{(HT)}}_{d,Y}
    & = &
        W^{(+)}_{d,0}
        \;+\;
        \underset{l \in s_{R}}{\sum}\!
        \left(\;
            w_{l}\,d_{l}
            \,+\,
            W^{(+)}_{d,l}
            \right)
        Y_{l}
        \;-\;
        \underset{k\,\in\,s}{\sum}\;w_{k}\,d_{k}\,Y_{k}
    \\
    & = &
        W^{(+)}_{d,0}
        \;+\;
        \underset{l \in s_{R}}{\sum}\,w_{l}\,d_{l}\,Y_{l}
        \;+\;
        \underset{l \in s_{R}}{\sum}\;W^{(+)}_{d,l}\,Y_{l}
        \;-\;
        \underset{k\,\in\,s}{\sum}\;w_{k}\,d_{k}\,Y_{k}
    \\
    & = &
        W^{(+)}_{d,0}
        \;+\;
        \underset{l \in s_{R}}{\sum}\;W^{(+)}_{d,l}\,Y_{l}
        \;-\;
        \underset{k \in s_{M}}{\sum}\;w_{k}\,d_{k}\,Y_{k}\,,
    \end{eqnarray*}
    as required.
    \qed
\end{enumerate}

\vskip 0.1cm
\begin{lemma}[Unbiased estimator for nonresponse variance; Equations (5.9) \& (5.10) in \cite{Beaumont2011}]
\label{DomainTotalNonresponseVariance}
\mbox{}
\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{DomainTotalFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{DomainTotalProbabilisticFramework}.
\vskip 0.1cm
\noindent
Then, the following statements are true:
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\mbox{}\;\;\,(\theenumi)$\quad$}
\begin{enumerate}
\item
    $
    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\,)^{2}
        \,\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    $\,
    is an $(\Sdot,\Rdot)$-unbiased estimator for the nonresponse variance\\
    \,$
    E_{SR}\!\left[\,
    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\,)^{2}
        \,\right\vert\,
        s,s_{R},\Xobs
        \,\right]
        \right]
    $.
\item
    $
    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\,)^{2}
        \,\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    $\,
    can be expressed in explicit computable form as follows:
    \begin{eqnarray*}
    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\,)^{2}
        \,\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    & = &
        \left(\;
            \underset{l \in s_{R}}{\sum}
                \left(W^{(+)}_{d,l}\right)^{2}\sigma_{l}^{2}
            \,+\,
            \underset{k \in s_{M}}{\sum}\,
                w_{k}^{2}\,d_{k}\,\sigma_{k}^{2}
            \,\right)
    \\
    &&
        +\;
        \left(\,
            W^{(+)}_{d,0}
            \,+\,
            \underset{l \in s_{R}}{\sum}\;W^{(+)}_{d,l}\,\mu_{l}
            \,-\,
            \underset{k \in s_{M}}{\sum}\;w_{k}\,d_{k}\,\mu_{k}
            \,\right)^{2}
    \end{eqnarray*}
\end{enumerate}
\end{lemma}
\proof
\begin{enumerate}
\item
    Nothing to prove.
\item
    \textbf{Claim 1:}\quad
    \begin{equation*}
    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\,)^{2}
        \,\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    \;\; = \;\;
        \Var_{Y}\!\left[\,
            \left.
            \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
        \,+\,
        E_{Y}\!\left[\,
            \left.
            \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]^{2}
    \end{equation*}
    \vskip 0.1cm
    \noindent
    Proof of Claim 1:\;\;
    Claim 1 follows immediately from the following well-known equality:
    \begin{equation*}
    \Var[\,X\,]
    \; := \;
        E\!\left[\,\overset{{\color{white}.}}{(X - \mu_{X})^{2}}\,\right]
    \; = \;
        E\!\left[\,\overset{{\color{white}.}}{X^{2} - 2\,\mu_{X}\,X + \mu_{X}^{2}}\,\right]
    \; = \;
        \cdots
    \; = \;
        E\!\left[\,X^{2}\,\right] \,-\, \mu_{X}^{2}\,,
    \end{equation*}
    This completes the proof of Claim 1.

    \vskip 0.5cm
    \textbf{Claim 2:}\quad
    \begin{equation*}
    \Var_{Y}\!\left[\,
        \left.
        \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
        \,\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    \;\; = \;\;
        \underset{l \in s_{R}}{\sum}
            \left(W^{(+)}_{d,l}\right)^{2}\sigma_{l}^{2}
        \,\;+\;
        \underset{k \in s_{M}}{\sum}\,
            w_{k}^{2}\,d_{k}^{2}\,\sigma_{k}^{2}
    \end{equation*}
    \vskip 0.1cm
    \noindent
    Proof of Claim 2:\;\;
    \begin{eqnarray*}
    &&
        \Var_{Y}\!\left[\,
            \left.
            \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \\
    & = &
        \Var_{Y}\!\left[\,
            \left.
            W^{(+)}_{d,0}
            \,+\,
            \underset{l \in s_{R}}{\sum}\,W^{(+)}_{d,l}\,Y_{l}
            \,-\,
            \underset{k \in s_{M}}{\sum}\;w_{k}\,d_{k}\,Y_{k}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right],
        \;\;
        \textnormal{by Lemma \ref{DomainTotalImputationError}(ii)}
    \\
    & = &
        \Var_{Y}\!\left[\;
            \left.
            \underset{l \in s_{R}}{\sum}\,W^{(+)}_{d,l}\,Y_{l}
            \,-\,
            \underset{k \in s_{M}}{\sum}\;w_{k}\,d_{k}\,Y_{k}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right],
        \;\;
        \textnormal{since $W^{(+)}_{d,0}$ is constant upon conditioning on $s, s_{R}$}
    \\
    & = &
        \underset{l \in s_{R}}{\sum}\,
        \underset{m \in s_{R}}{\sum}\,
        \Cov_{Y}\!\left[\;
            W^{(+)}_{d,l}\,Y_{l}
            \,,
            W^{(+)}_{m}\,Y_{m}
            \,\left\vert
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right]
        \; + \;
        \underset{k \in s_{M}}{\sum}\,
        \underset{m \in s_{M}}{\sum}\,
        \Cov_{Y}\!\left[\;
            w_{k}\,d_{k}\,Y_{k}
            \,,
            w_{m}\,d_{m}\,Y_{m}
            \,\left\vert
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right]
    \\
    &&
        \; - \;
        2 \cdot
        \underset{l \in s_{R}}{\sum}\;
        \underset{k \in s_{M}}{\sum}\,
        \Cov_{Y}\!\left[\;
            W^{(+)}_{d,l}\,Y_{l}
            \,,\,
            w_{k}\,d_{k}\,Y_{k}
            \,\left\vert
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right],
        \;\;
        \textnormal{by Lemma \ref{sRsMPullOutOfVariance}}
    \\
    & = &
        \underset{l \in s_{R}}{\sum}
        \!\left(W^{(+)}_{d,l}\right)^{2}
        \cdot
        \Var_{Y}\!\left[\;
            Y_{l}
            \,\left\vert
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right]
        \,\;+\;
        \underset{k \in s_{M}}{\sum}
        \!\left(w_{k}\,d_{k}\right)^{2}
        \cdot
        \Var_{Y}\!\left[\;
            Y_{k}
            \,\left\vert
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right],
    \\
    &&
        \quad\quad\quad
        \textnormal{since
            \,$
            \Cov\!\left(\,Y_{a},Y_{b}\,\left\vert\,s,s_{R},\Xobs\right.\,\right)
            \,=\,
            \Cov\!\left(\,Y_{a},Y_{b}\,\left\vert\,\Xobs\right.\,\right)
            \,=\,
            0$,\,
            whenever \,$a \neq b$
            }
    \\
    & = &
        \underset{l \in s_{R}}{\sum}
        \!\left(W^{(+)}_{d,l}\right)^{2}
        \cdot
        \Var_{Y}\!\left[\;
            Y_{l}
            \,\left\vert\,
            \Xobs
            \right.
            \right]
        \,\;+\;
        \underset{k \in s_{M}}{\sum}\,
        w_{k}^{2}\,d_{k}^{2}
        \cdot
        \Var_{Y}\!\left[\;Y_{k}
            \,\left\vert\,
            \Xobs
            \right.
            \,\right],
        \;\;
        \textnormal{since \,$\Ydot \perp (\Sdot,\Rdot) \;\;\vert\;\, \Xobs$}
    \\
    & = &
        \underset{l \in s_{R}}{\sum}
            \left(W^{(+)}_{d,l}\right)^{2}\sigma_{l}^{2}
        \,\;+\;
        \underset{k \in s_{M}}{\sum}\,
            w_{k}^{2}\,d_{k}^{2}\,\sigma_{k}^{2}\,,
        \;\;
        \textnormal{by definition of
            \,$\sigma_{a}^{2} \;:=\, \Var_{Y}\!\left[\;Y_{a}
                \,\left\vert\,
                \Xobs
                \right.
                \,\right]$}
    \end{eqnarray*}
    This completes the proof of Claim 2.

    \vskip 0.5cm
    \textbf{Claim 3:}\quad
    \begin{equation*}
    E_{Y}\!\left[\,
        \left.
        \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
        \,\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    \;\; = \;\;
        W^{(+)}_{d,0}
        \,\;+\;
        \underset{l \in s_{R}}{\sum}\;W^{(+)}_{d,l}\,\mu_{l}
        \,\;-\;
        \underset{k \in s_{M}}{\sum}\;w_{k}\,d_{k}\,\mu_{k}
    \end{equation*}
    \vskip 0.1cm
    \noindent
    Proof of Claim 3:\;\;
    \begin{eqnarray*}
    &&
        E_{Y}\!\left[\,
            \left.
            \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \\
    & = &
        E_{Y}\!\left[\,
            \left.
            W^{(+)}_{d,0}
            \,+\,
            \underset{l \in s_{R}}{\sum}\,W^{(+)}_{d,l}\,Y_{l}
            \,-\,
            \underset{k \in s_{M}}{\sum}\;w_{k}\,d_{k}\,Y_{k}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right],
        \quad
        \textnormal{by Lemma \ref{DomainTotalImputationError}(ii)}
    \\
    & = &
        W^{(+)}_{d,0}
        \;+\;
        \underset{l \in s_{R}}{\sum}\;
        W^{(+)}_{d,l}
        \cdot
        E_{Y}\!\left[\;
            Y_{l}
            \,\left\vert\,
            s,s_{R},\Xobs
            \right.
            \right]
        \,\;-\;
        \underset{k \in s_{M}}{\sum}\;
        w_{k}\,d_{k}
        \cdot
        E_{Y}\!\left[\;Y_{k}
            \,\left\vert\,
            s,s_{R},\Xobs
            \right.
            \,\right]
    \\
    & = &
        W^{(+)}_{d,0}
        \;+\;
        \underset{l \in s_{R}}{\sum}\;
        W^{(+)}_{d,l}
        \cdot
        E_{Y}\!\left[\;
            Y_{l}
            \,\left\vert\,
            \Xobs
            \right.
            \right]
        \,\;-\;
        \underset{k \in s_{M}}{\sum}\;
        w_{k}\,d_{k}
        \cdot
        E_{Y}\!\left[\;Y_{k}
            \,\left\vert\,
            \Xobs
            \right.
            \,\right],
        \;\;
        \textnormal{since \,$\Ydot \perp (\Sdot,\Rdot) \;\;\vert\;\, \Xobs$}
    \\
    & = &
        W^{(+)}_{d,0}
        \;+\;
        \underset{l \in s_{R}}{\sum}\;W^{(+)}_{d,l}\,\mu_{l}
        \,\;-\;
        \underset{k \in s_{M}}{\sum}\,
            w_{k}\,d_{k}\,\mu_{k}\,,
        \;\;
        \textnormal{by definition of
            \,$\mu_{k} \;:=\, E_{Y}\!\left[\;Y_{k}
                \,\left\vert\,
                \Xobs
                \right.
                \,\right]$}
    \end{eqnarray*}
    This completes the proof of Claim 3.

    \vskip 0.3cm
    \noindent
    The required result now follows immediately from Claim 1, Claim 2, and Claim 3.
    \qed
\end{enumerate}

\vskip 0.1cm
\begin{lemma}[Unbiased estimator for sampling-nonresponse covariance]
\label{DomainTotalMixedVariance}
\mbox{}
\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{DomainTotalFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{DomainTotalProbabilisticFramework}.
\vskip 0.1cm
\noindent
Then, the following statements are true:
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\mbox{}\;\;\,(\theenumi)$\quad$}
\begin{enumerate}
\item
    $
    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\,)
        (\,  \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}\,)
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    $\,
    is an $(\Sdot,\Rdot)$-unbiased estimator for the mixed variance\\
    \,$
    E_{SR}\!\left[\,
    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\,)
        (\,  \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}\,)
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
        \right]
    $.
\item
    The above estimator can be expressed as follows:
    \begin{eqnarray*}
    &&
        E_{Y}\!\left[\,
            \left.
            (\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\,)
            (\,  \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}\,)
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \\
    & = &
        \Cov_{Y}\!\left(\,
            \left.
            \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,,\,
            \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right)
    \\
    &&
        +\;
        E_{Y}\!\left[\;
            \left.
            \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]
        \cdot
        {\color{red}E_{Y}\!\left[\;
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]}
    \end{eqnarray*}
\item
    The above covariance term can be expressed in explicit computable form as follows:
    \begin{eqnarray*}
    \Cov_{Y}\!\left(\,
        \left.
        \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
        \,,\,
        \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right)
    & = &
        \underset{l \in s_{R}}{\sum}\,
            W^{(+)}_{d,l}\,(w_{k}-1)\,d_{l}\,\sigma_{l}^{2}
        \,-\,
        \underset{k \in s_{M}}{\sum}\,
            w_{k}\,(w_{1}-1)\,d_{k}^{2}\,\sigma_{k}^{2}
    \end{eqnarray*}
\end{enumerate}
\end{lemma}
\proof
\begin{enumerate}
\item
    Nothing to prove.
\item
    The required result follows from the following general fact:
    \begin{equation*}
    \Cov(\,X,Y\,)
    \;\; := \;\;
        E\!\left[\,\overset{{\color{white}.}}{(X - \mu_{X})(Y - \mu_{Y})}\,\right]
    \;\; = \;\;
        \cdots
    \;\; = \;\;
        E\!\left[\,\overset{{\color{white}.}}{XY}\,\right]
        \; - \;
        \mu_{X}\,\mu_{Y}\,,
    \end{equation*}
    which implies
    \begin{equation*}
    E\!\left[\,\overset{{\color{white}.}}{XY}\,\right]
    \;\; = \;\;
        \Cov(\,X,Y\,)
        \; + \;
        \mu_{X}\,\mu_{Y}
    \end{equation*}
\item
    We compute:
    \begin{eqnarray*}
    &&
        \Cov_{Y}\!\left(\,
            \left.
            \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,,\,
            \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right)
    \\
    & = &
        \Cov_{Y}\!\left(\,
            \left.
            W^{(+)}_{d,0}
            \,+\,
            \underset{l \in s_{R}}{\sum}\,W^{(+)}_{d,l}\,Y_{l}
            \;-
            \underset{k \in s_{M}}{\sum}\,w_{k}\,d_{k}\,Y_{k}
            \;,\;
            \underset{k \in s}{\sum}\;w_{k}d_{k}Y_{k}
            \;-
            \underset{k \in U}{\sum}\;d_{k}Y_{k}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right),
            \quad
            \textnormal{by Lemma \ref{DomainTotalImputationError}(ii)}
    \\
    & = &
        \Cov_{Y}\!\left(\;
            \left.
            \underset{l \in s_{R}}{\sum}\,W^{(+)}_{d,l}\,Y_{l}
            \;-
            \underset{k \in s_{M}}{\sum}\,w_{k}\,d_{k}\,Y_{k}
            \;,\;
            \underset{k \in s}{\sum}\;(w_{k}-1)\,d_{k}Y_{k}
            \;-
            \underset{k \in U \backslash s}{\sum}d_{k}Y_{k}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right)
    \\
    & = &
        \underset{l \in s_{R}}{\sum}\;
        \underset{k \in s}{\sum}\;
        \Cov_{Y}\!\left(\;
            W^{(+)}_{d,l}\,Y_{l}
            \;,\,
            (w_{k}-1)\,d_{k}Y_{k}
            \,\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right)
        \; - \;
        \underset{l \in s_{R}}{\sum}\;
        \underset{k \in U \backslash s}{\sum}\,
        \Cov_{Y}\!\left(\;
            W^{(+)}_{d,l}\,Y_{l}
            \;,\,
            d_{k}Y_{k}
            \,\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right)
    \\
    &&
        \; - \;
        \underset{k \in s_{M}}{\sum}\;
        \underset{l \in s}{\sum}\;
        \Cov_{Y}\!\left(\;
            w_{k}\,d_{k}\,Y_{k}
            \;,\,
            (w_{l}-1)\,d_{l}Y_{l}
            \,\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right)
        \; + \;
        \underset{k \in s_{M}}{\sum}\;
        \underset{l \in U \backslash s}{\sum}\,
        \Cov_{Y}\!\left(\;
            w_{k}\,d_{k}\,Y_{k}
            \;,\,
            d_{l}Y_{l}
            \,\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right),
    \\
    &&
        \quad\quad\quad
        \textnormal{by Lemma \ref{sRsMPullOutOfCovariance}}
    \\
    & = &
        \underset{l \in s_{R}}{\sum}\,
        W^{(+)}_{d,l}\,(w_{l}-1)\,d_{l}\,\Var_{Y}\!\left[\;
            Y_{l}
            \;\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right]
        \;-\;
        \underset{k \in s_{M}}{\sum}\,
        w_{k}\,(w_{k}-1)\,d_{k}^{2}\,\Var_{Y}\!\left[\;
            Y_{k}
            \;\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right],
    \\
    &&
        \quad\quad\quad
        \textnormal{since
            \,$
            \Cov\!\left(\,Y_{a},Y_{b}\,\left\vert\,s,s_{R},\Xobs\right.\right)
            \,=\,
            \Cov\!\left(\,Y_{a},Y_{b}\,\left\vert\,\Xobs\right.\right)
            \,=\,
            0
            $,\, whenever \,$a \neq b$}
    \\
    & = &
        \underset{l \in s_{R}}{\sum}\,
        W^{(+)}_{d,l}\,(w_{l}-1)\,d_{l}\,\Var_{Y}\!\left[\,
            Y_{l}
            \left\vert\,
            \Xobs
            \right.
            \right]
        \,-\,
        \underset{k \in s_{M}}{\sum}\,
        w_{k}\,(w_{k}-1)\,d_{k}^{2}\,\Var_{Y}\!\left[\,
            Y_{k}
            \left\vert\,
            \Xobs
            \right.
            \right],
        \;\;
        \textnormal{since $\Ydot \perp (\Sdot,\Rdot) \;\vert\; \Xobs$}
    \\
    & = &
        \underset{l \in s_{R}}{\sum}\,
        W^{(+)}_{d,l}\,(w_{l}-1)\,d_{l}\,\sigma_{l}^{2}
        \;\;-\;
        \underset{k \in s_{M}}{\sum}\,
        w_{k}\,(w_{k}-1)\,d_{k}^{2}\,\sigma_{k}^{2}
    \end{eqnarray*}
    % where
    % the third equality follows from the conditional independence
    % \,$Y_{k} \perp Y_{l} \;\,\vert\; \Xobs$ whenever \,$k \neq l$,\,
    % while
    % the fourth equality follows from the conditional independence
    % \,$\Ydot \perp (\Sdot,\Rdot) \;\vert\; \Xobs$.\,
    \vskip 0.1cm
    \noindent
    This completes the proof of the Lemma.
    \qed
\end{enumerate}

\subsection{Summary: variance estimation for composite linear imputation estimators in practice}
\setcounter{theorem}{0}

        %%%%% ~~~~~~~~~~~~~~~~~~~~ %%%%%

\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{DomainTotalFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{DomainTotalProbabilisticFramework}.
\vskip 0.1cm
\noindent
Then, the following summarizes the results in \cite{Beaumont2011}:
\begin{itemize}
\item
    By Lemma \ref{DecompositionMSETtilde}, we have the following decomposition:
    \begin{eqnarray*}
    \MSE\!\left(\;
        \left.
        \widetilde{T}_{d,Y}
        \;\right\vert\,
        \Xobs
        \,\right)
    & := &
        E\!\left[
            \left.
            \left(\,\widetilde{T}_{d,Y} - T_{d,Y}\right)^{2}
            \,\right\vert\,
            \Xobs
            \,\right]
    \\
    & = &
        E_{Y}\!\left[\,
        \Var_{SR}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,\right\vert\,
            \Ydot,
            \Xobs
            \,\right)
            \,\right]
        \;+\;
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\right)^{2}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \\
    &&
        +\;2\cdot
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{d,Y} \,-\, \widehat{T}^{\textnormal{(HT)}}_{d,Y}\right)
            \!\!
            \left(\,\widehat{T}^{\textnormal{(HT)}}_{d,Y} \,-\, T_{d,Y}\right)
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \end{eqnarray*}
    We use the following nomenclature to refer the three summands in the right-hand side:
    \begin{eqnarray*}
    \left(\begin{array}{c}
        \textnormal{sampling}
        \\
        \textnormal{\;\,variance component\;\,}
        \end{array}\right)
    & := &
        E_{Y}\!\left[\,
        \Var_{SR}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,\right\vert\,
            \Ydot,
            \Xobs
            \,\right)
            \,\right]
    \\
    \left(\begin{array}{c}
        \textnormal{nonresponse}
        \\
        \textnormal{\;\,variance component\;\,}
        \end{array}\right)
    & := &
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\right)^{2}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \\
    \left(\begin{array}{c}
        \textnormal{sampling-nonresponse}
        \\
        \textnormal{covariance}
        \end{array}\right)
    & := &
        2\cdot
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{d,Y} \,-\, \widehat{T}^{\textnormal{(HT)}}_{d,Y}\,\right)
            \!\!
            \left(\,\widehat{T}^{\textnormal{(HT)}}_{d,Y} \,-\, T_{d,Y}\,\right)
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \end{eqnarray*}
\item
    By Lemma \ref{DomainTotalSamplingVariance},
    the sampling variance component can be estimated as follows:
    \begin{eqnarray*}
    \left(\!\begin{array}{c}
        \textnormal{sampling}
        \\
        \textnormal{variance component}
        \end{array}\!\right)
    & := &
        E_{Y}\!\left[\,
        \Var_{SR}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,\right\vert\,
            \Ydot,
            \Xobs
            \,\right)
            \,\right]
    \;\; = \;\;
        E_{SR}\,
        E_{Y_{R}}\,
        E_{Y_{M}}\!\left[\;
            \widehat{V}^{\textnormal{(HT)}}(\,s\,;\dbullet\Ydot\,)
            \,\left\vert\!
            \begin{array}{c} s,s_{R},\YsubR, \\ \Xobs \end{array}
            \right.
            \!\!\right]
    \\
    & {\color{red}\approx} &
    E_{Y_{M}}\!\left[\;
        \widehat{V}^{\textnormal{(HT)}}(\,s\,;\dbullet\Ydot\,)
        \,\left\vert\!
        \begin{array}{c} s,s_{R},\YsubR, \\ \Xobs \end{array}
        \right.
        \!\!\right]
    \\
    & = &
        {\color{blue}
        \widehat{V}^{\textnormal{(HT)}}(\,s\,;\dbullet\widetilde{y}_{\bullet}\,)
        \;+\;
        \underset{k \in s_{M}}{\sum}\;
        d_{k}
        \left(1-\dfrac{1}{w_{k}}\right)
        w_{k}^{2}\,\sigma_{k}^{2}
        }\,,
    \end{eqnarray*}
    where
    \,$\widehat{V}^{\textnormal{(HT)}}(\,s\,;\dydot\,)$\,
    is the (design-unbiased) Horvitz-Thompson estimator for sampling variance, i.e.,
    \begin{equation*}
    \widehat{V}^{\textnormal{(HT)}}(\,s\,;\dydot\,)
    \;\; := \;\;
        \underset{k \in s}{\sum}\,
        \underset{l \in s}{\sum}\,
        \dfrac{\pi_{kl} - \pi_{k}\pi_{l}}{\pi_{kl}}\,
        (w_{k}d_{k}y_{k})\,(w_{l}d_{l}y_{l})\,,
    \end{equation*}
    and
    \begin{equation*}
    \widetilde{y}_{k}
    \;\; := \;\;
        \left\{\begin{array}{cl}
            y_{k}, & \textnormal{for \,$k \,\in\, s_{R}$}
            \\
            \overset{{\color{white}\textnormal{\large1}}}{\mu_{k}}, & \textnormal{for \,$k \,\in\, s_{M} \,:=\, s \,\backslash\, s_{R}$}
            \end{array}\right.
    \end{equation*}
\item
    By Lemma \ref{DomainTotalNonresponseVariance},
    the nonresponse variance component can be estimated as follows:
    \begin{eqnarray*}
    \left(\!\begin{array}{c}
        \textnormal{nonresponse}
        \\
        \textnormal{variance}
        \\
        \textnormal{{\color{white}..}component{\color{white}..}}
        \end{array}\!\right)
    & := &
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\right)^{2}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \;\; {\color{red}\approx} \;\;
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}\right)^{2}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \\
    & = &
        {\color{blue}
        \left(\;
            \underset{l \in s_{R}}{\sum}
                \left(W^{(+)}_{d,l}\right)^{2}\sigma_{l}^{2}
            \,+\,
            \underset{k \in s_{M}}{\sum}\,
                w_{k}^{2}\,d_{k}^{2}\,\sigma_{k}^{2}
            \,\right)
        \;+\;
        \left(\,
            W^{(+)}_{d,0}
            \,+\,
            \underset{l \in s_{R}}{\sum}\;W^{(+)}_{d,l}\,\mu_{l}
            \,-\,
            \underset{k \in s_{M}}{\sum}\;w_{k}\,d_{k}\,\mu_{k}
            \,\right)^{2}
        }
    \end{eqnarray*}
\item
    By Lemma \ref{DomainTotalMixedVariance},
    the sampling-nonresponse covariance component can be estimated as follows:
    \begin{eqnarray*}
    \left(\!\begin{array}{c}
        \textnormal{sampling-nonresponse}
        \\
        \textnormal{covariance}
        \end{array}\!\right)
    & := &
        2\cdot
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{d,Y} \,-\, \widehat{T}^{\textnormal{(HT)}}_{d,Y}\right)
            \!\!
            \left(\,\widehat{T}^{\textnormal{(HT)}}_{d,Y} \,-\, T_{d,Y}\right)
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \\
    & \approx &
        2\cdot
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{d,Y} \,-\, \widehat{T}^{\textnormal{(HT)}}_{d,Y}\right)
            \!\!
            \left(\,\widehat{T}^{\textnormal{(HT)}}_{d,Y} \,-\, T_{d,Y}\right)
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \\
    & = &
        2\cdot
        \Cov_{Y}\!\left(\,
            \left.
            \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,,\,
            \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right)
    \\
    &&
        +\;
        2\cdot
        E_{Y}\!\left[\;
            \left.
            \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]
        \cdot
        {\color{red}E_{Y}\!\left[\;
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]}
    \\
    & {\color{red}\approx} &
        2\cdot
        \Cov_{Y}\!\left(\,
            \left.
            \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
            \,,\,
            \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right)
    \\
    & = &
        {\color{blue}
        2\,\cdot\!\left(\;
            \underset{l \in s_{R}}{\sum}\,
                W^{(+)}_{d,l}\,(w_{l}-1)\,d_{l}\,\sigma_{l}^{2}
            \;\,-\;
            \underset{k \in s_{M}}{\sum}\,
                w_{k}\,(w_{1}-1)\,d_{k}^{2}\,\sigma_{k}^{2}
            \,\right)
        }
    \end{eqnarray*}
\item
    \textbf{Important Remark:}
    \vskip 0.05cm
    \noindent
    Note that the above estimation/approximation of the mixed variance component
    disregards the following term:
    \begin{equation*}
    2\,\cdot E_{Y}\!\left[\;
        \left.
        \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    \cdot
    {\color{red}E_{Y}\!\left[\;
        \left.
        \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]}
    \end{equation*}
    The reason this has to be done in practice is that, in practice,
    there is usually insufficient data/information to compute or estimate
    the quantity
    \,$
    E_{Y}\!\left[\;
        \left.
        \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    $\,
    (indicated in red above), as \,$T_{d,Y}$\, involves population units
    that are not selected in the sample.
    \vskip 0.1cm
    \noindent
    Obviously, the above estimation/approximation of the mixed variance component
    is acceptable only if the quantity
    \,$
    2\,\cdot E_{Y}\!\left[\;
        \left.
        \widetilde{T}_{d,Y} - \widehat{T}^{\textnormal{(HT)}}_{d,Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    \cdot
    {\color{black}E_{Y}\!\left[\;
        \left.
        \widehat{T}^{\textnormal{(HT)}}_{d,Y} - T_{d,Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]}
    $\,
    is small.
    \vskip 0.1cm
    \noindent
    For this, in practice, we thus {\color{red}tacitly assume} that the quantity
    \,$
    E_{Y}\!\left[\;
        \left.
        \widetilde{T}_{d,Y} \overset{{\color{white}.}}{-} \widehat{T}^{\textnormal{(HT)}}_{d,Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    $\,
    is small;
    recall that this quantity is called the \textbf{conditional model bias}
    (see Lemma \ref{biasTtilde}), and
    it can be expressed as follows:
    \begin{eqnarray*}
    E_{Y}\!\left[\;
        \left.
        \widetilde{T}_{d,Y} \overset{{\color{white}.}}{-} \widehat{T}^{\textnormal{(HT)}}_{d,Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    & = &
        \underset{k\in{\color{red}s_{M}}}{\sum}\,d_{k}\,w_{k}\,
        E_{Y}\!\left[\;
            \left.
            Y^{*}_{k} \overset{{\color{white}.}}{-} Y_{k}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \end{eqnarray*}
    This harks back to our earlier remark (Remark \ref{remarkSmallUnitBias}) that,
    in practice, we would like to choose an imputation strategy such that
    \begin{equation*}
    E_{Y}\!\left[\;
        \left.
        Y^{*}_{k} \overset{{\color{white}.}}{-} Y_{k}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    \;\; \approx \;\;
        0,
    \quad
    \textnormal{for each \,$k \in s_{M}$}
    \end{equation*}
    See the paragraph after Equation (5.2), p.175, \cite{Beaumont2011}.
\end{itemize}
