\section{SEVANI: Variance estimation methodology \cite{Beaumont2011} for {\color{red}Horvitz-Thompson} estimators under one-stage cluster sampling with composite linear imputation for {\color{red}generalized totals}}
\label{GeneralizedTotalSEVANI}

        %%%%% ~~~~~~~~~~~~~~~~~~~~ %%%%%

\subsection{Formulation of the target variability quantification problem}
\label{GeneralizedTotalFormulationOfVarianceEstimationProblem}
\setcounter{theorem}{0}

        %%%%% ~~~~~~~~~~~~~~~~~~~~ %%%%%

Suppose:
\begin{itemize}
\item
    We would like to estimate
    -- using data from a one-stage cluster sample --
    a \textbf{generalized total}
    \begin{equation*}
    T_{\alpha Y}
    \;\; := \;\;
        \underset{i \in U}{\sum}\;
        \underset{j \in \mathcal{C}_{j}}{\sum}\;
        \alpha_{ij}Y_{ij}\,,
    \end{equation*}
    where
    \begin{itemize}
    \item
        ${\color{red}U} = \{\,1,2,\ldots,N\,\} \subset \N$\,
        denotes the population of the primary sampling units (PSU),
    \item
        for each \,$i \in U$,\,
        \,${\color{red}\mathcal{C}_{j}}$\,
        denotes the collection of all secondary sampling units (SSU) belonging to PSU \,$i \in U$,
    \item
        $
        {\color{red}\Ydot} \, = \, \left\{\,
            \left.
            \overset{{\color{white}.}}{Y_{ij}}
            \,\right\vert\,
            i \in U, j \in \mathcal{C}_{i}
            \,\right\}
        $\,
        is an $\Re$-valued population characteristic defined at the level of the SSUs, and
    \item
        $
        {\color{red}\abullet} \, = \, \left\{\,
            \left.
            \overset{{\color{white}1}}{\alpha_{ij}}
            \;\right\vert\,
            i \in U, j \in \mathcal{C}_{i}
            \,\right\}
        $\,
        is the collection of coefficients defined at the level of the SSUs.
    \end{itemize}
\item
    Aside/Reminder:
    \vskip 0.01cm
    \noindent
    If there were no missing data in \,$\ydot$\, for the units in the selected sample,
    then the \textit{Horvitz-Thompson generalized total estimator\! \footnote{
        We will use the alternative notations
        \,$\widehat{T}_{\alpha\xi}^{\textnormal{(HT)}}$\,
        and
        \,$\widehat{T}^{\textnormal{(HT)}}(s,\xi_{\bullet}\,;\abullet)$\,
        interchangeably, noting that the former is notationally more succinct,
        while the latter allows us to emphasize that the population characteristic
        \,$\xi_{\bullet}$\,
        is being considered as an argument to the formula
        \begin{equation*}
        \widehat{T}^{\textnormal{(HT)}}(s,\xi_{\bullet}\,;\abullet)
        \;\; := \;\;
            \underset{i\,\in\,s}{\sum}\;
                \dfrac{1}{\pi_{i}}\!
                \left(\;
                    \underset{j \in \mathcal{C}_{j}}{\sum}\;\alpha_{ij}\,\xi_{ij}
                    \right)
        \end{equation*}
        for the Horvitz-Thompson population total estimator.
        }
    }
    \begin{eqnarray*}
    \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
    & :=: &
        % \widehat{T}^{\textnormal{(HT)}}(s\,;\dydot)
        \widehat{T}^{\textnormal{(HT)}}(\,\Sdot,\Ydot\,;\abullet\,)
    \;\; := \;\;
        \underset{i\,\in\,s}{\sum}\;
            \dfrac{1}{\pi_{i}}
            \left(\;
                \underset{j \in \mathcal{C}_{j}}{\sum}\;
                \alpha_{ij}Y_{ij}
                \right)
    \end{eqnarray*}
    is a design-unbiased estimator for
    \,$T_{\alpha Y}$,\,
    where
    \begin{itemize}
    \item
        ${\color{red}s} \subset U$\, is the (random) selected sample of PSUs, and
        \,$\Sdot \,=\, (S_{1},\ldots,S_{N})^{T}$\,
        is the vector of $\{0,1\}$-valued selection indicator variables,
    \item
        ${\color{red}w_{i}} \,:=\, \dfrac{1}{\pi_{i}}$,\, for each population unit \,$i \in U$,\,
        is the (\textit{a priori} known, non-random) design weight of \,$i \in U$,\,
        i.e., the reciprocal of the selection probability \,${\color{red}\pi_{i}}$\,
        (assumed strictly positive) of \,$i \in U$.\,
    \end{itemize}
    Recall (Result 4.3.1, p.137, \cite{Sarndal1992}) also that
    the sampling variance of the Horvitz-Thompson population total estimator
    under a two-stage cluster sampling design is:
    \begin{equation*}
    \Var_{\,\textnormal{2-stage}}
    \;\; = \;\;
        \Var_{\,\textnormal{PSU}}
        \; + \;
        \Var_{\,\textnormal{SSU}}
    \end{equation*}
    Since we are considering only one-stage cluster sampling
    (i.e., two-stage cluster sampling where the second-stage sampling is a census),
    the subsampling variance \,$\Var_{\,\textnormal{SSU}}$\, is zero.
    Hence, the sampling variance of
    \,$
    \widehat{T}^{\textnormal{(HT)}}_{\alpha\xi}
    % \,=\,
    %     \widehat{T}^{\textnormal{(HT)}}(s\,;\dbullet\xi_{\bullet})
    $\,
    has only the \,$\Var_{\,\textnormal{PSU}}$\, term, i.e.,
    \begin{equation*}
    \Var_{S}\!\left(\,
        \left.
        \widehat{T}^{\textnormal{(HT)}}_{\alpha\xi}
        \,\right\vert\!\!
        \begin{array}{c} \xi_{\bullet} \end{array}
        \!\!\right)
    \;\; = \;\;\,
        \underset{i \in U}{\sum}\,
        \underset{k \in U}{\sum}\,
        \left(\,\pi_{ik} \overset{{\color{white}.}}{-} \pi_{i}\pi_{k}\,\right)
        \cdot\!
        \left(\,
            \dfrac{1}{\pi_{i}}\,
            \underset{j\in\mathcal{C}_{i}}{\sum}\;\alpha_{ij}\,\xi_{ij}
            \,\right)
        \!\cdot\!
        \left(\,
            \dfrac{1}{\pi_{i}}\,
            \underset{l\in\mathcal{C}_{k}}{\sum}\;\alpha_{kl}\,\xi_{kl}
            \,\right),
    \end{equation*}
    and the following Horvitz-Thompson variance estimator
    \begin{equation*}
    \widehat{V}^{\textnormal{(HT)}}(\,s\,;\abullet,\xi_{\bullet}\,)
    \;\; := \;\;
        \underset{i \in s}{\sum}\,
        \underset{k \in s}{\sum}\;
        \dfrac{\pi_{ik} - \pi_{i}\pi_{k}}{\pi_{ik}}
        \cdot\!
        \left(\,
            \dfrac{1}{\pi_{i}}\,
            \underset{j\in\mathcal{C}_{i}}{\sum}\;\alpha_{ij}Y_{ij}
            \,\right)
        \!\cdot\!
        \left(\,
            \dfrac{1}{\pi_{i}}\,
            \underset{l\in\mathcal{C}_{k}}{\sum}\;\alpha_{kl}Y_{kl}
            \,\right),
    \end{equation*}
    is a design-unbiased estimator for the sampling variance of
    % \,$\widehat{T}_{d,\xi_{\bullet}}(s) \,=\, \widehat{T}_{d}(s\,;\xi_{\bullet})$,\,
    \,$
    \widehat{T}^{\textnormal{(HT)}}_{\alpha\xi}
    \,=\,
        \widehat{T}^{\textnormal{(HT)}}(s\,;\abullet\xi_{\bullet})
    $,\,
    i.e.,
    \begin{equation*}
    E_{S}\!\left[\;
        \left.
        \widehat{V}^{\textnormal{(HT)}}(\,s\,;\abullet,\xi_{\bullet}\,)
        \;\right\vert\!
        \begin{array}{c} \xi_{\bullet} \end{array}
        \!\right]
    \;\; = \;\;
        \Var_{S}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{\alpha\xi} %(s\,;\dbullet\xi_{\bullet})
            \,\right\vert\!
            \begin{array}{c} \xi_{\bullet} \end{array}
            \!\!\right)
    \end{equation*}
\item
    For each PSU \,$i \in s$,\,
    the values of \,$Y_{ij}$\, have been successfully observed
    only for SSUs \,$j \in \mathcal{C}_{i}$\,
    in the (random) ``responding'' subsample of SSUs
    \,${\color{red}(\mathcal{C}_{i})_{R}} \subset \mathcal{C}_{i} \subset U$;\,
    in other words, the measurement of \,$Y_{ij}$\, somehow failed for each SSU
    \,$
    j
    \in
        {\color{red}(\mathcal{C}_{i})_{M}}
    \, := \,
        \mathcal{C}_{i} \,\backslash\, (\mathcal{C}_{i})_{R}
    $.\,
    \,$(\mathcal{C}_{i})_{M}$\,
    is called the ``nonresponding'' subsample of PSU \,$i \in U$.\,
    We will also use
    \,${\color{red}s_{R}} \, := \, \underset{i \in s}{\bigcup}\,(\mathcal{C}_{i})_{R}$\,
    to denote the full (random) collection of selected and responding SSUs.
\item
    We would like to estimate \,$T_{\alpha Y}$\,
    with an imputation estimator of the following form:
    \begin{eqnarray*}
    \widetilde{T}_{\alpha Y}
    & = &
        \widetilde{T}_{\alpha Y}(\,
            \Sdot\,,\,
            \Rdot\,,\,
            \Ydot\,;\,
            \Xdotdot\,,\,
            \abullet
            \,)
    \\
    & = &
        \underset{i\,\in\,s}{\sum}\;
        \dfrac{1}{\pi_{i}}\!
        \left(\;
            \underset{j \in \mathcal{C}_{i}}{\sum}\;\alpha_{ij}\,Y^{\dagger}_{ij}
            \right)
    \;\; = \;\;
        \underset{i\,\in\,U}{\sum}\;
            \dfrac{S_{i}}{\pi_{i}}
            \left(\;
                \underset{j \in \mathcal{C}_{i}}{\sum}\;\alpha_{ij}\!
                \left(
                    R_{ij}\,Y_{ij}
                    \overset{{\color{white}.}}{+}
                    (1-R_{ij})\,Y^{*}_{ij}
                    \right)
                \right),
    \end{eqnarray*}
    where
    \begin{itemize}
    \item
        $
        \Rdot
        \,=\,
            \left\{\,
                R_{ij}
                \,\left\vert
                \begin{array}{c} i \in U, \\ j \in \mathcal{C}_{i} \end{array}
                \right.
                \!\!\right\}
        $\,
        is the collection of $\{0,1\}$-valued imputation-unit-level response indicator variables,
    \item
        $\Xdotdot \,=\, \left(\,X_{ij}\,\right) \,\in\, \overline{\Re}^{N \times r}$\,
        is a random matrix containing the values of the $\Re$predictor variables for $\Ydot$,
        where the $i^{\textnormal{th}}$ row of \,$\Xdotdot$\, corresponds to populaton unit \,$i \in U$,\,
        and where \,$\overline{R} \,:=\, \Re \sqcup \{\,\texttt{NA}\,\}$\,
        is an extension of $\Re$ alllowing for missing values in \,$\Xdotdot$,\, and
    \item
        for each \,$i \in s$\, and \,$j \in \mathcal{C}_{i}$,\,
        \,$Y^{\dagger}_{ij} \;=\; R_{ij}\,Y_{ij}+(1-R_{ij})\,Y^{*}_{ij}$,\,
        and
        \begin{equation*}
        Y^{*}_{ij}
        \;\; = \;\;
            Y^{*}_{ij}(\,\Sdot,\Rdot,\Ydot,\Xdotdot\,)\,,
        \quad
        \textnormal{for each \,$j \in (\mathcal{C}_{i})_{M} = \mathcal{C}_{i}\,\backslash\,(\mathcal{C}_{i})_{R}$},
        \end{equation*}
        is an imputed value for
        \,$Y_{ij}$\, and \,$Y^{*}_{ij}$\,
        may depend on
        \,$\Sdot,\Rdot,\Ydot,\Xdotdot$.\,
    \end{itemize}
\item
    Furthermore, \,$Y^{*}_{kl}$\, ($k \in s$,\, $l \in (\mathcal{C}_{k})_{M}$)
    is \textbf{composite linear}, i.e., % there exists \,$J \in \N$\, such that
    \begin{center}
    \begin{minipage}{6.3in}
    for each admissible sample \,$s \subset U$,\,
    each \,$s_{R} := \underset{i \in s}{\bigcup}\,(\mathcal{C}_{i})_{R}$,\,
    {\color{red}
        each
        \,$k \in s$,\,
        and each
        \,$l \in (\mathcal{C}_{k})_{M} := \mathcal{C}_{k} \,\backslash\, (\mathcal{C}_{k})_{R}${\color{black},}
    }
    \vskip 0.01cm
    \noindent
    there exist
    \begin{equation*}
    \left\{\;\;\begin{array}{ccl}
    % j
    % & = &
    %     j(\,k,s,s_{R}\,;\Xobs\,) \;\in\; \{1,2,\ldots,J\},
    % \\
    \varphi_{0,kl}
    & \overset{{\color{white}\textnormal{\small$1$}}}{=} &
        \varphi_{0,kl}(\,s,s_{R}\,;\Xdotdot\,),
        \quad
        \textnormal{and}
    \\
    \varphi_{ij,kl}
    & \overset{{\color{white}\textnormal{\Large$1$}}}{=} &
        \varphi_{ij,kl}(\,s,s_{R}\,;\Xdotdot\,),
        \;\;
        \textnormal{{\color{red}for each \,$i \in s$,\, $j \in (\mathcal{C}_{i})_{R}$}}
    \end{array}\right.
    \end{equation*}
    such that
    \begin{eqnarray*}
    Y_{{\color{red}kl}}^{*}
    & = &
        Y^{*}_{kl}(\,\Sdot,\Rdot,\Ydot,\Xdotdot\,)
    \\
    & = &
        \varphi_{0,{\color{red}kl}}
        \, + \,
        \underset{i \in s}{\sum}\;
        \underset{j \in (\mathcal{C}_{i})_{R}}{\sum}\,
        \varphi_{ij,{\color{red}kl}}\,Y_{ij}\,,
        \quad
        \textnormal{
            for each
            \,$k \in s$,
            \,$l \in (\mathcal{C}_{k})_{M} := \mathcal{C}_{k} \,\backslash\, (\mathcal{C}_{k})_{R}$
            },
    \end{eqnarray*}
    \end{minipage}
    \end{center}
\end{itemize}
\vskip 0.1cm

\vskip 1.0cm
\subsection{Probabilistic framework of SEVANI \cite{Beaumont2011}}
\label{GeneralizedTotalProbabilisticFramework}
\setcounter{theorem}{0}

        %%%%% ~~~~~~~~~~~~~~~~~~~~ %%%%%

\noindent
\textbf{Probability spaces}
\begin{itemize}
\item
    Let
    \,$U \,=\, \{1,2,\ldots,N\}$.\,
    Let
    \,$\mathcal{P}(U) \,=\, \mathcal{P}(\{1,2,\ldots,N\})$\,
    denote the power set (set of all subsets) of \,$U$.\,
    Let
    \,$\left(\,\mathcal{P}(U)\,,\mathcal{P}(\mathcal{P}(U))\,,\overset{{\color{white}-}}{\pi}\,\right)$\,
    be the probability space induced by a sampling design, i.e.,
    a probability mass function defined on
    \,$\mathcal{P}(U)$.\,
\item
    Let \,$\left(\,\Omega,\mathscr{A},\lambda\,\right)$\, be a probability space.
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Random variables}
\begin{itemize}
\item
    $
    \Ydot
    \, = \,
        \left\{\;
            Y_{ij} : \Omega \longrightarrow \Re
            \,\left\vert
            \begin{array}{c} i \in U, \\ j \in \mathcal{C}_{i} \end{array}
            \right.
            \!\!\right\}
    $\,
    is a collection of $\Re$-valued random variables
    (i.e., $\mathscr{A}$-measurable maps)
    defined on \,$\Omega$,\,
    where \,$\mathcal{C}_{i}$\, is a finite set, for each \,$i \in U$.
\item
    $
    \Xdotdot
    \, = \,
        \left\{\,
            X_{ij} : \Omega \longrightarrow \Re^{r}
            \left\vert
            \begin{array}{c} i \in U, \\ j \in \mathcal{C}_{i} \end{array}
            \right.
            \!\!\right\}
    $\,
    is a collection of $\Re^{r}$-valued random variables defined on \,$\Omega$.\,
\item
    $\Sdot \,=\, (S_{1},S_{2},\ldots,S_{N})^{T} : \mathcal{P}(U) \longrightarrow \{0,1\}^{N \times 1}$\,
    is a $\{0,1\}^{N \times 1}$-valued random variable defined on \,$\mathcal{P}(U)$,\,
    i.e., $\mathcal{P}(\mathcal{P}(U))$-measurable map.
\item
    % $\Rdot \,=\, (R_{1},R_{2},\ldots,R_{N})^{T} : {\color{red}\Omega \times \mathcal{P}(U)} \longrightarrow \{0,1\}^{N \times 1}$\,
    % \vskip 0.1cm
    $
    \Rdot
    \, = \,
        \left\{\,
            R_{ij} : {\color{red}\Omega \times \mathcal{P}(U)} \longrightarrow \{0,1\}
            \,\left\vert
            \begin{array}{c} i \in U, \\ j \in \mathcal{C}_{i} \end{array}
            \right.
            \!\!\right\}
    $\,
    is a collection of $\{0,1\}$-valued random variables
    % defined on \,$\Omega$,\,
    % where \,$\mathcal{C}_{i}$\, is a finite set, for each \,$i \in U$.
    % is a $\{0,1\}^{N \times 1}$-valued random variable
    defined on \,$\Omega \times \mathcal{P}(U)$,\,
    i.e.,
    $\left(\mathscr{A} \overset{{\color{white}.}}{\otimes} \mathcal{P}(\mathcal{P}(U))\right)$-measurable maps, that satisfies:
    \begin{equation*}
    (\,\lambda\otimes\pi\,)\!\left(\;\,
        \underset{i \in U}{\bigcup}\left\{\;
            (\omega,s) \in \Omega\times\mathcal{P}(U)
            \;\left\vert
            \begin{array}{c}
            S_{i}(s) = 0, %\,\textnormal{and}\;
            \\
            \overset{{\color{white}.}}{R_{ij}(\omega,s) = 1},\;
            \textnormal{for some $j \in \mathcal{C}_{i}$}
            \end{array}
            \right.
            \!\!\right\}
        \;\right)
    \; = \;
        0
    \end{equation*}
    Note that this condition stipulates that,
    whenver a PSU \,$i \in U$\, is not selected in the (random) sample \,$s \subset U$,\,
    then none of its SSUs can be a respondents.
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Interpretation}
\begin{itemize}
\item
    $U = \{1,2,\ldots,N\}$
    \,--\,
    index set of the target (finite) population of primary sampling units (PSU),
    and \,$N$\, is the (finite) size of the target population of PSUs.
\item
    $\left(\,\Omega,\mathscr{A},\lambda\,\right)$
    \,--\,
    underlying source of stochasticity of the imputation model (or superpopulation model).
\item
    $\Ydot$
    \,--\,
    $\Re$-valued target variable (population characteristic defined at the level of the SSUs).
\item
    $\Xdotdot$
    \,--\,
    $\Re^{r}$-valued auxiliary variable (to be used as predictor variables for $\Ydot$). 
\item
    $\Sdot$
    \,--\,
    $\{0,1\}$-valued binary variable indicating, for each \,$s \subset U$,\,
    whether a population unit \,$k \in U$\, is selected in \,$s$.\,
    More precisely, for each \,$s \subset U$,\, we have:
    \begin{equation*}
    S_{k}(s)
    \;\; = \;\;
        \left\{\begin{array}{cl}
            1\,, & \textnormal{if \,$k \in s$},
            \\
            0\,, & \textnormal{otherwise}.
            \end{array}\right.
    \end{equation*}
\item
    $\Rdot$
    \,--\,
    $\{0,1\}$-valued binary variable indicating
    % for almost every outcome
    % \,$\omega \in \Omega$\,
    % and each
    % \,$(s,s_{R}) \in \mathcal{P}(U) \times \mathcal{P}(U)$\,
    % satisfying
    % \,$s_{R} \subset s$,\,
    whether an SSU in a selected PSU \,$i \in s$\,
    is a respondent SSU or not.
    % More precisely, for each \,$s \subset U$ and each \,$k \in s$,\, we have:
    % \begin{equation*}
    % R_{k}(\omega,s_{R})
    % \;\; = \;\;
    %     \left\{\begin{array}{cl}
    %         1\,, & \textnormal{if \,$k \in s_{R}$},
    %         \\
    %         0\,, & \textnormal{otherwise}.
    %         \end{array}\right.
    % \end{equation*}
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Probabilistic assumptions}
\begin{itemize}
\item
    The unit-level selection probabilities are all strictly positive, i.e.,
    \begin{equation*}
    \pi_{i}
    \; := \;
        P(\,S_{i} = 1\,)
    \; := \;
        \pi\!\left(\,\left\{\,
            \left.
            s \overset{{\color{white}.}}{\subset} U
            \,\right\vert\,
            i \in s
            \,\right\}
        \,\right)
    \;\; > \;\;
        0,\,
    \textnormal{for each \,$i \in U$\,
    % and \,$w_{i} \,:=\, \dfrac{1}{P(\,S_{i} = 1\,)}$,\, for each \,$i \in U$.
    }
    \end{equation*}
\item
    The second-order joint selection probabilities are all strictly positive, i.e.,
    \begin{equation*}
    \pi_{ij}
    \; := \;
        P(\,S_{i} = S_{j} = 1\,)
    \; := \;
        \pi\!\left(\,\left\{\,
            \left.
            s \overset{{\color{white}.}}{\subset} U
            \,\right\vert\,
            i, j \in s
            \,\right\}
        \,\right)
    \;\; > \;\;
        0,\,
    \textnormal{for each \,$i,j \in U$,\, with $i \neq j$}
    \end{equation*}
\item
    $\left(\,\Ydot\,,\Xdotdot\,\right)$\, and \,$\Sdot$\, are (unconditionally) independent.
\item
    $\Ydot$\, and \,$(\,\Sdot\,,\Rdot\,)$\, are {\color{red}conditionally independent} given \,$\Xdotdot$.
\item
    $\Cov\!\left(\;Y_{ij}\,,\,Y_{kl}\;\left\vert\;\Xdotdot\right.\right) \,=\, 0$,\,
    for each \,$(i,j) \neq (k,l)$.
\end{itemize}

\vskip 0.2cm
\noindent

            W^{(+)}_{\alpha,0}
            \; + \;
            \underset{i \in {\color{black}s}}{\sum}\;
            \underset{j \in (\mathcal{C}_{i})_{R}}{\sum}\!\!
            \left(\,
                \dfrac{
                    \alpha_{ij} %Y_{ij}
                    }{
                    \pi_{i}
                    }
                \; + \,
                W^{(+)}_{\alpha,ij}
                \,\right)
            Y_{ij}
            \,\right)
        \; - \;
        \left(\;
            \underset{i \in s}{\sum}\;
            \underset{j \in \mathcal{C}_{i}}{\sum}\,
            \dfrac{\alpha_{ij}Y_{ij}}{\pi_{i}}
            \right)
    \\
    & = &
        W^{(+)}_{\alpha,0}
        \; + \;
            \underset{i \in {\color{black}s}}{\sum}\;
            \underset{j \in (\mathcal{C}_{i})_{R}}{\sum}\!
            \dfrac{
                \alpha_{ij}Y_{ij}
                }{
                \pi_{i}
                }
            \; + \,
            \underset{i \in {\color{black}s}}{\sum}\,
            \underset{j \in (\mathcal{C}_{i})_{R}}{\sum}\!
            W^{(+)}_{\alpha,ij}Y_{ij}
        \; - \;
        \underset{i \in s}{\sum}\;
        \underset{j \in \mathcal{C}_{i}}{\sum}\,
        \dfrac{\alpha_{ij}Y_{ij}}{\pi_{i}}
    \\
    & = &
        W^{(+)}_{\alpha,0}
        \; + \,
        \underset{i \in {\color{black}s}}{\sum}\,
        \underset{j \in (\mathcal{C}_{i})_{R}}{\sum}\!
        W^{(+)}_{\alpha,ij}\,Y_{ij}
        \; - \;
        \underset{i \in {\color{black}s}}{\sum}\;
        \underset{j \in (\mathcal{C}_{i})_{M}}{\sum}\!
        \dfrac{ \alpha_{ij}Y_{ij} }{ \pi_{i} }\,,
    \end{eqnarray*}
    as required.
    \qed
\end{enumerate}

\begin{lemma}[Unbiased estimator for nonresponse variance; Equations (5.9) \& (5.10) in \cite{Beaumont2011}]
\label{GeneralizedTotalNonresponseVariance}
\mbox{}
\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{GeneralizedTotalFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{GeneralizedTotalProbabilisticFramework}.
\vskip 0.1cm

Then, the following statements are true:
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\mbox{}\;\;\,(\theenumi)$\quad$}
\begin{enumerate}
\item

        \left.
        (\,\widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\,)^{2}
        \,\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    $\,

    \,$

        \right]

\item

        (\,\widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\,)^{2}
        \,\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    $\,

    \begin{eqnarray*}
    E_{Y}\!\left[\,

\end{lemma}
\proof
\begin{enumerate}
\item
    Nothing to prove.
\item
    \textbf{Claim 1:}\quad

    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\,)^{2}
        \,\right\vert\,
        s,s_{R},\Xobs
        \,\right]

            \left.
            \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}

            \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}

    Proof of Claim 1:\;\;

    \begin{equation*}

        E\!\left[\,\overset{{\color{white}.}}{(X - \mu_{X})^{2}}\,\right]
    \; = \;
        E\!\left[\,\overset{{\color{white}.}}{X^{2} - 2\,\mu_{X}\,X + \mu_{X}^{2}}\,\right]
    \; = \;
        \cdots
    \; = \;
        E\!\left[\,X^{2}\,\right] \,-\, \mu_{X}^{2}\,,

    This completes the proof of Claim 1.

    \vskip 0.5cm

    \begin{equation*}

        \,\right\vert\,
        s,s_{R},\Xobs
        \,\right]

        \underset{l \in s_{R}}{\sum}
            \left(W^{(+)}_{\alpha,kl}\right)^{2}\sigma_{l}^{2}

            w_{k}^{2}\,d_{k}^{2}\,\sigma_{k}^{2}

    \vskip 0.1cm
    \noindent
    Proof of Claim 2:\;\;
    \begin{eqnarray*}
    &&
        \Var_{Y}\!\left[\,
            \left.
            \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \\
    & = &
        \Var_{Y}\!\left[\,
            \left.
            W^{(+)}_{\alpha,0}
            \,+\,
            \underset{l \in s_{R}}{\sum}\,W^{(+)}_{\alpha,kl}\,Y_{l}
            \,-\,
            \underset{k \in s_{M}}{\sum}\;w_{k}\,d_{k}\,Y_{k}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right],
        \;\;
        \textnormal{by Lemma \ref{GeneralizedTotalImputationError}(ii)}
    \\
    & = &
        \Var_{Y}\!\left[\;
            \left.
            \underset{l \in s_{R}}{\sum}\,W^{(+)}_{\alpha,kl}\,Y_{l}
            \,-\,
            \underset{k \in s_{M}}{\sum}\;w_{k}\,d_{k}\,Y_{k}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right],
        \;\;
        \textnormal{since $W^{(+)}_{\alpha,0}$ is constant upon conditioning on $s, s_{R}$}
    \\
    & = &
        \underset{l \in s_{R}}{\sum}\,
        \underset{m \in s_{R}}{\sum}\,
        \Cov_{Y}\!\left[\;
            W^{(+)}_{\alpha,kl}\,Y_{l}
            \,,
            W^{(+)}_{m}\,Y_{m}
            \,\left\vert
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right]
        \; + \;
        \underset{k \in s_{M}}{\sum}\,
        \underset{m \in s_{M}}{\sum}\,
        \Cov_{Y}\!\left[\;
            w_{k}\,d_{k}\,Y_{k}
            \,,
            w_{m}\,d_{m}\,Y_{m}
            \,\left\vert
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right]
    \\
    &&
        \; - \;
        2 \cdot
        \underset{l \in s_{R}}{\sum}\;

            w_{k}\,d_{k}\,Y_{k}

            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}

        \;\;
        \textnormal{by Lemma \ref{sRsMPullOutOfVariance}}
    \\

            \!\right]
        \,\;+\;
        \underset{k \in s_{M}}{\sum}

            Y_{k}
            \,\left\vert
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.

    &&
        \quad\quad\quad
        \textnormal{since

            \Cov\!\left(\,Y_{a},Y_{b}\,\left\vert\,s,s_{R},\Xobs\right.\,\right)
            \,=\,

            }
    \\
    & = &
        \underset{l \in s_{R}}{\sum}

        \Var_{Y}\!\left[\;
            Y_{l}

            \right]

            \,\left\vert\,
            \Xobs
            \right.
            \,\right],

        \;\;
        \textnormal{by definition of

                \Xobs
                \right.
                \,\right]$}

    \vskip 0.1cm
    \noindent
    Proof of Claim 3:\;\;
    \begin{eqnarray*}
    &&
        E_{Y}\!\left[\,
            \left.
            \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}

            s,s_{R},\Xobs
            \,\right]

        \underset{l \in s_{R}}{\sum}\;W^{(+)}_{\alpha,kl}\,\mu_{l}

    \end{eqnarray*}
    This completes the proof of Claim 3.

    \vskip 0.3cm
    \noindent
    The required result now follows immediately from Claim 1, Claim 2, and Claim 3.
    \qed
\end{enumerate}

\vskip 0.1cm
\begin{lemma}[Unbiased estimator for sampling-nonresponse covariance]
\label{GeneralizedTotalMixedVariance}
\mbox{}
\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{GeneralizedTotalFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{GeneralizedTotalProbabilisticFramework}.
\vskip 0.1cm
\noindent
Then, the following statements are true:
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\mbox{}\;\;\,(\theenumi)$\quad$}
\begin{enumerate}
\item
    $
    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\,)
        (\,  \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}\,)
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    $\,
    is an $(\Sdot,\Rdot)$-unbiased estimator for the mixed variance\\
    \,$
    E_{SR}\!\left[\,
    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\,)
        (\,  \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}\,)
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
        \right]
    $.
\item
    The above estimator can be expressed as follows:
    \begin{eqnarray*}
    &&
        E_{Y}\!\left[\,
            \left.
            (\,\widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\,)
            (\,  \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}\,)
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \\
    & = &
        \Cov_{Y}\!\left(\,
            \left.
            \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \,,\,
            \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right)
    \\
    &&
        +\;
        E_{Y}\!\left[\;
            \left.
            \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]
        \cdot
        {\color{red}E_{Y}\!\left[\;
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]}
    \end{eqnarray*}
\item
    The above covariance term can be expressed in explicit computable form as follows:
    \begin{eqnarray*}
    \Cov_{Y}\!\left(\,
        \left.
        \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
        \,,\,
        \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right)
    & = &
        \underset{l \in s_{R}}{\sum}\,
            W^{(+)}_{\alpha,kl}\,(w_{k}-1)\,d_{l}\,\sigma_{l}^{2}
        \,-\,
        \underset{k \in s_{M}}{\sum}\,
            w_{k}\,(w_{1}-1)\,d_{k}^{2}\,\sigma_{k}^{2}
    \end{eqnarray*}
\end{enumerate}
\end{lemma}
\proof
\begin{enumerate}
\item
    Nothing to prove.
\item
    The required result follows from the following general fact:
    \begin{equation*}
    \Cov(\,X,Y\,)
    \;\; := \;\;
        E\!\left[\,\overset{{\color{white}.}}{(X - \mu_{X})(Y - \mu_{Y})}\,\right]
    \;\; = \;\;
        \cdots
    \;\; = \;\;
        E\!\left[\,\overset{{\color{white}.}}{XY}\,\right]
        \; - \;
        \mu_{X}\,\mu_{Y}\,,
    \end{equation*}
    which implies
    \begin{equation*}
    E\!\left[\,\overset{{\color{white}.}}{XY}\,\right]
    \;\; = \;\;
        \Cov(\,X,Y\,)
        \; + \;
        \mu_{X}\,\mu_{Y}
    \end{equation*}
\item
    We compute:
    \begin{eqnarray*}
    &&
        \Cov_{Y}\!\left(\,
            \left.
            \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \,,\,
            \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right)
    \\
    & = &
        \Cov_{Y}\!\left(\,
            \left.
            W^{(+)}_{\alpha,0}
            \,+\,
            \underset{l \in s_{R}}{\sum}\,W^{(+)}_{\alpha,kl}\,Y_{l}
            \;-
            \underset{k \in s_{M}}{\sum}\,w_{k}\,d_{k}\,Y_{k}
            \;,\;
            \underset{k \in s}{\sum}\;w_{k}d_{k}Y_{k}
            \;-
            \underset{k \in U}{\sum}\;d_{k}Y_{k}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right),
            \quad
            \textnormal{by Lemma \ref{GeneralizedTotalImputationError}(ii)}
    \\
    & = &
        \Cov_{Y}\!\left(\;
            \left.
            \underset{l \in s_{R}}{\sum}\,W^{(+)}_{\alpha,kl}\,Y_{l}
            \;-
            \underset{k \in s_{M}}{\sum}\,w_{k}\,d_{k}\,Y_{k}
            \;,\;
            \underset{k \in s}{\sum}\;(w_{k}-1)\,d_{k}Y_{k}
            \;-
            \underset{k \in U \backslash s}{\sum}d_{k}Y_{k}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right)
    \\
    & = &
        \underset{l \in s_{R}}{\sum}\;
        \underset{k \in s}{\sum}\;
        \Cov_{Y}\!\left(\;
            W^{(+)}_{\alpha,kl}\,Y_{l}
            \;,\,
            (w_{k}-1)\,d_{k}Y_{k}
            \,\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right)
        \; - \;
        \underset{l \in s_{R}}{\sum}\;
        \underset{k \in U \backslash s}{\sum}\,
        \Cov_{Y}\!\left(\;
            W^{(+)}_{\alpha,kl}\,Y_{l}
            \;,\,
            d_{k}Y_{k}
            \,\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right)
    \\
    &&
        \; - \;
        \underset{k \in s_{M}}{\sum}\;
        \underset{l \in s}{\sum}\;
        \Cov_{Y}\!\left(\;
            w_{k}\,d_{k}\,Y_{k}
            \;,\,
            (w_{l}-1)\,d_{l}Y_{l}
            \,\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right)
        \; + \;
        \underset{k \in s_{M}}{\sum}\;
        \underset{l \in U \backslash s}{\sum}\,
        \Cov_{Y}\!\left(\;
            w_{k}\,d_{k}\,Y_{k}
            \;,\,
            d_{l}Y_{l}
            \,\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right),
    \\
    &&
        \quad\quad\quad
        \textnormal{by Lemma \ref{sRsMPullOutOfCovariance}}
    \\
    & = &
        \underset{l \in s_{R}}{\sum}\,
        W^{(+)}_{\alpha,kl}\,(w_{l}-1)\,d_{l}\,\Var_{Y}\!\left[\;
            Y_{l}
            \;\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right]
        \;-\;
        \underset{k \in s_{M}}{\sum}\,
        w_{k}\,(w_{k}-1)\,d_{k}^{2}\,\Var_{Y}\!\left[\;
            Y_{k}
            \;\left\vert\!
            \begin{array}{c} s,s_{R}, \\ \Xobs \end{array}
            \right.
            \!\right],
    \\
    &&
        \quad\quad\quad
        \textnormal{since
            \,$
            \Cov\!\left(\,Y_{a},Y_{b}\,\left\vert\,s,s_{R},\Xobs\right.\right)
            \,=\,
            \Cov\!\left(\,Y_{a},Y_{b}\,\left\vert\,\Xobs\right.\right)
            \,=\,
            0
            $,\, whenever \,$a \neq b$}
    \\
    & = &
        \underset{l \in s_{R}}{\sum}\,
        W^{(+)}_{\alpha,kl}\,(w_{l}-1)\,d_{l}\,\Var_{Y}\!\left[\,
            Y_{l}
            \left\vert\,
            \Xobs
            \right.
            \right]
        \,-\,
        \underset{k \in s_{M}}{\sum}\,
        w_{k}\,(w_{k}-1)\,d_{k}^{2}\,\Var_{Y}\!\left[\,
            Y_{k}
            \left\vert\,
            \Xobs
            \right.
            \right],
        \;\;
        \textnormal{since $\Ydot \perp (\Sdot,\Rdot) \;\vert\; \Xobs$}
    \\
    & = &
        \underset{l \in s_{R}}{\sum}\,
        W^{(+)}_{\alpha,kl}\,(w_{l}-1)\,d_{l}\,\sigma_{l}^{2}
        \;\;-\;
        \underset{k \in s_{M}}{\sum}\,
        w_{k}\,(w_{k}-1)\,d_{k}^{2}\,\sigma_{k}^{2}
    \end{eqnarray*}
    % where
    % the third equality follows from the conditional independence
    % \,$Y_{k} \perp Y_{l} \;\,\vert\; \Xobs$ whenever \,$k \neq l$,\,
    % while
    % the fourth equality follows from the conditional independence
    % \,$\Ydot \perp (\Sdot,\Rdot) \;\vert\; \Xobs$.\,
    \vskip 0.1cm
    \noindent
    This completes the proof of the Lemma.
    \qed
\end{enumerate}

\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{GeneralizedTotalFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{GeneralizedTotalProbabilisticFramework}.
\vskip 0.1cm
\noindent
Then, the following summarizes the results in \cite{Beaumont2011}:
\begin{itemize}
\item
    By Lemma \ref{DecompositionMSETtilde}, we have the following decomposition:
    \begin{eqnarray*}
    \MSE\!\left(\;
        \left.
        \widetilde{T}_{\alpha Y}
        \;\right\vert\,
        \Xobs
        \,\right)
    & := &
        E\!\left[
            \left.
            \left(\,\widetilde{T}_{\alpha Y} - T_{\alpha Y}\right)^{2}
            \,\right\vert\,
            \Xobs
            \,\right]
    \\
    & = &
        E_{Y}\!\left[\,
        \Var_{SR}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \,\right\vert\,
            \Ydot,
            \Xobs
            \,\right)
            \,\right]
        \;+\;
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\right)^{2}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \\
    &&
        +\;2\cdot
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{\alpha Y} \,-\, \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\right)
            \!\!
            \left(\,\widehat{T}^{\textnormal{(HT)}}_{\alpha Y} \,-\, T_{\alpha Y}\right)
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \end{eqnarray*}
    We use the following nomenclature to refer the three summands in the right-hand side:
    \begin{eqnarray*}
    \left(\begin{array}{c}
        \textnormal{sampling}
        \\
        \textnormal{\;\,variance component\;\,}
        \end{array}\right)
    & := &
        E_{Y}\!\left[\,
        \Var_{SR}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \,\right\vert\,
            \Ydot,
            \Xobs
            \,\right)
            \,\right]
    \\
    \left(\begin{array}{c}
        \textnormal{nonresponse}
        \\
        \textnormal{\;\,variance component\;\,}
        \end{array}\right)
    & := &
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\right)^{2}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \\
    \left(\begin{array}{c}
        \textnormal{sampling-nonresponse}
        \\
        \textnormal{covariance}
        \end{array}\right)
    & := &
        2\cdot
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{\alpha Y} \,-\, \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\,\right)
            \!\!
            \left(\,\widehat{T}^{\textnormal{(HT)}}_{\alpha Y} \,-\, T_{\alpha Y}\,\right)
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \end{eqnarray*}
\item
    By Lemma \ref{GeneralizedTotalSamplingVariance},
    the sampling variance component can be estimated as follows:
    \begin{eqnarray*}
    \left(\!\begin{array}{c}
        \textnormal{sampling}
        \\
        \textnormal{variance component}
        \end{array}\!\right)
    & := &
        E_{Y}\!\left[\,
        \Var_{SR}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \,\right\vert\,
            \Ydot,
            \Xobs
            \,\right)
            \,\right]
    \;\; = \;\;
        E_{SR}\,
        E_{Y_{R}}\,
        E_{Y_{M}}\!\left[\;
            \widehat{V}^{\textnormal{(HT)}}(\,s\,;\dbullet\Ydot\,)
            \,\left\vert\!
            \begin{array}{c} s,s_{R},\YsubR, \\ \Xobs \end{array}
            \right.
            \!\!\right]
    \\
    & {\color{red}\approx} &
    E_{Y_{M}}\!\left[\;
        \widehat{V}^{\textnormal{(HT)}}(\,s\,;\dbullet\Ydot\,)
        \,\left\vert\!
        \begin{array}{c} s,s_{R},\YsubR, \\ \Xobs \end{array}
        \right.
        \!\!\right]
    \\
    & = &
        {\color{blue}
        \widehat{V}^{\textnormal{(HT)}}(\,s\,;\dbullet\widetilde{y}_{\bullet}\,)
        \;+\;
        \underset{k \in s_{M}}{\sum}\;
        d_{k}
        \left(1-\dfrac{1}{w_{k}}\right)
        w_{k}^{2}\,\sigma_{k}^{2}
        }\,,
    \end{eqnarray*}
    where
    \,$\widehat{V}^{\textnormal{(HT)}}(\,s\,;\dydot\,)$\,
    is the (design-unbiased) Horvitz-Thompson estimator for sampling variance, i.e.,
    \begin{equation*}
    \widehat{V}^{\textnormal{(HT)}}(\,s\,;\dydot\,)
    \;\; := \;\;
        \underset{k \in s}{\sum}\,
        \underset{l \in s}{\sum}\,
        \dfrac{\pi_{kl} - \pi_{k}\pi_{l}}{\pi_{kl}}\,
        (w_{k}d_{k}y_{k})\,(w_{l}d_{l}y_{l})\,,
    \end{equation*}
    and
    \begin{equation*}
    \widetilde{y}_{k}
    \;\; := \;\;
        \left\{\begin{array}{cl}
            y_{k}, & \textnormal{for \,$k \,\in\, s_{R}$}
            \\
            \overset{{\color{white}\textnormal{\large1}}}{\mu_{k}}, & \textnormal{for \,$k \,\in\, s_{M} \,:=\, s \,\backslash\, s_{R}$}
            \end{array}\right.
    \end{equation*}
\item
    By Lemma \ref{GeneralizedTotalNonresponseVariance},
    the nonresponse variance component can be estimated as follows:
    \begin{eqnarray*}
    \left(\!\begin{array}{c}
        \textnormal{nonresponse}
        \\
        \textnormal{variance}
        \\
        \textnormal{{\color{white}..}component{\color{white}..}}
        \end{array}\!\right)
    & := &
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\right)^{2}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \;\; {\color{red}\approx} \;\;
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\right)^{2}
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \\
    & = &
        {\color{blue}
        \left(\;
            \underset{l \in s_{R}}{\sum}
                \left(W^{(+)}_{\alpha,kl}\right)^{2}\sigma_{l}^{2}
            \,+\,
            \underset{k \in s_{M}}{\sum}\,
                w_{k}^{2}\,d_{k}^{2}\,\sigma_{k}^{2}
            \,\right)
        \;+\;
        \left(\,
            W^{(+)}_{\alpha,0}
            \,+\,
            \underset{l \in s_{R}}{\sum}\;W^{(+)}_{\alpha,kl}\,\mu_{l}
            \,-\,
            \underset{k \in s_{M}}{\sum}\;w_{k}\,d_{k}\,\mu_{k}
            \,\right)^{2}
        }
    \end{eqnarray*}
\item
    By Lemma \ref{GeneralizedTotalMixedVariance},
    the sampling-nonresponse covariance component can be estimated as follows:
    \begin{eqnarray*}
    \left(\!\begin{array}{c}
        \textnormal{sampling-nonresponse}
        \\
        \textnormal{covariance}
        \end{array}\!\right)
    & := &
        2\cdot
        E_{SR}\!\left[\,
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{\alpha Y} \,-\, \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\right)
            \!\!
            \left(\,\widehat{T}^{\textnormal{(HT)}}_{\alpha Y} \,-\, T_{\alpha Y}\right)
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
            \,\right]
    \\
    & \approx &
        2\cdot
        E_{Y}\!\left[
            \left.
            \left(\,\widetilde{T}_{\alpha Y} \,-\, \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}\right)
            \!\!
            \left(\,\widehat{T}^{\textnormal{(HT)}}_{\alpha Y} \,-\, T_{\alpha Y}\right)
            \,\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \\
    & = &
        2\cdot
        \Cov_{Y}\!\left(\,
            \left.
            \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \,,\,
            \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right)
    \\
    &&
        +\;
        2\cdot
        E_{Y}\!\left[\;
            \left.
            \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]
        \cdot
        {\color{red}E_{Y}\!\left[\;
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]}
    \\
    & {\color{red}\approx} &
        2\cdot
        \Cov_{Y}\!\left(\,
            \left.
            \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
            \,,\,
            \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right)
    \\
    & = &
        {\color{blue}
        2\,\cdot\!\left(\;
            \underset{l \in s_{R}}{\sum}\,
                W^{(+)}_{\alpha,kl}\,(w_{l}-1)\,d_{l}\,\sigma_{l}^{2}
            \;\,-\;
            \underset{k \in s_{M}}{\sum}\,
                w_{k}\,(w_{1}-1)\,d_{k}^{2}\,\sigma_{k}^{2}
            \,\right)
        }
    \end{eqnarray*}
\item
    \textbf{Important Remark:}
    \vskip 0.05cm
    \noindent
    Note that the above estimation/approximation of the mixed variance component
    disregards the following term:
    \begin{equation*}
    2\,\cdot E_{Y}\!\left[\;
        \left.
        \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    \cdot
    {\color{red}E_{Y}\!\left[\;
        \left.
        \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]}
    \end{equation*}
    The reason this has to be done in practice is that, in practice,
    there is usually insufficient data/information to compute or estimate
    the quantity
    \,$
    E_{Y}\!\left[\;
        \left.
        \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    $\,
    (indicated in red above), as \,$T_{\alpha Y}$\, involves population units
    that are not selected in the sample.
    \vskip 0.1cm
    \noindent
    Obviously, the above estimation/approximation of the mixed variance component
    is acceptable only if the quantity
    \,$
    2\,\cdot E_{Y}\!\left[\;
        \left.
        \widetilde{T}_{\alpha Y} - \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    \cdot
    {\color{black}E_{Y}\!\left[\;
        \left.
        \widehat{T}^{\textnormal{(HT)}}_{\alpha Y} - T_{\alpha Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]}
    $\,
    is small.
    \vskip 0.1cm
    \noindent
    For this, in practice, we thus {\color{red}tacitly assume} that the quantity
    \,$
    E_{Y}\!\left[\;
        \left.
        \widetilde{T}_{\alpha Y} \overset{{\color{white}.}}{-} \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    $\,
    is small;
    recall that this quantity is called the \textbf{conditional model bias}
    (see Lemma \ref{biasTtilde}), and
    it can be expressed as follows:
    \begin{eqnarray*}
    E_{Y}\!\left[\;
        \left.
        \widetilde{T}_{\alpha Y} \overset{{\color{white}.}}{-} \widehat{T}^{\textnormal{(HT)}}_{\alpha Y}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    & = &
        \underset{k\in{\color{red}s_{M}}}{\sum}\,d_{k}\,w_{k}\,
        E_{Y}\!\left[\;
            \left.
            Y^{*}_{k} \overset{{\color{white}.}}{-} Y_{k}
            \;\right\vert\,
            s,s_{R},\Xobs
            \,\right]
    \end{eqnarray*}
    This harks back to our earlier remark (Remark \ref{remarkSmallUnitBias}) that,
    in practice, we would like to choose an imputation strategy such that
    \begin{equation*}
    E_{Y}\!\left[\;
        \left.
        Y^{*}_{k} \overset{{\color{white}.}}{-} Y_{k}
        \;\right\vert\,
        s,s_{R},\Xobs
        \,\right]
    \;\; \approx \;\;
        0,
    \quad
    \textnormal{for each \,$k \in s_{M}$}
    \end{equation*}
    See the paragraph after Equation (5.2), p.175, \cite{Beaumont2011}.
\end{itemize}
