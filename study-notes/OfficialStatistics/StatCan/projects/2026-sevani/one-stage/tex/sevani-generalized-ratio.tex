\subsection{Formulation of the target variability quantification problem}
\label{GeneralizedRatioFormulationOfVarianceEstimationProblem}
\setcounter{theorem}{0}

        %%%%% ~~~~~~~~~~~~~~~~~~~~ %%%%%

Suppose:
\begin{itemize}
\item
    We would like to estimate
    -- using data from a one-stage cluster sample --
    a domain ratio (the ratio of two population totals)
    \begin{equation*}
    R_{\alpha,Y/Z}
    \;\; := \;\;
        \dfrac{T_{\alpha,Y}}{\overset{{\color{white}.}}{T_{\alpha,Z}}}
    \;\; = \;\;
        \dfrac{
            {\color{white}.}
            \underset{i \in U}{\sum}\;
            \underset{j \in \mathcal{C}_{j}}{\sum}\;
            \alpha_{ij}\,Y_{ij}
            {\color{white}.}
            }{
            \overset{{\color{white}.}}{
                \underset{i \in U}{\sum}\;
                \underset{j \in \mathcal{C}_{j}}{\sum}\;
                \alpha_{ij}\,Z_{ij}
                }
            }\,,
    \end{equation*}
    where
    \begin{itemize}
    \item
        ${\color{red}U} = \{\,1,2,\ldots,N\,\} \subset \N$\,
        denotes the population of the primary sampling units (PSU),
    \item
        for each \,$i \in U$,\,
        \,${\color{red}\mathcal{C}_{j}}$\,
        denotes the collection of all secondary sampling units (SSU) belonging to PSU \,$i \in U$,
    \item
        $
        {\color{red}\Ydot} \, = \, \left\{\,
            \left.
            \overset{{\color{white}.}}{Y_{ij}}
            \,\right\vert\,
            i \in U, j \in \mathcal{C}_{i}
            \,\right\}
        $\,
        and
        \,$
        {\color{red}\Zdot} \, = \, \left\{\,
            \left.
            \overset{{\color{white}.}}{Z_{ij}}
            \,\right\vert\,
            i \in U, j \in \mathcal{C}_{i}
            \,\right\}
        $\,
        are $\Re$-valued population characteristics defined at the level of the SSUs, and
    \item
        $
        {\color{red}\abullet} \, = \, \left\{\,
            \left.
            \overset{{\color{white}1}}{\alpha_{ij}}
            \;\right\vert\,
            i \in U, j \in \mathcal{C}_{i}
            \,\right\}
        $\,
        is the collection of coefficients defined at the level of the SSUs.
    \end{itemize}
\item
    Aside/Reminder:
    If there were no missing data for the units in the (random) selected sample,
    then the \textit{Horvitz-Thompson domain ratio estimator}
    \begin{eqnarray*}
    \widehat{R}^{\textnormal{(HT)}}_{\alpha,Y/Z}
    & = &
        \widehat{R}^{\textnormal{(HT)}}_{\alpha,Y/Z}(\,
            \Sdot\,,\,
            \Ydot\,,\,
            \Zdot\,;\,
            \abullet
            \,)
    \;\; := \;\;
        \dfrac{
            \widehat{T}^{\textnormal{(HT)}}_{\alpha,Y}
            }{
            \overset{{\color{white}.}}{
                \widehat{T}^{\textnormal{(HT)}}_{\alpha,Z}
                }
            }
    \;\; :=: \;\;
        \dfrac{
            \widehat{T}^{\textnormal{(HT)}}(s\,;\abullet,\Ydot)
            }{
            \overset{{\color{white}.}}{
                \widehat{T}^{\textnormal{(HT)}}(s\,;\abullet,\Zdot)
                }
            }
    \\
    & = &
        \dfrac{
            {\color{white}.}
            \underset{i \in s}{\sum}\;
                \dfrac{1}{\pi_{i}}\!
                \left(\;
                    \underset{j \in \mathcal{C}_{j}}{\sum}\;
                    \alpha_{ij}Y_{ij}
                    \right)
            {\color{white}.}
            }{
            \overset{{\color{white}.}}{
                \underset{i \in s}{\sum}\;
                    \dfrac{1}{\pi_{i}}\!
                    \left(\;
                        \underset{j \in \mathcal{C}_{j}}{\sum}\;
                        \alpha_{ij}Z_{ij}
                        \right)
                }
            }
    \;\; = \;\;
        \dfrac{
            {\color{white}.}
            \underset{i \in U}{\sum}\;
                \dfrac{S_{i}}{\pi_{i}}\!
                \left(\;
                    \underset{j \in \mathcal{C}_{j}}{\sum}\;
                    \alpha_{ij}Y_{ij}
                    \right)
            {\color{white}.}
            }{
            \overset{{\color{white}.}}{
                \underset{i \in U}{\sum}\;
                    \dfrac{S_{i}}{\pi_{i}}\!
                    \left(\;
                        \underset{j \in \mathcal{C}_{j}}{\sum}\;
                        \alpha_{ij}Z_{ij}
                        \right)
                }
            }
    \end{eqnarray*}
    is a common choice as an estimator for
    \,$R_{\alpha,Y/Z}$,\,
    where
    \begin{itemize}
    \item
        $s \subset U$\, is the (random) selected sample, and
        \,$\Sdot \,=\, (S_{1},\ldots,S_{N})^{T}$\,
        is the vector of $\{0,1\}$-valued selection indicator variables,
    \item
        $w_{i} \,:=\, \dfrac{1}{\pi_{i}}$,\, for each population unit \,$i \in U$,\,
        is the (\textit{a priori} known, non-random) design weight of \,$i \in U$,\,
        i.e., the reciprocal of the selection probability \,$\pi_{i}$\,
        (assumed strictly positive) of \,$i \in U$,\,
    \item
        $\widehat{T}^{\textnormal{(HT)}}_{\alpha,\xi} \, :=: \,\widehat{T}^{\textnormal{(HT)}}(s\,;\abullet,\xi_{\bullet})$\,
        is the \textit{Horvitz-Thompson estimator\! \footnote{
        We will use the alternative notations
            \,$\widehat{T}^{\textnormal{(HT)}}_{\alpha,\xi}$\, and \,$\widehat{T}^{\textnormal{(HT)}}(s\,;\abullet,\xi_{\bullet})$\,
            interchangeably, noting that the former is notationally more succinct,
            while the latter allows us to emphasize that
            \,$\xi_{\bullet}$\,
            is being considered as an argument to the formula
            \begin{equation*}
            \widehat{T}^{\textnormal{(HT)}}(s\,;\abullet,\xi_{\bullet})
            \;\; := \;\;
                \underset{i\,\in\,s}{\sum}\;w_{i}\,d_{i}\,\xi_{i}
            \end{equation*}
            for the Horvitz-Thompson population total estimator.
            }
        for the generalized total}
        \,$
        T_{\alpha,\xi}
        \, := \,
            \underset{i \in U}{\sum}\;
            \underset{j \in \mathcal{C}_{j}}{\sum}\;
            \alpha_{ij}\,\xi_{ij}
        $\,
        of an arbitrary $\Re$-valued population characteristic
        \,$\xi_{ij}$ ($i \in U$, $j \in \mathcal{C}_{i}$),\, i.e.,
        \begin{equation*}
        \widehat{T}^{\textnormal{(HT)}}_{\alpha,\xi}
        \;\; :=: \;\;
            \widehat{T}^{\textnormal{(HT)}}(s\,;\abullet,\xi_{\bullet})
        \;\; = \;\;
            \underset{i \in s}{\sum}\;
                \dfrac{1}{\pi_{i}}\!
                \left(\;
                    \underset{j \in \mathcal{C}_{j}}{\sum}\;
                    \alpha_{ij}\,\xi_{ij}
                    \right)
        \;\; = \;\;
            \underset{i \in U}{\sum}\;
                \dfrac{S_{i}}{\pi_{i}}\!
                \left(\;
                    \underset{j \in \mathcal{C}_{j}}{\sum}\;
                    \alpha_{ij}\,\xi_{ij}
                    \right)
        \end{equation*}
    \end{itemize}
    Recall (Result 4.3.1, p.137, \cite{Sarndal1992}) also that
    the sampling variance of the Horvitz-Thompson population total estimator
    under a two-stage cluster sampling design is:
    \begin{equation*}
    \Var_{\,\textnormal{2-stage}}
    \;\; = \;\;
        \Var_{\,\textnormal{PSU}}
        \; + \;
        \Var_{\,\textnormal{SSU}}
    \end{equation*}
    Since we are considering only one-stage cluster sampling
    (i.e., two-stage cluster sampling where the second-stage sampling is a census),
    the subsampling variance \,$\Var_{\,\textnormal{SSU}}$\, is zero.
    Hence, the sampling variance of
    \,$
    \widehat{T}^{\textnormal{(HT)}}_{\alpha,\xi}
    % \,=\,
    %     \widehat{T}^{\textnormal{(HT)}}(s\,;\dbullet\xi_{\bullet})
    $\,
    has only the \,$\Var_{\,\textnormal{PSU}}$\, term, i.e.,
    \begin{equation*}
    \Var_{S}\!\left(\,
        \left.
        \widehat{T}^{\textnormal{(HT)}}_{\alpha,\xi}
        \,\right\vert\!\!
        \begin{array}{c} \xi_{\bullet} \end{array}
        \!\!\right)
    \;\; = \;\;\,
        V^{\textnormal{(HT)}}(\,\abullet,\xi_{\bullet}\,)
    \;\; = \;\;\,
        \underset{i \in U}{\sum}\,
        \underset{k \in U}{\sum}\,
        \left(\,\pi_{ik} \overset{{\color{white}.}}{-} \pi_{i}\pi_{k}\,\right)
        \cdot\!
        \left(\,
            \dfrac{1}{\pi_{i}}\,
            \underset{j\in\mathcal{C}_{i}}{\sum}\;\alpha_{ij}\,\xi_{ij}
            \,\right)
        \!\cdot\!
        \left(\,
            \dfrac{1}{\pi_{i}}\,
            \underset{l\in\mathcal{C}_{k}}{\sum}\;\alpha_{kl}\,\xi_{kl}
            \,\right),
    \end{equation*}
    and the following Horvitz-Thompson variance estimator
    \begin{equation*}
    \widehat{V}^{\textnormal{(HT)}}(\,s\,;\abullet,\xi_{\bullet}\,)
    \;\; := \;\;
        \underset{i \in s}{\sum}\,
        \underset{k \in s}{\sum}\;
        \dfrac{\pi_{ik} - \pi_{i}\pi_{k}}{\pi_{ik}}
        \cdot\!
        \left(\,
            \dfrac{1}{\pi_{i}}\,
            \underset{j\in\mathcal{C}_{i}}{\sum}\;\alpha_{ij}\,\xi_{ij}
            \,\right)
        \!\cdot\!
        \left(\,
            \dfrac{1}{\pi_{i}}\,
            \underset{l\in\mathcal{C}_{k}}{\sum}\;\alpha_{kl}\,\xi_{kl}
            \,\right),
    \end{equation*}
    is a design-unbiased estimator for the sampling variance of
    % \,$\widehat{T}_{d,\xi_{\bullet}}(s) \,=\, \widehat{T}_{d}(s\,;\xi_{\bullet})$,\,
    \,$
    \widehat{T}^{\textnormal{(HT)}}_{\alpha,\xi}
    \,=\,
        \widehat{T}^{\textnormal{(HT)}}(s\,;\abullet,\xi_{\bullet})
    $,\,
    i.e.,
    \begin{equation*}
    E_{S}\!\left[\;
        \left.
        \widehat{V}^{\textnormal{(HT)}}(\,s\,;\abullet,\xi_{\bullet}\,)
        \;\right\vert\!
        \begin{array}{c} \xi_{\bullet} \end{array}
        \!\right]
    \;\; = \;\;
        \Var_{S}\!\left(\,
            \left.
            \widehat{T}^{\textnormal{(HT)}}_{\alpha,\xi} %(s\,;\dbullet\xi_{\bullet})
            \,\right\vert\!
            \begin{array}{c} \xi_{\bullet} \end{array}
            \!\!\right)
    \end{equation*}
\item
    For each selected PSU \,$i \in s \subset U$,\,
    the values of \,$Y_{ij}$\, have been successfully observed
    only for SSUs \,$j \in \mathcal{C}_{i}$\,
    in the (random) ``responding'' subsample of SSUs
    \,${\color{red}(\mathcal{C}_{i})_{R}} \subset \mathcal{C}_{i}$;\,
    in other words, the measurement of \,$Y_{ij}$\, somehow failed for each SSU
    \,$
    j
    \in
        {\color{red}(\mathcal{C}_{i})_{M}}
    \, := \,
        \mathcal{C}_{i} \,\backslash\, (\mathcal{C}_{i})_{R}
    $.\,
    \,$(\mathcal{C}_{i})_{M}$\,
    is called the ``nonresponding'' subsample of the selected PSU \,$i \in s \subset U$.\,
    We will also use
    \,${\color{red}s_{R}} \, := \, \underset{i \in s}{\bigcup}\,(\mathcal{C}_{i})_{R}$\,
    to denote the full (random) collection of selected and responding SSUs.
    \vskip 0.1cm
    On the other hand, we assume that
    the value of \,$Z_{ij}$\, has been successfully observed or measured
    for each selected PSU \,$i \in s \subset U$\, and
    for each SSU \,$j \in \mathcal{C}_{i}$\, in the selected PSU \,$i \in s \subset U$.
\item
    We would like to estimate \,$R_{\alpha,Y/Z}$\,
    with an imputation estimator of the following form:
    \begin{eqnarray*}
    \widetilde{R}_{\alpha,Y/Z}
    & = &
        \widetilde{R}_{\alpha,Y/Z}(\,
            \Sdot\,,\,
            \Rdot\,,\,
            \Ydot\,,\,
            \Zdot\,;\,
            \Xdotdot\,,\,
            \abullet
            \,)
    \;\; := \;\;
        \dfrac{
            \widetilde{T}_{\alpha,Y}
            }{
            \overset{{\color{white}.}}{
                \widehat{T}^{\textnormal{(HT)}}_{\alpha,Z}
                }
            }
    \;\; :=: \;\;
        \dfrac{
            \widehat{T}(s\,;\abullet,Y^{\dagger}_{\bullet})
            }{
            \overset{{\color{white}.}}{
                \widehat{T}^{\textnormal{(HT)}}(s\,;\abullet,\Zdot)
                }
            }
    \\
    & = &
        \dfrac{
            {\color{white}.}
            \underset{i \in s}{\sum}\;
                \dfrac{1}{\pi_{i}}\!
                \left(\;
                    \underset{j \in \mathcal{C}_{j}}{\sum}\;
                    \alpha_{ij}Y_{ij}^{\dagger}
                    \right)
            {\color{white}.}
            }{
            \overset{{\color{white}.}}{
                \underset{i \in s}{\sum}\;
                    \dfrac{1}{\pi_{i}}\!
                    \left(\;
                        \underset{j \in \mathcal{C}_{j}}{\sum}\;
                        \alpha_{ij}Z_{ij}
                        \right)
                }
            }
    % \;\; = \;\;
    %     \dfrac{
    %         \underset{i\,\in\,s}{\sum}\;w_{i}\,d_{i}\,Y^{\dagger}_{i}
    %         }{
    %         \overset{{\color{white}.}}{
    %             \underset{i\,\in\,s}{\sum}\;w_{i}\,d_{i}\,Z_{i}
    %             }
    %         }
    \;\; = \;\;
        \dfrac{
            % \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,d_{i}\left(\,
            %     R_{i}\,Y_{i}
            %     \overset{{\color{white}.}}{+}
            %     (1-R_{i})\,Y^{*}_{i}
            %     \,\right)

                \dfrac{S_{i}}{\pi_{i}}

                    \left(
                        R_{ij}\,Y_{ij}
                        \overset{{\color{white}.}}{+}

                        \right)
                    \right)
            }{
            \overset{{\color{white}.}}{
                % \underset{i\,\in\,U}{\sum}\;w_{i}\,S_{i}\,d_{i}\,Z_{i}
                \underset{i \in U}{\sum}\;
                    \dfrac{S_{i}}{\pi_{i}}\!
                    \left(\;
                        \underset{j \in \mathcal{C}_{j}}{\sum}\;
                        \alpha_{ij}Z_{ij}
                        \right)
                }
            }\,,
    \end{eqnarray*}
    where
    \begin{itemize}
    % \item
    %     $w_{i} \,:=\, \dfrac{1}{\pi_{i}}$,\, for each population unit \,$i \in U$,\,
    %     is the (\textit{a priori} known, non-random) design weight of \,$i \in U$,\,
    %     i.e., reciprocal of the selection probability \,$\pi_{i}$\,
    %     (assumed strictly positive) of \,$i \in U$,\,
    % \item
    %     $\Sdot \,=\, (S_{1},\ldots,S_{N})^{T}$ is the vector of $\{0,1\}$-valued selection indicator variables,
    \item
        $\Rdot \,=\, (R_{1},\ldots,R_{N})^{T}$ is the vector of $\{0,1\}$-valued response indicator variables,
    \item
        $\Xdotdot \,=\, \left(\,X_{ij}\,\right) \,\in\, \overline{\Re}^{N \times r}$\,
        is a random matrix containing the predictor variables for $\Ydot$,
        where the $i^{\textnormal{th}}$ row of \,$\Xdotdot$\, corresponds to populaton unit \,$i \in U$,\,
        and where \,$\overline{R} \,:=\, \Re \sqcup \{\,\texttt{NA}\,\}$\,
        is an extension of $\Re$ alllowing for missing values in \,$\Xdotdot$,\, and
    \item
        for each \,$i \in s$,\, $j \in \mathcal{C}_{i}$,\, we have

\textbf{The variability quantification problem$\overset{{\color{white}1}}{\textnormal{\color{white}.}}$\& proposed solution}
\end{center}
We seek to quantify the variability of
\,$\widetilde{R}_{\alpha,Y/Z}$\,
as an estimator for
\,$R_{\alpha,Y/Z} \; := \; \dfrac{T_{\alpha,Y}}{T_{\alpha,Z}}$.
\vskip 0.1cm
Following \cite{Beaumont2011} (which treats generalized totals),
the conditional mean squared error (MSE) given \,$\Zdot$\, and \,$\Xdotdot$\,
\begin{equation*}
\MSE\!\left(\;
    \left.
    \widetilde{R}_{\alpha,Y/Z}
    \;\right\vert\!
    \begin{array}{c} \Zdot\,,\Xdotdot \end{array}
    \!\right)
\;\; := \;\;
    E\!\left[\;\left(\,
        \left.
        \widetilde{R}_{\alpha,Y/Z}
        \, - \,
        R_{\alpha,Y/Z}
        \,\right)^{\!2}
        \;\right\vert\!
        \begin{array}{c} \Zdot\,,\Xdotdot \end{array}
        \!\right]
\end{equation*}
of
\,$\widetilde{R}_{\alpha,Y/Z}$\,
as estimator for
\,$R_{\alpha,Y/Z}$\,
is proposed for this purpose.
\vskip 0.3cm
\noindent
In addition, an approximate, but computable, estimator for
\,$
\MSE\!\left(\;
    \left.
    \widetilde{R}_{\alpha,Y/Z}
    \;\right\vert\!
    \begin{array}{c} \Zdot\,,\Xdotdot \end{array}
    \!\right)
$\,
will be proposed.

\subsection{Probabilistic framework}
\label{DomainRatioProbabilisticFramework}
\setcounter{theorem}{0}

        %%%%% ~~~~~~~~~~~~~~~~~~~~ %%%%%

\noindent
\textbf{Probability spaces}
\begin{itemize}
\item
    Let
    \,$U \,=\, \{1,2,\ldots,N\}$.\,
    Let
    \,$\mathcal{P}(U) \,=\, \mathcal{P}(\{1,2,\ldots,N\})$\,
    denote the power set (set of all subsets) of \,$U$.\,
    Let
    \,$\left(\,\mathcal{P}(U)\,,\mathcal{P}(\mathcal{P}(U))\,,\overset{{\color{white}-}}{\pi}\,\right)$\,
    be the probability space induced by a sampling design, i.e.,
    a probability mass function defined on
    \,$\mathcal{P}(U)$.\,
\item
    Let \,$\left(\,\Omega,\mathscr{A},\lambda\,\right)$\, be a probability space.
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Random variables}
\begin{itemize}
\item
    $
    \Ydot
    \, = \,
        \left\{\;
            Y_{ij} : \Omega \longrightarrow \Re
            \,\left\vert
            \begin{array}{c} i \in U, \\ j \in \mathcal{C}_{i} \end{array}
            \right.
            \!\!\right\}
    $\,
    and
    \,$
    \Zdot
    \, = \,
        \left\{\;
            Z_{ij} : \Omega \longrightarrow \Re
            \,\left\vert
            \begin{array}{c} i \in U, \\ j \in \mathcal{C}_{i} \end{array}
            \right.
            \!\!\right\}
    $\,
    are two collections of $\Re$-valued random variables
    (i.e., $\mathscr{A}$-measurable maps)
    defined on \,$\Omega$,\,
    where \,$\mathcal{C}_{i}$\, is a finite set, for each \,$i \in U$.
\item
    $
    \Xdotdot
    \, = \,
        \left\{\,
            X_{ij} : \Omega \longrightarrow \Re^{r}
            \left\vert
            \begin{array}{c} i \in U, \\ j \in \mathcal{C}_{i} \end{array}
            \right.
            \!\!\right\}
    $\,
    is a collection of $\Re^{r}$-valued random variables defined on \,$\Omega$.\,
\item
    $\Sdot \,=\, (S_{1},S_{2},\ldots,S_{N})^{T} : \mathcal{P}(U) \longrightarrow \{0,1\}^{N \times 1}$\,
    is a $\{0,1\}^{N \times 1}$-valued random variable defined on \,$\mathcal{P}(U)$,\,
    i.e., $\mathcal{P}(\mathcal{P}(U))$-measurable map.
\item
    % $\Rdot \,=\, (R_{1},R_{2},\ldots,R_{N})^{T} : {\color{red}\Omega \times \mathcal{P}(U)} \longrightarrow \{0,1\}^{N \times 1}$\,
    % \vskip 0.1cm
    $
    \Rdot
    \, = \,
        \left\{\,
            R_{ij} : {\color{red}\Omega \times \mathcal{P}(U)} \longrightarrow \{0,1\}
            \,\left\vert
            \begin{array}{c} i \in U, \\ j \in \mathcal{C}_{i} \end{array}
            \right.
            \!\!\right\}
    $\,
    is a collection of $\{0,1\}$-valued random variables
    % defined on \,$\Omega$,\,
    % where \,$\mathcal{C}_{i}$\, is a finite set, for each \,$i \in U$.
    % is a $\{0,1\}^{N \times 1}$-valued random variable
    defined on \,$\Omega \times \mathcal{P}(U)$,\,
    i.e.,
    $\left(\mathscr{A} \overset{{\color{white}.}}{\otimes} \mathcal{P}(\mathcal{P}(U))\right)$-measurable maps, that satisfies:
    \begin{equation*}
    (\,\lambda\otimes\pi\,)\!\left(\;\,
        \underset{i \in U}{\bigcup}\left\{\;
            (\omega,s) \in \Omega\times\mathcal{P}(U)
            \;\left\vert
            \begin{array}{c}
            S_{i}(s) = 0, %\,\textnormal{and}\;
            \\
            \overset{{\color{white}.}}{R_{ij}(\omega,s) = 1},\;
            \textnormal{for some $j \in \mathcal{C}_{i}$}
            \end{array}
            \right.
            \!\!\right\}
        \;\right)
    \; = \;
        0
    \end{equation*}
    Note that this condition stipulates that,
    whenver a PSU \,$i \in U$\, is not selected in the (random) sample \,$s \subset U$,\,
    then none of its SSUs can be a respondents.
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Interpretation}
\begin{itemize}
\item
    $U = \{1,2,\ldots,N\}$
    \,--\,
    index set of the target (finite) population of primary sampling units (PSU),
    and \,$N$\, is the (finite) size of the target population.
\item
    $\left(\,\Omega,\mathscr{A},\lambda\,\right)$
    \,--\,
    underlying source of stochasticity of the superpopulation model,
    encoding the probabilistic relations among
    \,$\Ydot$,\, $\Zdot$,\, $\Xdotdot$,\, and \,$\Rdot$.\,
\item
    $\Ydot$
    \,--\,
    numerator of the $\Re$-valued target variable
    (population characteristic defined at the level of the SSUs).
\item
    $\Zdot$
    \,--\,
    denominator of the $\Re$-valued target variable
    (population characteristic defined at the level of the SSUs).
\item
    $\Xdotdot$
    \,--\,
    $\Re^{r}$-valued auxiliary variable (to be used as predictor variables for $\Ydot$).
\item
    $\Sdot$
    \,--\,
    $\{0,1\}$-valued binary variable indicating, for each \,$s \subset U$,\,
    whether a primary sampling unit (PSU) \,$k \in U$\, is selected in \,$s$.\,
    More precisely, for each \,$s \subset U$,\, we have:
    \begin{equation*}
    S_{k}(s)
    \;\; = \;\;
        \left\{\begin{array}{cl}
            1\,, & \textnormal{if \,$k \in s$},
            \\
            0\,, & \textnormal{otherwise}.
            \end{array}\right.
    \end{equation*}
\item
    $\Rdot$
    \,--\,
    $\{0,1\}$-valued binary variable indicating
    whether an SSU in a selected PSU \,$i \in s$\,
    is a respondent SSU or not.
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Probabilistic assumptions}
\begin{itemize}
% \item
%     $\pi_{i} \,:=\, P(\,S_{i} = 1\,) \,:=\, \pi\!\left(\,\left\{\,\left. s \overset{{\color{white}.}}{\subset} U \,\right\vert\, i \in s\,\right\}\,\right) \,>\, 0$,\,
%     for each \,$i \in U$,\,
%     and
%     \,$w_{i} \,:=\, \dfrac{1}{P(\,S_{i} = 1\,)}$,\, for each \,$i \in U$.
\item
    The unit-level selection probabilities are all strictly positive, i.e.,
    \begin{equation*}
    \pi_{i}
    \; := \;
        P(\,S_{i} = 1\,)
    \; := \;
        \pi\!\left(\,\left\{\,
            \left.
            s \overset{{\color{white}.}}{\subset} U
            \,\right\vert\,
            i \in s
            \,\right\}
        \,\right)
    \;\; > \;\;
        0,\,
    \textnormal{for each \,$i \in U$\,
    % and \,$w_{i} \,:=\, \dfrac{1}{P(\,S_{i} = 1\,)}$,\, for each \,$i \in U$.
    }
    \end{equation*}
\item
    The second-order joint selection probabilities are all strictly positive, i.e.,
    \begin{equation*}
    \pi_{ij}
    \; := \;
        P(\,S_{i} = S_{j} = 1\,)
    \; := \;
        \pi\!\left(\,\left\{\,
            \left.
            s \overset{{\color{white}.}}{\subset} U
            \,\right\vert\,
            i, j \in s
            \,\right\}
        \,\right)
    \;\; > \;\;
        0,\,
    \textnormal{for each \,$i,j \in U$,\, with $i \neq j$}
    \end{equation*}
\item
    $\left(\,\Ydot\,,\Zdot\,,\Xdotdot\,\right)$\, and \,$\Sdot$\, are (unconditionally) independent.
\item
    % $\left(\,\Ydot\,,\Zdot\,\right)$\,
    $\Ydot$\,
    and
    \,$(\,\Sdot\,,\Rdot\,)$\,
    are {\color{red}conditionally independent} given \,$\left(\,\Zdot\,,\Xdotdot\,\right)$.
\item
    $\Cov\!\left(\;Y_{ij}\,,\,Y_{kl}\;\left\vert\;\Xdotdot\right.\right) \,=\, 0$,\,
    for each \,$(i,j) \neq (k,l)$.
\end{itemize}

                \,\right)^{\!2}
            \,\right\}
        }
    \end{eqnarray*}
\item

\item
    \textbf{Important Remarks:}
    \begin{itemize}
    \item
        Note that the above estimation/approximation
        of the sampling-nonresponse covariance component
        disregards the following term:
        \begin{equation*}
        2
        \cdot
        \dfrac{
            1
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d}(s\,;\zdot)
                }
            }
        \cdot
        E_{Y}\!\left[\;
            \widetilde{T}_{\alpha,Y}
            \, - \,
            \widehat{T}^{\textnormal{(HT)}}_{\alpha,Y}
            \;\left\vert\!
            \begin{array}{c}
                s,s_{R},
                \\
                \Zdot\,,\Xobs
                \end{array}
            \right.
            \!\right]
        \cdot
        {\color{red}E_{Y}\!\left[\;\,
            \widehat{R}^{\textnormal{(HT)}}_{\alpha,Y/Z} - R_{\alpha,Y/Z}
            \,\left\vert
            \begin{array}{c}
                s,s_{R},
                \\
                \Zdot\,,\Xobs
                \end{array}
            \right.
            \!\right]}
        \end{equation*}
        The reason this has to be done in practice is that, in practice,
        there is usually insufficient data/information to compute or estimate
        the quantity
        \,$
        E_{Y}\!\left[\;
            \left.
            \widehat{R}^{\textnormal{(HT)}}_{\alpha,Y/Z} - R_{\alpha,Y/Z}
            \;\right\vert\,
            s,s_{R},\Zdot,\Xobs
            \,\right]
        $\,
        (indicated in red above), as \,$R_{\alpha,Y/Z}$\, involves population units
        that are not selected in the sample (hence, not observed).
        \vskip 0.1cm
        \noindent
        Obviously, the above estimation/approximation
        of the sampling-nonresponse covariance component
        is acceptable only if the quantity
        \begin{equation*}
        2
        \cdot
        \dfrac{
            1
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d}(s\,;\zdot)
                }
            }
        \cdot
        E_{Y}\!\left[\;
            \widetilde{T}_{\alpha,Y}
            \, - \,
            \widehat{T}^{\textnormal{(HT)}}_{\alpha,Y}
            \;\left\vert\!
            \begin{array}{c}
                s,s_{R},
                \\
                \Zdot\,,\Xobs
                \end{array}
            \right.
            \!\right]
        \cdot
        {\color{black}E_{Y}\!\left[\;\,
            \widehat{R}^{\textnormal{(HT)}}_{\alpha,Y/Z} - R_{\alpha,Y/Z}
            \,\left\vert
            \begin{array}{c}
                s,s_{R},
                \\
                \Zdot\,,\Xobs
                \end{array}
            \right.
            \!\right]}
        \end{equation*}
        is small.
        \vskip 0.1cm
        \noindent
        For this, in practice, we thus {\color{red}tacitly assume} that the quantity
        \,$
        E_{Y}\!\left[\;
            \left.
            \widetilde{T}_{\alpha,Y} \overset{{\color{white}.}}{-} \widehat{T}^{\textnormal{(HT)}}_{\alpha,Y}
            \;\right\vert\,
            s,s_{R},\Zdot,\Xobs
            \,\right]
        $\,
        is small;
        recall that this quantity is called the \textbf{conditional model bias}
        (see Lemma \ref{biasTtilde}), and
        it can be expressed as follows:
        \begin{eqnarray*}
        E_{Y}\!\left[\;
            \left.
            \widetilde{T}_{\alpha,Y} \overset{{\color{white}.}}{-} \widehat{T}^{\textnormal{(HT)}}_{\alpha,Y}
            \;\right\vert\,
            s,s_{R},\Zdot,\Xobs
            \,\right]
        & = &
            \underset{k\in{\color{red}s_{M}}}{\sum}\,w_{k}\,d_{k}\,
            E_{Y}\!\left[\;
                \left.
                Y^{*}_{k} \overset{{\color{white}.}}{-} Y_{k}
                \;\right\vert\,
                s,s_{R},\Zdot,\Xobs
                \,\right]
        \end{eqnarray*}
        This harks back to our earlier remark (Remark \ref{remarkSmallUnitBias}) that,
        in practice, we would like to choose an imputation strategy such that
        \begin{equation*}
        E_{Y}\!\left[\;
            \left.
            Y^{*}_{k} \overset{{\color{white}.}}{-} Y_{k}
            \;\right\vert\,
            s,s_{R},\Zdot,\Xobs
            \,\right]
        \;\; \approx \;\;
            0,
        \quad
        \textnormal{for each \,$k \in s_{M}$}
        \end{equation*}
        See the paragraph after Equation (5.2), p.175, \cite{Beaumont2011}.
        \vskip 0.3cm
    \item
        The second summand
        \begin{equation*}
        2
        \cdot
        \left(\,
            \dfrac{1}{T_{\alpha,Z} \slash N}
            \,\right)
        \left(\,
            \dfrac{
                {\color{white}.}
                {\color{red}T_{\alpha,Z}} \,-\, \widehat{T}_{d}(s\,;\zdot)
                {\color{white}.}
                }{
                \overset{{\color{white}.}}{
                    N
                    }
                }           
            \,\right)
        \cdot
        \dfrac{
            1
            }{
            \overset{{\color{white}.}}{
                \left. \widehat{T}_{d}(s\,;\zdot)^{2} \right\slash N^{2}
                }
            }
        \cdot
        \left(\,
            \dfrac{
                \underset{l \in s_{R}}{\sum}\,
                W^{(+)}_{d,l}\,d_{l}\,\sigma_{l}^{2}
                \;\;-\;
                \underset{k \in s_{M}}{\sum}\,
                w_{k}\,d_{k}^{2}\,\sigma_{k}^{2}
                }{
                \overset{{\color{white}.}}{N^{2}}                
                }
            \,\right)
        \end{equation*}
        in the expression for the sampling-nonresponse covariance component
        would also probably have to be disregarded in practice,
        due to the presence of the term \,$T_{\alpha,Z}$
        (generalized total of the population characteristic $\Zdot$),\,
        which is usually unknown in practice.
        \vskip 0.1cm
        In order to justify the omission of the second summand,
        one could attempt to find (practically usefull) ``regularity conditions"
        that are sufficient to force the second summand to be
        \,$O_{p}\!\left(\,1/n^{\delta}\,\right)$,\,
        for some \,$\delta > 0$,\,
        where \,$n$\, here denotes the sample size.
        We may not be too far from such a sufficent collection of regularity conditions.
        Recall that
        \begin{itemize}
        \item
            we have the general asymptotic result for the Horvitz-Thompson generalized total estimator:
            \begin{equation*}
            \dfrac{
                {\color{white}.}
                \widehat{T}_{d}(s\,;\zdot) \,-\, {\color{black}T_{\alpha,Z}}
                {\color{white}.}
                }{
                \overset{{\color{white}.}}{
                    N
                    }
                }
            \;\; = \;\;
                O_{p}\!\left(\dfrac{1}{\sqrt{\textnormal{\,sample size\,}}}\right),
            \quad
            \textnormal{under suitable regularity conditions on \,$\Zdot$,\, and}
            \end{equation*}
        \item
            $T_{\alpha,Z} \slash N$\, being asymptotically bounded away from zero
            is one of the regularity conditions often assumed in order to establish
            consistency for the Horvitz-Thomposon ratio estimator.
        \end{itemize}
        Hence, it may be sufficient provided one could find
        a ``suitable'' (practically applicable) regularity condition on
        \,$W^{(+)}_{d,l}$,\, $\sigma^{2}_{k}$,\, and \,$w_{k} = 1/\pi_{k}$\,
        such that
        \begin{equation*}
        \dfrac{
            {\color{white}.}
            \underset{l \in s_{R}}{\sum}\,
            d_{l}\,W^{(+)}_{d,l}\,\sigma_{l}^{2}
            \;\,-\,
            \underset{k \in s_{M}}{\sum}\,
            w_{k}\,d_{k}\,\sigma_{k}^{2}
            {\color{white}.}
            }{
            \overset{{\color{white}.}}{N^{2}}                
            }
        \;\; = \;\;
            O_{p}\!\left(\,1\,\right)
        \end{equation*}
        If such a regularity condition is available,
        we should be able to invoke it, in conjuction with the other two more standard regularity conditions above,
        so as to conclude that
        \begin{equation*}
        2
        \cdot
        \left(\,
            \dfrac{1}{T_{\alpha,Z} \slash N}
            \,\right)
        \left(\,
            \dfrac{
                {\color{white}.}
                {\color{black}T_{\alpha,Z}} \,-\, \widehat{T}_{d}(s\,;\zdot)
                {\color{white}.}
                }{
                \overset{{\color{white}.}}{
                    N
                    }
                }           
            \,\right)
        \cdot
        \dfrac{
            1
            }{
            \overset{{\color{white}.}}{
                \left. \widehat{T}_{d}(s\,;\zdot)^{2} \right\slash N^{2}
                }
            }
        \cdot
        \left(\,
            \dfrac{
                \underset{l \in s_{R}}{\sum}\,
                d_{l}\,W^{(+)}_{d,l}\,\sigma_{l}^{2}
                \;\;-\;
                \underset{k \in s_{M}}{\sum}\,
                w_{k}\,d_{k}\,\sigma_{k}^{2}
                }{
                \overset{{\color{white}.}}{N^{2}}                
                }
            \,\right)
        \;\; = \;\;
            O_{p}\!\left(\,\dfrac{1}{n^{\delta}}\,\right),
        \end{equation*}
        for some \,$\delta > 0$,\,
        where \,$n$\, denotes the sample size.
        Such an asymptotic result would suffice to justify disregarding the second summand,
        when estimating sampling-nonresponse covariance component in practice.
    \end{itemize}
\end{itemize}
