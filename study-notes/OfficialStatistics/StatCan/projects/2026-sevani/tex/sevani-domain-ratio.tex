\section{Variance estimation for composite linear imputation estimators for {\color{red}domain ratios}}

        %%%%% ~~~~~~~~~~~~~~~~~~~~ %%%%%

\subsection{Formulation of the target variability quantification problem}
\label{DomainRatioFormulationOfVarianceEstimationProblem}
\setcounter{theorem}{0}

        %%%%% ~~~~~~~~~~~~~~~~~~~~ %%%%%

Suppose:
\begin{itemize}
\item
    We would like to estimate
    -- using data collected according to a probability sample --
    a domain ratio (the ratio of two population totals)
    \begin{equation*}
    R_{d,Y/Z}
    \;\; := \;\;
        \dfrac{T_{d,Y}}{\overset{{\color{white}.}}{T_{d,Z}}}
    \;\; = \;\;
        \dfrac{
            {\color{white}.}
            \underset{i \in U}{\sum}\;d_{i}Y_{i}
            {\color{white}.}
            }{
            \overset{{\color{white}.}}{\underset{i \in U}{\sum}\;d_{i}Z_{i}}
            }\,,
    \end{equation*}
    where \,$U = \{\,1,2,\ldots,N\,\} \subset \N$\, represents the population,
    $d_{\bullet} = (d_{1},\ldots,d_{N}) \in \{0,1\}^{N}$ is the vector of domain membership indicators,
    and
    \,$\Ydot = (Y_{1},\ldots,Y_{N})$,\,
    \,$\Zdot = (Z_{1},\ldots,Z_{N})$\,
    are two $\Re$-valued population characteristics.
\item
    Aside/Reminder:
    If there are no missing data for the units in the selected sample,
    then the \textit{Horvitz-Thompson ratio estimator}
    \begin{eqnarray*}
    \widehat{R}_{d,Y/Z}
    & = &
        \widehat{R}_{d,Y/Z}(\,
            \Sdot\,,\,
            \Ydot\,,\,
            \Zdot\,;\,
            \dbullet
            \,)
    \;\; := \;\;
        \dfrac{
            \widehat{T}_{d,Y}
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d,Z}
                }
            }
    \;\; :=: \;\;
        \dfrac{
            \widehat{T}_{d}(s\,;\Ydot)
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d}(s\,;\Zdot)
                }
            }
    \\
    & = &
        \dfrac{
            {\color{white}.}
            \underset{i\,\in\,s}{\sum}\;d_{i}\,w_{i}\,Y_{i}
            {\color{white}.}
            }{
            \overset{{\color{white}.}}{
                \underset{i\,\in\,s}{\sum}\;d_{i}\,w_{i}\,Z_{i}
                }
            }
    \;\; = \;\;
        \dfrac{
            {\color{white}.}
            \underset{i\,\in\,U}{\sum}\;d_{i}\,w_{i}\,S_{i}\,Y_{i}
            {\color{white}.}
            }{
            \overset{{\color{white}.}}{
                \underset{i\,\in\,U}{\sum}\;d_{i}\,w_{i}\,S_{i}\,Z_{i}
                }
            }\,,
    \end{eqnarray*}
    is a common choice as an estimator for
    \,$R_{d,Y/Z}$,\,
    where
    \begin{itemize}
    \item
        $w_{i} \,:=\, \dfrac{1}{\pi_{i}}$,\, for each population unit \,$i \in U$,\,
        is the (\textit{a priori} known, non-random) design weight of \,$i \in U$,\,
        i.e., the reciprocal of the selection probability \,$\pi_{i}$\,
        (assumed strictly positive) of \,$i \in U$,\,
    \item
        $\Sdot \,=\, (S_{1},\ldots,S_{N})^{T}$\,
        is the vector of $\{0,1\}$-valued selection indicator variables, and
    \item
        $\widehat{T}_{d,\xi} \, :=: \,\widehat{T}_{d}(s\,;\xi_{\bullet})$\,
        is the \textit{Horvitz-Thompson estimator for the domain total}
        \,$T_{d,\xi} \,:=\, \underset{i\,\in\,U}{\sum}\,d_{i}\xi_{i}$\,
        of an arbitrary $\Re$-valued population characteristic \,$\xi_{i}$ ($i \in U$),\, i.e.,
        \begin{equation*}
        \widehat{T}_{d,\xi}(s)
        \;\; :=: \;\;
            \widehat{T}_{d}(s\,;\xi_{\bullet})
        \;\; := \;\;
            \underset{i\,\in\,s}{\sum}\;d_{i}\,w_{i}\,\xi_{i}
        \;\; = \;\;
            \underset{i\,\in\,U}{\sum}\;d_{i}\,w_{i}\,S_{i}\,\xi_{i}
        \end{equation*}
        We will use the alternative notations
        \,$\widehat{T}_{d,\xi}(s)$\, and \,$\widehat{T}_{d}(s\,;\xi_{\bullet})$\,
        interchangeably, noting that the former is notationally more succinct,
        while the latter allows us to emphasize that
        \,$\xi_{\bullet}$\,
        is being considered as an argument to the formula
        \begin{equation*}
        \widehat{T}_{d}(s\,;\xi_{\bullet})
        \;\; := \;\;
            \underset{i\,\in\,s}{\sum}\;d_{i}\,w_{i}\,\xi_{i}
        \end{equation*}
        for the Horvitz-Thompson population total estimator.
    \end{itemize}
    Recall also that the following Horvitz-Thompson variance estimator
    \begin{equation*}
    v_{\textnormal{HT}}(\,s\,;\xi_{\bullet}\,)
    \;\; := \;\;
        \underset{i \in s}{\sum}\,
        \underset{j \in s}{\sum}\;
        \dfrac{\pi_{ij} - \pi_{i}\pi_{j}}{\pi_{ij}}
        \cdot
        \dfrac{d_{i}\,\xi_{i}}{\pi_{i}}
        \cdot
        \dfrac{d_{j}\,\xi_{j}}{\pi_{j}}
    \end{equation*}
    is a design-unbiased estimator for the sampling variance of
    \,$\widehat{T}_{d,\xi_{\bullet}}(s) \,=\, \widehat{T}_{d}(s\,;\xi_{\bullet})$,\,
    i.e.,
    \begin{equation*}
    E_{S}\!\left[\;
        v_{\textnormal{HT}}(\,s\,;\xi_{\bullet}\,)
        \,\left\vert
        \begin{array}{c} \xi_{\bullet} \end{array}
        \right.
        \!\right]
    \;\; = \;\;
        \Var_{S}\!\left(\,
            \left.
            \widehat{T}_{d}(s\,;\xi_{\bullet})
            \;\right\vert\!
            \begin{array}{c} \xi_{\bullet} \end{array}
            \!\!\right)
    \end{equation*}
\item
    The value of \,$Z_{i}$\, has been successfully observed or measured
    for each population unit \,$i$\, in the selected sample \,$s \subset U$.\,
    On the other hand, the value of \,$Y_{l}$\, has been successfully observed
    only for units \,$l$\, in the (random) ``responding'' subsample
    \,$s_{R} \subset s \subset U$;\,
    in other words, the measurement of \,$Y_{k}$\, somehow failed for each unit \,$k$\,
    in the ``nonresponding'' subsample
    \,$s_{M} \, := \,s \,\backslash\, s_{R} \,\subset\, U$.\,
\item
    We would like to estimate \,$R_{d,Y/Z}$\,
    with an imputation estimator of the following form:
    \begin{eqnarray*}
    \widetilde{R}_{d,Y/Z}
    & = &
        \widetilde{R}_{d,Y/Z}(\,
            \Sdot\,,\,
            \Rdot\,,\,
            \Ydot\,,\,
            \Zdot\,;\,
            \Xdotdot\,,\,
            \dbullet
            \,)
    \;\; := \;\;
        \dfrac{
            \widetilde{T}_{d,Y}
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d,Z}
                }
            }
    \;\; :=: \;\;
        \dfrac{
            \widehat{T}_{d}(s\,;Y^{\dagger}_{\bullet})
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d}(s\,;\Zdot)
                }
            }
    \\
    & = &
        \dfrac{
            \underset{i\,\in\,s}{\sum}\;d_{i}\,w_{i}\,Y^{\dagger}_{i}
            }{
            \overset{{\color{white}.}}{
                \underset{i\,\in\,s}{\sum}\;d_{i}\,w_{i}\,Z_{i}
                }
            }
    \;\; = \;\;
        \dfrac{
            \underset{i\,\in\,U}{\sum}\;d_{i}\,w_{i}\,S_{i}\left(\,
                R_{i}\,Y_{i}
                \overset{{\color{white}.}}{+}
                (1-R_{i})\,Y^{*}_{i}
                \,\right)
            }{
            \overset{{\color{white}.}}{
                \underset{i\,\in\,U}{\sum}\;d_{i}\,w_{i}\,S_{i}\,Z_{i}
                }
            }\,,
    \end{eqnarray*}
    where
    \begin{itemize}
    % \item
    %     $w_{i} \,:=\, \dfrac{1}{\pi_{i}}$,\, for each population unit \,$i \in U$,\,
    %     is the (\textit{a priori} known, non-random) design weight of \,$i \in U$,\,
    %     i.e., reciprocal of the selection probability \,$\pi_{i}$\,
    %     (assumed strictly positive) of \,$i \in U$,\,
    % \item
    %     $\Sdot \,=\, (S_{1},\ldots,S_{N})^{T}$ is the vector of $\{0,1\}$-valued selection indicator variables,
    \item
        $\Rdot \,=\, (R_{1},\ldots,R_{N})^{T}$ is the vector of $\{0,1\}$-valued response indicator variables,
    \item
        $\Xdotdot \,=\, \left(\,X_{ij}\,\right) \,\in\, \overline{\Re}^{N \times r}$\,
        is a random matrix containing the predictor variables for $\Ydot$,
        where the $i^{\textnormal{th}}$ row of \,$\Xdotdot$\, corresponds to populaton unit \,$i \in U$,\,
        and where \,$\overline{R} \,:=\, \Re \sqcup \{\,\texttt{NA}\,\}$\,
        is an extension of $\Re$ alllowing for missing values in \,$\Xdotdot$,\, and
    \item
        for each \,$i \in s$,\,
        \,$Y^{\dagger}_{i} \;=\; R_{i}\,Y_{i}+(1-R_{i})\,Y^{*}_{i}$,\,
        and
        \begin{equation*}
        Y^{*}_{k}
        \;\; = \;\;
            Y^{*}_{k}(\,\Sdot,\Rdot,\Ydot,\Xdotdot\,)\,,
        \quad
        \textnormal{for each \,$k \in s_{M} = s\,\backslash\,s_{R}$},
        \end{equation*}
        is an imputed value for
        \,$Y_{k}$\, and \,$Y^{*}_{k}$\,
        may depend on
        \,$\Sdot,\Rdot,\Ydot,\Xdotdot$.\,
    % \item
    %     $\widehat{T}(s\,;\pi,d_{\bullet},\xi_{\bullet})$\,
    %     is the Horvitz-Thompson estimator for the domain total
    %     \,$T_{d,\xi} \,:=\, \underset{i\,\in\,U}{\sum}\,d_{i}\xi_{i}$\,
    %     of an arbitrary population characteristic \,$\xi_{i}$ ($i \in U$),\, i.e.,
    %     \begin{equation*}
    %     \widehat{T}(s\,;\pi,d_{\bullet},\xi_{\bullet})
    %     \;\; := \;\;
    %         \underset{i\,\in\,s}{\sum}\;d_{i}\,w_{i}\,\xi_{i}
    %     \;\; = \;\;
    %         \underset{i\,\in\,U}{\sum}\;d_{i}\,w_{i}\,S_{i}\,\xi_{i}
    %     \end{equation*}
    \end{itemize}
\item
    Furthermore, \,$Y^{*}_{k}$\, ($k \in s_{M}$) is \textbf{composite-linear},
    i.e., there exists \,$J \in \N$\, such that
    \begin{center}
    \begin{minipage}{6.2in}
    for each admissible sample \,$s \subset U$,\,
    each \,$s_{R} \subset s$,\,
    {\color{red}each} \,{\color{red}$k \in s_{M} := s \,\backslash\, s_{R}$},\,
    there exist
    \begin{equation*}
    \left\{\;\;\begin{array}{ccl}
    j
    & = &
        j(\,k,s,s_{R}\,;\Xobs\,) \;\in\; \{1,2,\ldots,J\},
    \\
    \varphi^{(j)}_{0k}
    & \overset{{\color{white}\textnormal{\Large$1$}}}{=} &
        \varphi^{(j)}_{0k}(\,s,s_{R}\,;X_{\bullet\bullet}\,),
        \quad
        \textnormal{and}
    \\
    \varphi^{(j)}_{lk}
    & \overset{{\color{white}\textnormal{\Large$1$}}}{=} &
        \varphi^{(j)}_{lk}(\,s,s_{R}\,;X_{\bullet\bullet}\,),
        \;\;
        \textnormal{{\color{red}for each \,$l \in s_{R}$}}
    \end{array}\right.
    \end{equation*}
    such that
    \begin{equation*}
    Y_{{\color{red}k}}^{*}
    \;\; = \;\;
        Y^{*}_{k}(\,\Sdot,\Rdot,\Ydot,\Xdotdot\,)
    \;\; = \;\;
        \varphi^{(j)}_{0k}
        \,+\,
        \underset{l \in s_{R}}{\sum}\,\varphi^{(j)}_{l{\color{red}k}}\,Y_{l}\,,
        \quad
        \textnormal{for each \,$k \in s_{M} := s \,\backslash\, s_{R}$},
    % \quad
    % \textnormal{for some \,$j \in \{1,2,\ldots,J\}$\, depending on \,$s,\,s_{R},\,\Xobs$}
    \end{equation*}
    \end{minipage}
    \end{center}
\end{itemize}

\,$R_{d,Y/Z} \; := \; \dfrac{T_{d,Y}}{T_{d,Z}}$.

\begin{equation*}
\MSE\!\left(\;
    \left.
    \widetilde{R}_{d,Y/Z}
    \;\right\vert\!
    \begin{array}{c} \Zdot\,,\Xdotdot \end{array}
    \!\right)
\;\; := \;\;
    E\!\left[\;\left(\,
        \left.
        \widetilde{R}_{d,Y/Z}
        \, - \,
        R_{d,Y/Z}
        \,\right)^{\!2}
        \;\right\vert\!
        \begin{array}{c} \Zdot\,,\Xdotdot \end{array}
        \!\right]
\end{equation*}
of
\,$\widetilde{R}_{d,Y/Z}$\,
as estimator for
\,$R_{d,Y/Z}$,\,
given
\,$\Zdot$\, and \,$\Xdotdot$,\,
is proposed for this purpose.

\noindent

\,$
\MSE\!\left(\;
    \left.
    \widetilde{R}_{d,Y/Z}
    \;\right\vert\!
    \begin{array}{c} \Zdot\,,\Xdotdot \end{array}
    \!\right)
$\,
will be proposed.

\subsection{Probabilistic framework}
\label{DomainRatioProbabilisticFramework}
\setcounter{theorem}{0}

        %%%%% ~~~~~~~~~~~~~~~~~~~~ %%%%%

\noindent
\textbf{Probability spaces}
\begin{itemize}
\item
    Let
    \,$U \,=\, \{1,2,\ldots,N\}$.\,
    Let
    \,$\mathcal{P}(U) \,=\, \mathcal{P}(\{1,2,\ldots,N\})$\,
    denote the power set (set of all subsets) of \,$U$.\,
    Let
    \,$\left(\,\mathcal{P}(U)\,,\mathcal{P}(\mathcal{P}(U))\,,\overset{{\color{white}-}}{\pi}\,\right)$\,
    be the probability space induced by a sampling design, i.e.,
    a probability mass function defined on
    \,$\mathcal{P}(U)$.\,
\item
    Let \,$\left(\,\Omega,\mathscr{A},\lambda\,\right)$\, be a probability space.
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Random variables}
\begin{itemize}
\item
    $\Ydot \,=\, (Y_{1},Y_{2},\ldots,Y_{N})^{T} : \Omega \longrightarrow \Re^{N \times 1}$\,
    is an $\Re^{N \times 1}$-valued random variable defined on \,$\Omega$,\,
    i.e., $\mathscr{A}$-measurable map.
\item
    $\Zdot \,=\, (Z_{1},Z_{2},\ldots,Z_{N})^{T} : \Omega \longrightarrow \Re^{N \times 1}$\,
    is an $\Re^{N \times 1}$-valued random variable defined on \,$\Omega$.\,
\item
    $\Xdotdot\,=\,\left(\,\overset{{\color{white}.}}{X_{ij}}\,\right) : \Omega \longrightarrow \Re^{N \times r}$\,
    is an $\Re^{N \times r}$-valued random variable defined on \,$\Omega$.
\item
    $\Sdot \,=\, (S_{1},S_{2},\ldots,S_{N})^{T} : \mathcal{P}(U) \longrightarrow \{0,1\}^{N \times 1}$\,
    is a $\{0,1\}^{N \times 1}$-valued random variable defined on \,$\mathcal{P}(U)$,\,
    i.e., $\mathcal{P}(\mathcal{P}(U))$-measurable map.
\item
    $\Rdot \,=\, (R_{1},R_{2},\ldots,R_{N})^{T} : {\color{red}\Omega \times \mathcal{P}(U)} \longrightarrow \{0,1\}^{N \times 1}$\,
    is a $\{0,1\}^{N \times 1}$-valued random variable defined on the product probability space
    \,$\Omega \times \mathcal{P}(U)$,\,
    i.e.,
    \,$\left(\mathscr{A} \overset{{\color{white}.}}{\otimes} \mathcal{P}(\mathcal{P}(U))\right)$-measurable map
    that satisfies:
    % \begin{equation*}
    % \mu\!\left(\left\{\;
    %     \omega \in \Omega
    %     \;\left\vert\;
    %     S_{k}(s) = 0
    %     \,\overset{{\color{white}1}}{\Longrightarrow}\,
    %     R_{k}(\omega,s_{R}) = 0
    %     \right.
    %     \;\right\}\right)
    % \; = \;
    %     1\,,
    % \quad
    % \textnormal{for each \,$k \in U$,\, and each \,$s_{R} \subset s \subset U$}
    % \end{equation*}
    % \begin{equation*}
    % (\,\mu\times\pi\,)\!\left(\left\{\;
    %     (\omega,s) \in \Omega
    %     \;\left\vert\;
    %     \begin{array}{c}
    %     S_{k}(s) = 0, \,\textnormal{and}\; R_{k}(\omega,s) = 1,
    %     \\
    %     \overset{{\color{white}.}}{\textnormal{for one or more $k \in U$}}
    %     \end{array}
    %     \right.
    %     \;\right\}\right)
    % \; = \;
    %     0
    % \end{equation*}
    \begin{equation*}
    (\,\mu\otimes\pi\,)\!\left(\;\,
        \underset{k \in U}{\bigcup}\left\{\;
            (\omega,s) \in \Omega\times\mathcal{P}(U)
            \;\left\vert
            \begin{array}{c}
            S_{k}(s) = 0, %\,\textnormal{and}\;
            \\
            \overset{{\color{white}.}}{R_{k}(\omega,s) = 1}
            \end{array}
            \right.
            \!\!\right\}
        \;\right)
    \; = \;
        0
    \end{equation*}
    Note that this condition stipulates that,
    whenver a population unit \,$k \in U$\, is not selected in the (random) sample \,$s \subset U$,\,
    then it is not a respondent either.
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Interpretation}
\begin{itemize}
\item
    $U = \{1,2,\ldots,N\}$
    \,--\,
    index set of the target (finite) population of interest,
    and \,$N$\, is the (finite) size of the target population.
\item
    $\left(\,\Omega,\mathscr{A},\lambda\,\right)$
    \,--\,
    underlying source of stochasticity of the superpopulation model,
    encoding the probabilistic relations among
    \,$\Ydot$,\, $\Zdot$,\, $\Xdotdot$,\, and \,$\Rdot$.\,
\item
    $\Ydot$
    \,--\,
    numerator of the $\Re$-valued target variable (population characteristic).
\item
    $\Zdot$
    \,--\,
    denominator of the $\Re$-valued target variable (population characteristic).
\item
    $\Xdotdot$
    \,--\,
    $\Re^{r}$-valued auxiliary variable.
\item
    $\Sdot$
    \,--\,
    $\{0,1\}$-valued binary variable indicating, for each \,$s \subset U$,\,
    whether a population unit \,$k \in U$\, is selected in \,$s$.\,
    More precisely, for each \,$s \subset U$,\, we have:
    \begin{equation*}
    S_{k}(s)
    \;\; = \;\;
        \left\{\begin{array}{cl}
            1\,, & \textnormal{if \,$k \in s$},
            \\
            0\,, & \textnormal{otherwise}.
            \end{array}\right.
    \end{equation*}
\item
    $\Rdot$
    \,--\,
    $\{0,1\}$-valued binary variable indicating, for almost every outcome
    \,$\omega \in \Omega$\,
    and each
    \,$(s,s_{R}) \in \mathcal{P}(U) \times \mathcal{P}(U)$\,
    satisfying
    \,$s_{R} \subset s$,\,
    whether a selected population unit \,$ k \in s$\,
    is a respondent or not.
    % More precisely, for each \,$s \subset U$ and each \,$k \in s$,\, we have:
    % \begin{equation*}
    % R_{k}(\omega,s_{R})
    % \;\; = \;\;
    %     \left\{\begin{array}{cl}
    %         1\,, & \textnormal{if \,$k \in s_{R}$},
    %         \\
    %         0\,, & \textnormal{otherwise}.
    %         \end{array}\right.
    % \end{equation*}
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{Probabilistic assumptions}
\begin{itemize}
\item
    $\pi_{i} \,:=\, P(\,S_{i} = 1\,) \,:=\, \pi\!\left(\,\left\{\,\left. s \overset{{\color{white}.}}{\subset} U \,\right\vert\, i \in s\,\right\}\,\right) \,>\, 0$,\,
    for each \,$i \in U$,\,
    and
    \,$w_{i} \,:=\, \dfrac{1}{P(\,S_{i} = 1\,)}$,\, for each \,$i \in U$.
\item
    $\left(\,\Ydot\,,\Zdot\,,\Xdotdot\,\right)$\, and \,$\Sdot$\, are (unconditionally) independent.
\item
    % $\left(\,\Ydot\,,\Zdot\,\right)$\,
    $\Ydot$\,
    and
    \,$(\,\Sdot\,,\Rdot\,)$\,
    are {\color{red}conditionally independent} given \,$\left(\,\Zdot\,,\Xdotdot\,\right)$.
\item
    $\Cov\!\left(\;Y_{k}\,,\,Y_{l}\;\left\vert\;\Zdot\,,\Xdotdot\right.\right) \,=\, 0$,\,
    for each \,$k, l \in U$\, with \,$k \neq l$.
\end{itemize}

\vskip 0.2cm
\noindent
\textbf{(Non-random) parameters}
\begin{itemize}
\item
    $\dbullet \,=\, (d_{1},d_{2},\ldots,d_{N}) \in \{0,1\}^{N}$\,
    is a (non-random, binary) domain membership indicator.

\begin{lemma}
\label{DomainRatioDecompositionMSETtilde}
\mbox{}
\vskip 0.1cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{DomainRatioFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{DomainRatioProbabilisticFramework}.
Then, the condition mean squared error
\,$
\MSE\!\left(\;
    \left.
    \widetilde{R}_{d,Y/Z}
    \;\right\vert\,
    \Zdot\,,\Xobs
    \,\right)
$\,
of \,$\widetilde{R}_{d,Y/Z}$\,
as estimator for
\,$R_{d,Y/Z}$,\,
given \,$\Zdot$\, and \,$\Xdotdot$,\,
can be additively decomposed as follows:
\begin{eqnarray*}
\MSE\!\left(\;
    \left.
    \widetilde{R}_{d,Y/Z}
    \;\right\vert\,
    \Zdot\,,\Xobs
    \,\right)
& := &
    E\!\left[\,
        \left.
        (\,\widetilde{R}_{d,Y/Z} - R_{d,Y/Z}\,)^{2}
        \,\right\vert\,
        \Zdot\,,\Xobs
        \,\right]
\\
& = &
    E_{Y}\!\left[\,
    \Var_{S}\!\left(\,
        \left.
        \widehat{R}_{d,Y/Z}
        \,\right\vert\,
        \Ydot,
        \Zdot\,,\Xobs
        \,\right)
        \,\right]
\\
&&
    \;+\;
    E_{SR}\!\left[\,
    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{R}_{d,Y/Z} - \widehat{R}_{d,Y/Z}\,)^{2}
        \,\right\vert\,
        s,s_{R},\Zdot\,,\Xobs
        \,\right]
        \,\right]
\\
&&
    \;+\;
    2\cdot
    E_{SR}\!\left[\,
    E_{Y}\!\left[\,
        \left.
        (\,\widetilde{R}_{d,Y/Z} \,-\, \widehat{R}_{d,Y/Z}\,)(\,\widehat{R}_{d,Y/Z} \,-\, R_{d,Y/Z}\,)
        \,\right\vert\,
        s,s_{R},\Zdot\,,\Xobs
        \,\right]
        \,\right]
\end{eqnarray*}
\end{lemma}

\vskip 0.2cm
\begin{lemma}[Approximately unbiased estimator for sampling variance component for ratio estimators]
\label{DomainRatioSamplingVariance}
\mbox{}
\vskip -0.3cm
\noindent
Suppose we are given quantities as described in Subsection
\ref{DomainRatioFormulationOfVarianceEstimationProblem}
and
that they satisfy the probabilistic assumptions as described in Subsection
\ref{DomainRatioProbabilisticFramework}.
Then, the following statements are true:
\renewcommand{\theenumi}{\roman{enumi}}
\renewcommand{\labelenumi}{\mbox{}\;\;\,(\theenumi)$\quad$}
\begin{enumerate}
\item
    If
    \;$v(\,s\,;y_{\bullet},\zdot\,)$\,
    is an (arbitrary) design-unbiased estimator for the sampling variance, i.e.,
    \begin{equation*}
    E_{S}\!\left[\;
        v(\,s\,;\Ydot,\Zdot\,)
        \,\left\vert\;
        \Ydot,\Zdot\,,\Xobs
        \right.
        \,\right]
    \;\; = \;\;
        \Var_{S}\!\left(\,
            \left.
            \widehat{R}_{d,Y/Z}
            \,\right\vert\,
            \Ydot,
            \Zdot,
            \Xobs
            \,\right),
    \end{equation*}
    then
    \,$
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\Ydot,\Zdot\,)
        \left\vert
        \begin{array}{c}
            s,s_{R}, \YsubR, \\ \Zdot\,,\Xobs
            \end{array}
        \right.
        \!\!\right]
    $\,
    is an $(\Sdot,\Rdot,\YsubR)$-unbiased estimator for
    \,$
    E_{Y}\!\left[\,
        \Var_{S}\!\left(\,
            \widehat{R}_{d,Y/Z}
            \,\left\vert\!
            \begin{array}{c} \Ydot,\\ \Zdot, \Xobs \end{array}
            \right.
            \!\!\right)
    \,\right]
    $,\,
    i.e.,
    \begin{equation*}
    E_{SR}\,
    E_{Y_{R}}\,
    E_{Y_{M}}\!\left[\,
        v(\,s\,;\Ydot,\Zdot\,)
        \,\left\vert\!
        \begin{array}{c}
        s,s_{R},\YsubR, \\ \Zdot\,,\Xobs
        \end{array}
        \right.
        \!\right]
    \;\; = \;\;
        E_{Y}\!\left[\,
            \Var_{S}\!\left(\,
                \widehat{R}_{d,Y/Z}
                \,\left\vert\!
                \begin{array}{c}
                \Ydot, \\ \Zdot, \Xobs
                \end{array}
                \right.
                \!\!\right)
            \,\right]
    \end{equation*}
\item
    If we choose in turn to approximate
    \,$v(\,s\,;y_{\bullet}\,)$\,
    by the {\color{red}approximately} design-unbiased linearization-based estimator
    \,$v_{L(Y/Z)}(\,s\,;y_{\bullet},\zdot\,)$\,
    for the sampling variance (see Result 5.6.2, p.178, \cite{Sarndal1992}), i.e.,
    \begin{eqnarray*}
    &&

    \\

        \dfrac{1}{
            \overset{{\color{white}.}}{\widehat{T}_{d}(s\,;\zdot)^{2}}
            }
        \cdot
        \underset{k \in s}{\sum}\,
        \underset{l \in s}{\sum}\;
        \dfrac{\pi_{kl} - \pi_{k}\pi_{l}}{\pi_{kl}}
        \left(
            \dfrac{
                d_{k}\left(\,y_{k}-\widehat{R}_{d,Y/Z}(s\,;\ydot,\zdot) \cdot z_{k}\,\right)
                }{
                \pi_{k}
                }
            \right)
        \left(
            \dfrac{
                d_{l}\left(\,y_{l}-\widehat{R}_{d,Y/Z}(s\,;\ydot,\zdot) \cdot z_{l}\,\right)
                }{
                \pi_{l}
                }
            \right)
    \\
    & = &
        \dfrac{1}{
            \overset{{\color{white}.}}{\widehat{T}_{d}(s\,;\zdot)^{2}}
            }
        \cdot
        \underset{k \in s}{\sum}\,
        \underset{l \in s}{\sum}\;
        \dfrac{\pi_{kl} - \pi_{k}\pi_{l}}{\pi_{kl}}\,
        \left(\,
            \dfrac{d_{k}}{\pi_{k}}
            \left(\,
                y_{k}
                \, - \,
                \dfrac{
                    \widehat{T}_{d}(s\,;\ydot)
                    }{
                    \overset{{\color{white}.}}{\widehat{T}_{d}(s\,;\zdot)}
                    }
                \cdot
                z_{k}
                \,\right)
            \right)
        \left(\,
            \dfrac{d_{l}}{\pi_{l}}
            \left(\,
                y_{l}
                \, - \,
                \dfrac{
                    \widehat{T}_{d}(s\,;\ydot)
                    }{
                    \overset{{\color{white}.}}{\widehat{T}_{d}(s\,;\zdot)}
                    }
                \cdot
                z_{l}
                \,\right)
            \right)
    \\
    & = &
        \dfrac{
            v_{\textnormal{HT}}(\,s\,;\ydot\,)
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d}(s\,;\zdot)^{2}
                }
            }
        \; + \;
        \dfrac{
            (-2)\left.\widehat{T}_{d}(s\,;\ydot)\right\slash\widehat{T}_{d}(s\,;\zdot)
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d}(s\,;\zdot)^{2}
                }
            }
        \left(\,
            \underset{k \in s}{\sum}\;
            \underset{l \in s}{\sum}
                \left(\dfrac{\pi_{kl}-\pi_{k}\pi_{l}}{\pi_{kl}}\right)
                \dfrac{d_{k}\,y_{k}}{\pi_{k}}\,
                \dfrac{d_{l}\,z_{l}}{\pi_{l}}
            \right)
    \\
    &&
        \; + \;
        \dfrac{
            \left.
            \widehat{T}_{d}(s\,;\ydot)^{2}
            \right\slash
            \widehat{T}_{d}(s\,;\zdot)^{2}
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d}(s\,;\zdot)^{2}
                }
            }
        \cdot
        v_{\textnormal{HT}}(s\,;\zdot)
    \end{eqnarray*}
    then
    \,$
    E_{Y_{M}}\!\left[\,
        v_{L(Y/Z)}(\,s\,;\Ydot,\Zdot\,)
        \left\vert\,
        s,s_{R},\YsubR = y_{R},\Zdot\,,\Xobs
        \right.
        \right]
    $\,
    can be expressed as follows:
    \begin{eqnarray*}
    &&

            \begin{array}{c}

            \right.
            \!\right]

        \; + \;

            1
            }{
            \overset{{\color{white}.}}{

            \right)
        \; + \;
        \dfrac{
            \underset{k \in s_{M}}{\sum}\,d_{k}^{\,2}\sigma^{2}_{k}/\pi_{k}^{2}
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d}(s\,;\zdot)^{2}
                }
            }
        \cdot
        \dfrac{
            v_{\textnormal{HT}}(s\,;\zdot)
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d}(s\,;\zdot)^{2}
                }
            }
    \end{eqnarray*}
    where
    \,$\widetilde{y}_{\bullet}$\,
    is given by:
    \begin{equation*}
    \widetilde{y}_{k}
    \;\; := \;\;
        \left\{\begin{array}{cl}
            y_{k}, & \textnormal{for \,$k \,\in\, s_{R}$}
            \\
            % \overset{{\color{white}1}}{E_{Y}\!\left[\;Y_{k}\,\left\vert\;\Xobs\right.\right]}, & \textnormal{for \,$k \,\in\, s_{M} \,:=\, s \,\backslash\, s_{R}$}
            \overset{{\color{white}\textnormal{\large1}}}{\mu_{k}}, & \textnormal{for \,$k \,\in\, s_{M} \,:=\, s \,\backslash\, s_{R}$}
            \end{array}\right.,
    \end{equation*}
    and, for \,$k \in s_{M}$,\, the parameters
    \,$\mu_{k}$\, and \,$\sigma_{k}^{2}$\,
    are given by:
    \begin{eqnarray*}
    \mu_{k}
    & := &
        {\color{white}...}E_{Y}\!\left[\;Y_{k}\,\left\vert\;\Zdot\,,\Xobs\right.\right]
    \\
    \sigma_{k}^{2}
    & := &
        \Var_{Y}\!\left[\;
            Y_{k}
            \,\left\vert\;
            \Zdot\,,\Xobs
            \right.
            \right]
    \end{eqnarray*}
\end{enumerate}
\end{lemma}

\item
    \textbf{Important Remarks:}
    \begin{itemize}
    \item
        Note that the above estimation/approximation
        of the sampling-nonresponse covariance component
        disregards the following term:
        \begin{equation*}
        2
        \cdot
        \dfrac{
            1
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d}(s\,;\zdot)
                }
            }
        \cdot
        E_{Y}\!\left[\;
            \widetilde{T}_{d,Y}
            \, - \,
            \widehat{T}_{d,Y}
            \;\left\vert\!
            \begin{array}{c}
                s,s_{R},
                \\
                \Zdot\,,\Xobs
                \end{array}
            \right.
            \!\right]
        \cdot
        {\color{red}E_{Y}\!\left[\;\,
            \widehat{R}_{d,Y/Z} - R_{d,Y/Z}
            \,\left\vert
            \begin{array}{c}
                s,s_{R},
                \\
                \Zdot\,,\Xobs
                \end{array}
            \right.
            \!\right]}
        \end{equation*}
        The reason this has to be done in practice is that, in practice,
        there is usually insufficient data/information to compute or estimate
        the quantity
        \,$
        E_{Y}\!\left[\;
            \left.
            \widehat{R}_{d,Y/Z} - R_{d,Y/Z}
            \;\right\vert\,
            s,s_{R},\Zdot,\Xobs
            \,\right]
        $\,
        (indicated in red above), as \,$R_{d,Y/Z}$\, involves population units
        that are not selected in the sample (hence, not observed).
        \vskip 0.1cm
        \noindent
        Obviously, the above estimation/approximation
        of the sampling-nonresponse covariance component
        is acceptable only if the quantity
        \begin{equation*}
        2
        \cdot
        \dfrac{
            1
            }{
            \overset{{\color{white}.}}{
                \widehat{T}_{d}(s\,;\zdot)
                }
            }
        \cdot
        E_{Y}\!\left[\;
            \widetilde{T}_{d,Y}
            \, - \,
            \widehat{T}_{d,Y}
            \;\left\vert\!
            \begin{array}{c}
                s,s_{R},
                \\
                \Zdot\,,\Xobs
                \end{array}
            \right.
            \!\right]
        \cdot
        {\color{black}E_{Y}\!\left[\;\,
            \widehat{R}_{d,Y/Z} - R_{d,Y/Z}
            \,\left\vert
            \begin{array}{c}
                s,s_{R},
                \\
                \Zdot\,,\Xobs
                \end{array}
            \right.
            \!\right]}
        \end{equation*}
        is small.
        \vskip 0.1cm
        \noindent
        For this, in practice, we thus {\color{red}tacitly assume} that the quantity
        \,$
        E_{Y}\!\left[\;
            \left.
            \widetilde{T}_{d,Y} \overset{{\color{white}.}}{-} \widehat{T}_{d,Y}
            \;\right\vert\,
            s,s_{R},\Zdot,\Xobs
            \,\right]
        $\,
        is small;
        recall that this quantity is called the \textbf{conditional model bias}
        (see Lemma \ref{biasTtilde}), and
        it can be expressed as follows:
        \begin{eqnarray*}
        E_{Y}\!\left[\;
            \left.
            \widetilde{T}_{d,Y} \overset{{\color{white}.}}{-} \widehat{T}_{d,Y}
            \;\right\vert\,
            s,s_{R},\Zdot,\Xobs
            \,\right]
        & = &
            \underset{k\in{\color{red}s_{M}}}{\sum}\,d_{k}\,w_{k}\,
            E_{Y}\!\left[\;
                \left.
                Y^{*}_{k} \overset{{\color{white}.}}{-} Y_{k}
                \;\right\vert\,
                s,s_{R},\Zdot,\Xobs
                \,\right]
        \end{eqnarray*}
        This harks back to our earlier remark (Remark \ref{remarkSmallUnitBias}) that,
        in practice, we would like to choose an imputation strategy such that
        \begin{equation*}
        E_{Y}\!\left[\;
            \left.
            Y^{*}_{k} \overset{{\color{white}.}}{-} Y_{k}
            \;\right\vert\,
            s,s_{R},\Zdot,\Xobs
            \,\right]
        \;\; \approx \;\;
            0,
        \quad
        \textnormal{for each \,$k \in s_{M}$}
        \end{equation*}
        See the paragraph after Equation (5.2), p.175, \cite{Beaumont2011}.
        \vskip 0.3cm
    \item
        The second summand
        \begin{equation*}
        2
        \cdot
        \left(\,
            \dfrac{1}{T_{d,Z} \slash N}
            \,\right)
        \left(\,
            \dfrac{
                {\color{white}.}
                {\color{red}T_{d,Z}} \,-\, \widehat{T}_{d}(s\,;\zdot)
                {\color{white}.}
                }{
                \overset{{\color{white}.}}{
                    N
                    }
                }           
            \,\right)
        \cdot
        \dfrac{
            1
            }{
            \overset{{\color{white}.}}{
                \left. \widehat{T}_{d}(s\,;\zdot)^{2} \right\slash N^{2}
                }
            }
        \cdot
        \left(\,
            \dfrac{
                \underset{l \in s_{R}}{\sum}\,
                d_{l}\,W^{(+)}_{l}\,\sigma_{l}^{2}
                \;\;-\;
                \underset{k \in s_{M}}{\sum}\,
                d_{k}\,w_{k}\,\sigma_{k}^{2}
                }{
                \overset{{\color{white}.}}{N^{2}}                
                }
            \,\right)
        \end{equation*}
        in the expression for the sampling-nonresponse covariance component
        would also probably have to be disregarded in practice,
        due to the presence of the term \,$T_{d,Z}$
        (domain total of the population characteristic $\Zdot$),\,
        which is usually unknown in practice.
        \vskip 0.1cm
        In order to justify the omission of the second summand,
        one could attempt to find (practically usefull) ``regularity conditions"
        that are sufficient to force the second summand to be
        \,$O_{p}\!\left(\,1/n^{\delta}\,\right)$,\,
        for some \,$\delta > 0$,\,
        where \,$n$\, here denotes the sample size.
        We may not be too far from such a sufficent collection of regularity conditions.
        Recall that
        \begin{itemize}
        \item
            we have the general asymptotic result for the Horvitz-Thompson domain total estimator:
            \begin{equation*}
            \dfrac{
                {\color{white}.}
                \widehat{T}_{d}(s\,;\zdot) \,-\, {\color{black}T_{d,Z}}
                {\color{white}.}
                }{
                \overset{{\color{white}.}}{
                    N
                    }
                }
            \;\; = \;\;
                O_{p}\!\left(\dfrac{1}{\sqrt{\textnormal{\,sample size\,}}}\right),
            \quad
            \textnormal{under suitable regularity conditions on \,$\Zdot$,\, and}
            \end{equation*}
        \item
            $T_{d,Z} \slash N$\, being asymptotically bounded away from zero
            is one of the regularity conditions often assumed in order to establish
            consistency for the Horvitz-Thomposon ratio estimator.
        \end{itemize}
        Hence, it may be sufficient provided one could find
        a ``suitable'' (practically applicable) regularity condition on
        \,$W^{(+)}_{l}$,\, $\sigma^{2}_{k}$,\, and \,$w_{k} = 1/\pi_{k}$\,
        such that
        \begin{equation*}
        \dfrac{
            {\color{white}.}
            \underset{l \in s_{R}}{\sum}\,
            d_{l}\,W^{(+)}_{l}\,\sigma_{l}^{2}
            \;\,-\,
            \underset{k \in s_{M}}{\sum}\,
            d_{k}\,w_{k}\,\sigma_{k}^{2}
            {\color{white}.}
            }{
            \overset{{\color{white}.}}{N^{2}}                
            }
        \;\; = \;\;
            O_{p}\!\left(\,1\,\right)
        \end{equation*}
        If such a regularity condition is available,
        we should be able to invoke it, in conjuction with the other two more standard regularity conditions above,
        so as to conclude that
        \begin{equation*}
        2
        \cdot
        \left(\,
            \dfrac{1}{T_{d,Z} \slash N}
            \,\right)
        \left(\,
            \dfrac{
                {\color{white}.}
                {\color{black}T_{d,Z}} \,-\, \widehat{T}_{d}(s\,;\zdot)
                {\color{white}.}
                }{
                \overset{{\color{white}.}}{
                    N
                    }
                }           
            \,\right)
        \cdot
        \dfrac{
            1
            }{
            \overset{{\color{white}.}}{
                \left. \widehat{T}_{d}(s\,;\zdot)^{2} \right\slash N^{2}
                }
            }
        \cdot
        \left(\,
            \dfrac{
                \underset{l \in s_{R}}{\sum}\,
                d_{l}\,W^{(+)}_{l}\,\sigma_{l}^{2}
                \;\;-\;
                \underset{k \in s_{M}}{\sum}\,
                d_{k}\,w_{k}\,\sigma_{k}^{2}
                }{
                \overset{{\color{white}.}}{N^{2}}                
                }
            \,\right)
        \;\; = \;\;
            O_{p}\!\left(\,\dfrac{1}{n^{\delta}}\,\right),
        \end{equation*}
        for some \,$\delta > 0$,\,
        where \,$n$\, denotes the sample size.
        Such an asymptotic result would suffice to justify disregarding the second summand,
        when estimating sampling-nonresponse covariance component in practice.
    \end{itemize}
\end{itemize}
